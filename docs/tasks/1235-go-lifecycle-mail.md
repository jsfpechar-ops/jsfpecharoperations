## 1. Objective

Add the new Go package `go/internal/lifecyclemail/` that does what `App/app/lifecycle_mail.py` `run_daily` does (candidates, claim, queue rows, once-a-day gate), with `mail_notify.build_lifecycle` and `workspace_contact_emails`. A Python golden script writes a fixture database and the expected rows and bytes; the Go test loads the same fixture into Postgres and compares.

## 2. Context

Plan: [go-mail-job](../plans/go-mail-job.md) §3 and §4 (task 1235). The package is a library; the job wiring is task 1236. Nothing else imports it yet.

Exported API (for 1236): `ConfigFromEnv(getenv func(string) string) Config`, `RunDaily(ctx, pool *pgxpool.Pool, cfg Config, now time.Time) (map[string]int, error)` (empty map when it did nothing), `Config.BuildLifecycle`, `Config.UnsubscribeURL`, `UnsubscribeToken`, `EmailHash`, `WorkspaceContactEmails`, `IsOptedOut`, `Config.IsSuppressed`, `Candidates`, `LastRunKey`.

Parity notes the reviewer should know:
- Queue rows go through `ubyport.EnqueueMail` (read-only package) after the guards of `mail.enqueue` (backend console or ses, `NormaliseEmail`, non-empty). The payload is written by hand as `json.dumps` would: keys text, html, lang, reply_to, list_unsubscribe, default separators, ensure_ascii (`ubyport.PyString`).
- The unsubscribe token is an itsdangerous `URLSafeSerializer` (no timestamp, salt `ubyhost-lifecycle-unsubscribe`, django-concat key, HMAC-SHA1). Python zlib-compresses the JSON when that is shorter, which a hex hash always is. Go's `compress/zlib` writes different bytes than C zlib for the same JSON (often longer, so Go then keeps the plain JSON, as itsdangerous does when compression does not help). So a Go token is not byte-identical to a Python token for a hashed address; with an empty hash both are plain and identical. Every Go token verifies in Python (`URLSafeSerializer.loads` was run on Go tokens, compressed and plain). The test therefore strips tokens before the byte-exact payload compare, and decodes each of the 3 tokens per row (text, HTML, list_unsubscribe) on both sides and compares the signed JSON. Porting C zlib's deflate bit for bit was judged not worth ~500 lines for an opaque link token. All other output (subject, text, HTML, payload apart from the token, SQL effects) is byte-exact.
- Insert race on the outbox key: `ubyport.EnqueueMail` returns the winner's id (counts as queued); Python logs and returns None. Only matters when two runtimes run the step at once.
- Only English is ported (`HOST_MAIL_LANGUAGE` is `en`); another language returns `ErrLanguage`.
- `ConfigFromEnv` reads the secret like `config._load_secret` but never creates one: empty secret makes `RunDaily` return an error before it writes the day marker.
- An unset and an empty `UBYHOST_PUBLIC_BASE_URL` both give the Python default `http://127.0.0.1:8080` (Python keeps an empty one).
- Errors and logs carry no e-mail address, body or secret.

Env the job reads: `UBYHOST_LIFECYCLE_MAIL` (1/true/yes), `UBYHOST_MAIL_BACKEND` (console|ses), `UBYHOST_OPERATOR_NAME`, `_ICO`, `_ADDRESS`, `UBYHOST_OPERATOR_EMAIL` (reply-to, default support@ubyhost.com), `UBYHOST_PUBLIC_BASE_URL`, `UBYHOST_APP_BASE_URL`, `UBYHOST_SECRET_KEY` or `$UBYHOST_DATA_DIR/secret_key`. The time zone is fixed Europe/Prague as in `config.TIMEZONE`.

	return c.OperatorName != "" && c.OperatorICO != "" && c.OperatorAddress != ""
}

package lifecyclemail

import (
	"bytes"
	"compress/zlib"
	"crypto/hmac"
	"crypto/sha1"
	"crypto/sha256"
	"encoding/base64"
	"encoding/hex"
	"encoding/json"

	"github.com/jsfpechar-ops/jsfpecharoperations/go/internal/ubyport"
)

// unsubscribeSalt is the salt lifecycle_mail._serializer passes to itsdangerous.
const unsubscribeSalt = "ubyhost-lifecycle-unsubscribe"

// EmailHash is lifecycle_mail.email_hash: HMAC-SHA256, keyed with the app
// secret, over "mail-suppression\0" and the normalised address, as lower-case
// hex. "" for a non-address.
func EmailHash(secret, address string) string {
	normalised := ubyport.NormaliseEmail(address)
	if normalised == "" {
		return ""
	}
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte("mail-suppression\x00" + normalised))
	return hex.EncodeToString(mac.Sum(nil))
}

// tokenPayload is {"u": account id, "h": address hash}; field order is the
// key order of the Python dict.
type tokenPayload struct {
	U int64  `json:"u"`
	H string `json:"h"`
}

// UnsubscribeToken is lifecycle_mail.unsubscribe_token: an itsdangerous
// URLSafeSerializer (no timestamp, django-concat key, HMAC-SHA1) over
// {"u": user_id, "h": email_hash(address)}.
//
// Python compresses the JSON with zlib when that is shorter, and a hex hash
// always is. Go's deflate may pick different bytes than zlib for the same
// input, so a token can differ from Python's while still verifying there:
// the signature covers whatever bytes are written and zlib reads both.
func UnsubscribeToken(secret string, userID int64, address string) (string, error) {
	var jb bytes.Buffer
	enc := json.NewEncoder(&jb)
	enc.SetEscapeHTML(false)
	if err := enc.Encode(tokenPayload{U: userID, H: EmailHash(secret, address)}); err != nil {
		return "", err
	}
	payload := bytes.TrimSuffix(jb.Bytes(), []byte("\n"))
	prefix := ""
	var zb bytes.Buffer
	w, err := zlib.NewWriterLevel(&zb, zlib.DefaultCompression)
	if err != nil {
		return "", err
	}
	if _, err := w.Write(payload); err != nil {
		return "", err
	}
	if err := w.Close(); err != nil {
		return "", err
	}
	if zb.Len() < len(payload)-1 {
		payload = zb.Bytes()
		prefix = "."
	}
	value := prefix + base64.RawURLEncoding.EncodeToString(payload)
	sig, err := signValue(secret, value)
	if err != nil {
		return "", err
	}
	return value + "." + sig, nil
}

// UnsubscribeURL is lifecycle_mail.unsubscribe_url. It uses the public base
// URL, not the app one.
func (c Config) UnsubscribeURL(userID int64, address, lang string) (string, error) {
	token, err := UnsubscribeToken(c.SecretKey, userID, address)
	if err != nil {
		return "", err
	}
	return c.PublicBaseURL + "/mail/unsubscribe/" + token + "?lang=" + lang, nil
}

// signValue is itsdangerous Signer.get_signature with the django-concat key.
func signValue(secret, value string) (string, error) {
	derived := sha1.Sum([]byte(unsubscribeSalt + "signer" + secret))
	mac := hmac.New(sha1.New, derived[:])
	mac.Write([]byte(value))
	return base64.RawURLEncoding.EncodeToString(mac.Sum(nil)), nil
}

	links map[string]link

// parts is _parts: re.split alternates text and marker names.
func (f footerSentence) parts() (texts []string, marks []link) {
	last := 0
	for _, m := range markerRE.FindAllStringSubmatchIndex(f.text, -1) {
		texts = append(texts, f.text[last:m[0]])
		marks = append(marks, f.links[f.text[m[2]:m[3]]])
		last = m[1]
	}
	texts = append(texts, f.text[last:])
	return texts, marks
}

func (f footerSentence) plain() string {
	texts, marks := f.parts()
	var b strings.Builder
	for i, t := range texts {
		b.WriteString(t)
		if i < len(marks) {
			if marks[i].label != "" {
				b.WriteString(marks[i].label + " (" + marks[i].href + ")")
			} else {
				b.WriteString(marks[i].href)

		Text:    text,

// iso is _iso: UTC, whole seconds, "+00:00".

// WorkspaceContactEmails is mail_notify.workspace_contact_emails: every
// distinct normalised contact address on the workspace's legal entities.
func WorkspaceContactEmails(ctx context.Context, pool *pgxpool.Pool, ownerUserID int64) ([]string, error) {
	rows, err := pool.Query(ctx,
		`SELECT contact_email FROM legal_entity WHERE owner_user_id = $1 ORDER BY id`, ownerUserID)
	if err != nil {
		return nil, errDB
	}
	defer rows.Close()
	found := []string{}
	for rows.Next() {
		var raw *string
		if err := rows.Scan(&raw); err != nil {
			return nil, errDB
		}
		if raw == nil {
			continue
		}
		address := ubyport.NormaliseEmail(*raw)
		if address == "" {
			continue
		}
		if !slices.Contains(found, address) {
			found = append(found, address)
		}
	}
	if err := rows.Err(); err != nil {
		return nil, errDB
	}
	return found, nil
}

// IsSuppressed is lifecycle_mail.is_suppressed.
func (c Config) IsSuppressed(ctx context.Context, pool *pgxpool.Pool, address string) (bool, error) {
	digest := EmailHash(c.SecretKey, address)

// Candidates is lifecycle_mail.candidates: host accounts due for kind at now.
func Candidates(ctx context.Context, pool *pgxpool.Pool, kind string, now time.Time) ([]int64, error) {
	r, ok := rules[kind]
	if !ok {
		return nil, fmt.Errorf("%w %s", ErrKind, kind)
	}
	days := 0
	for _, t := range triggers {
		if t.kind == kind {
			days = t.days
		}
	}
	dueBefore := iso(now.Add(-time.Duration(days) * 24 * time.Hour))
	tooLateBefore := iso(now.Add(-time.Duration(days+maxLateDays) * 24 * time.Hour))
	rows, err := pool.Query(ctx,
		"SELECT u.id FROM user_account u WHERE u.role = 'host' AND u.active = 1 "+
			"AND u.deletion_due_at IS NULL AND COALESCE(u.onboarding_emails_opt_out, 0) = 0 "+
			"AND NOT EXISTS (SELECT 1 FROM lifecycle_mail_sent m "+
			"WHERE m.user_account_id = u.id AND m.kind = $1) "+
			"AND "+r.moot+" "+
			"AND "+r.anchor+" <= $2 AND "+r.anchor+" >= $3 "+
			"ORDER BY u.id LIMIT 500",
		kind, dueBefore, tooLateBefore)
	if err != nil {
		return nil, errDB
	}
	defer rows.Close()
	var ids []int64
	for rows.Next() {
		var id int64
		if err := rows.Scan(&id); err != nil {
			return nil, errDB
		}
		ids = append(ids, id)
	}
	if err := rows.Err(); err != nil {
		return nil, errDB
	}
	return ids, nil
}

// claim is _claim: write the sent marker first; false when a run already did.
func claim(ctx context.Context, pool *pgxpool.Pool, userID int64, kind string) (bool, error) {
	tag, err := pool.Exec(ctx,
		`INSERT INTO lifecycle_mail_sent (user_account_id, kind, sent_at) VALUES ($1, $2, $3)
		 ON CONFLICT (user_account_id, kind) DO NOTHING`, userID, kind, ubyport.UTCNow())
	if err != nil {
		return false, errDB
	}
	return tag.RowsAffected() == 1, nil
}

// release is _release.
func release(ctx context.Context, pool *pgxpool.Pool, userID int64, kind string) error {
	if _, err := pool.Exec(ctx,
		`DELETE FROM lifecycle_mail_sent WHERE user_account_id = $1 AND kind = $2`, userID, kind); err != nil {
		return errDB
	}
	return nil
}

// pyPayload is json.dumps of the tip's payload: text, html, lang, reply_to,
// list_unsubscribe, in that order, default separators, ensure_ascii.
func pyPayload(text, html, lang, replyTo, listUnsubscribe string) string {
	return `{"text": ` + ubyport.PyString(text) +
		`, "html": ` + ubyport.PyString(html) +
		`, "lang": ` + ubyport.PyString(lang) +
		`, "reply_to": ` + ubyport.PyString(replyTo) +
		`, "list_unsubscribe": ` + ubyport.PyString(listUnsubscribe) + `}`
}

// enqueue is mail.enqueue for a lifecycle kind: nothing when the backend is
// off or the address does not normalise, otherwise the outbox row (an
// existing idempotency key returns the row already there). It reports whether
// a row exists.
func (c Config) enqueue(ctx context.Context, pool *pgxpool.Pool, kind, key, to, subject, payload string, ownerUserID int64) bool {
	if !c.mailEnabled() {
		return false
	}
	to = ubyport.NormaliseEmail(to)
	if to == "" {
		return false
	}
	owner := ownerUserID
	id, err := ubyport.EnqueueMail(ctx, pool, ubyport.OutboxMail{
		Kind:        kind,
		Key:         key,
		OwnerUserID: &owner,
		To:          to,
		Subject:     subject,
		Payload:     payload,
	})
	if err != nil {
		slog.Error("lifecycle mail enqueue failed", "kind", kind, "user", ownerUserID)
		return false
	}
	return id != 0
}

// sendOne is send_one: queue kind to every contact address of the account.
func (c Config) sendOne(ctx context.Context, pool *pgxpool.Pool, userID int64, kind string) (int, error) {
	opted, err := IsOptedOut(ctx, pool, userID)
	if err != nil || opted {
		return 0, err
	}
	contacts, err := WorkspaceContactEmails(ctx, pool, userID)
	if err != nil {
		return 0, err
	}
	var addresses []string
	for _, address := range contacts {
		suppressed, err := c.IsSuppressed(ctx, pool, address)
		if err != nil {
			return 0, err
		}
		if !suppressed {
			addresses = append(addresses, address)
		}
	}
	if len(addresses) == 0 {
		// Nowhere to send yet. No marker, so a later run can still reach them.
		return 0, nil
	}
	won, err := claim(ctx, pool, userID, kind)
	if err != nil || !won {
		return 0, err
	}
	queued := 0
	for _, address := range addresses {
		link, err := c.UnsubscribeURL(userID, address, hostMailLanguage)
		if err != nil {
			return queued, errDB
		}
		content, err := c.BuildLifecycle(kind, link, hostMailLanguage)
		if err != nil {
			return queued, err
		}
		payload := pyPayload(content.Text, content.HTML, hostMailLanguage, c.OperatorEmail, link)
		if c.enqueue(ctx, pool, kind, fmt.Sprintf("lifecycle:%s:%d:%s", kind, userID, address),
			address, content.Subject, payload, userID) {
			queued++
		}
	}
	if queued == 0 {
		if err := release(ctx, pool, userID, kind); err != nil {
			return 0, err
		}
	}
	return queued, nil
}

// run is run(): evaluate every trigger once. Empty while the switch is off.
func (c Config) run(ctx context.Context, pool *pgxpool.Pool, now time.Time) (map[string]int, error) {
	summary := map[string]int{}
	if !c.LifecycleMail || !c.mailEnabled() {
		return summary, nil
	}
	if !c.senderIdentified() {
		slog.Warn("lifecycle mail skipped: UBYHOST_OPERATOR_NAME, _ICO or _ADDRESS is empty")
		return summary, nil
	}
	if strings.TrimSpace(c.SecretKey) == "" {
		return summary, errors.New("lifecyclemail: no secret key (UBYHOST_SECRET_KEY or the key file)")
	}
	for _, t := range triggers {
		summary[t.kind] = 0
	}
	for _, t := range triggers {
		ids, err := Candidates(ctx, pool, t.kind, now)
		if err != nil {
			return summary, err
		}
		for _, userID := range ids {
			n, err := c.sendOne(ctx, pool, userID, t.kind)
			if err != nil {
				slog.Error("lifecycle mail failed", "kind", t.kind, "user", userID, "err", err.Error())
				continue
			}
			if n > 0 {
				summary[t.kind]++
			}
		}
	}
	return summary, nil
}

// RunDaily is run_daily: the mail job's step. At most one run per local day,
// after 09:00 Europe/Prague. It returns an empty map when it did nothing.
func RunDaily(ctx context.Context, pool *pgxpool.Pool, cfg Config, now time.Time) (map[string]int, error) {
	if !cfg.LifecycleMail || !cfg.mailEnabled() {
		return map[string]int{}, nil
	}
	loc, err := time.LoadLocation(Timezone)
	if err != nil {
		return nil, errors.New("lifecyclemail: time zone unavailable")
	}
	local := now.In(loc)
	today := local.Format("2006-01-02")
	if local.Hour() < sendHourLocal {
		return map[string]int{}, nil
	}
	last, _, err := getSetting(ctx, pool, LastRunKey)
	if err != nil {
		return nil, err
	}
	if last == today {
		return map[string]int{}, nil
	}
	if cfg.senderIdentified() && strings.TrimSpace(cfg.SecretKey) == "" {
		// Before the marker, so a missing key does not use up the day.
		return nil, errors.New("lifecyclemail: no secret key (UBYHOST_SECRET_KEY or the key file)")
	}
	if err := setSetting(ctx, pool, LastRunKey, today); err != nil {
		return nil, err
	}
	return cfg.run(ctx, pool, now)
}
