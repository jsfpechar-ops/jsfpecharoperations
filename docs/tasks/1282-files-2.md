// readOnlyOpenPrefixes and readOnlyOpenPaths are auth.py
// _READ_ONLY_OPEN_PREFIXES and _READ_ONLY_OPEN_PATHS: a paused plan may still post there.
var (
	readOnlyOpenPrefixes = []string{"/predplatne", "/logout", "/account", "/settings", "/exports"}
	readOnlyOpenPaths    = regexp.MustCompile(`^/(reservations/submit-ready|reservations/\d+/submit|guests/\d+/resend)$`)
)

func readOnlyOpen(path string) bool {
	for _, p := range readOnlyOpenPrefixes {
		if strings.HasPrefix(path, p) {
			return true
		}
	}
	return readOnlyOpenPaths.MatchString(path)
}

// openPost is the method Login sees for a POST to a read-only-open path. It
// is neither POST (so Login skips the paused check) nor GET or HEAD (so
// http.Redirect writes no HTML body, as RedirectResponse does not).
const openPost = "POST-READ-ONLY-OPEN"

// login is auth.require_login. hostweb.Login checks the paused plan on every
// POST; Python skips that check on the read-only-open paths, so Login sees a
// copy of the request with method openPost there. ok is false when the
// response is written.
func (d deps) login(w http.ResponseWriter, r *http.Request, at time.Time) (*session.Host, bool) {
	req := r
	if r.Method == http.MethodPost && readOnlyOpen(r.URL.Path) {
		req = r.WithContext(r.Context())
		req.Method = openPost
	}
	h, err := hostweb.Login(bare{w}, req, d.pool, d.getenv, at)
	if err != nil {
		serverError(w, err)
		return nil, false
	}
	return h, h != nil
}

// bare drops what http.Redirect adds to a GET redirect (Content-Type and a
// small HTML body): Starlette's RedirectResponse sends neither.
type bare struct{ http.ResponseWriter }

func (b bare) WriteHeader(code int) {
	if code >= 300 && code < 400 {
		b.Header()["Content-Type"] = nil
	}
	b.ResponseWriter.WriteHeader(code)
}

func (b bare) Write(p []byte) (int, error) {
	if loc := b.Header().Get("Location"); loc != "" && b.Header()["Content-Type"] == nil {
		return len(p), nil
	}
	return b.ResponseWriter.Write(p)
}

// current is auth.current_user (with workspace_user): the signed-in host or
// nil, without the require_login gates. ok is false after a 500.
func (d deps) current(w http.ResponseWriter, r *http.Request, at time.Time) (*session.Host, bool) {
	h, err := session.Load(r.Context(), d.pool, hostweb.Cookie(r, session.CookieName), at, d.getenv)
	if err != nil {
		serverError(w, err)
		return nil, false
	}
	return h, true
}

// form parses the body once; hostweb.CSRF may already have done it.
func form(r *http.Request) {
	if r.PostForm == nil {
		_ = r.ParseForm()
	}
}

// formLast is Starlette FormData.get: the last value of a repeated key.
func formLast(r *http.Request, key string) (string, bool) {
	form(r)
	vals, ok := r.PostForm[key]
	if !ok || len(vals) == 0 {
		return "", false
	}
	return vals[len(vals)-1], true
}

// formStr is admin_helpers.form_str: the stripped last value, or def.
func formStr(r *http.Request, key, def string) string {
	v, ok := formLast(r, key)
	if !ok || v == "" {
		return pyStrip(def)
	}
	return pyStrip(v)
}

// formHas is `key in form`.
func formHas(r *http.Request, key string) bool {
	form(r)
	_, ok := r.PostForm[key]
	return ok
}

// pySpace is str.isspace for one rune: Go's set plus U+001C..U+001F.
func pySpace(c rune) bool { return unicode.IsSpace(c) || (c >= 0x1c && c <= 0x1f) }

// pyStrip is str.strip().
func pyStrip(s string) string { return strings.TrimFunc(s, pySpace) }

// pyIsDigit is str.isdigit for ASCII input; any other rune reports false.
func pyIsDigit(s string) bool {
	if s == "" {
		return false
	}
	for _, c := range s {
		if c < '0' || c > '9' {
			return false
		}
	}
	return true
}

type validationItem struct {

	Loc   []string `json:"loc"`
	Msg   string   `json:"msg"`

// pathID is FastAPI's int path parameter: a 422 with pydantic's int_parsing
// detail when the segment is not an integer.
func pathID(w http.ResponseWriter, r *http.Request, name string) (int64, bool) {
	raw := r.PathValue("id")
	if n, err := strconv.ParseInt(strings.TrimSpace(raw), 10, 64); err == nil {
		return n, true
	}
	writeJSON(w, http.StatusUnprocessableEntity, map[string][]validationItem{"detail": {{
		Type: "int_parsing", Loc: []string{"path", name},
		Msg: "Input should be a valid integer, unable to parse string as an integer", Input: raw,
	}}})
	return 0, false
}

// writeJSON is Starlette's JSONResponse: compact, ensure_ascii=False.
func writeJSON(w http.ResponseWriter, status int, body any) {
	var b strings.Builder
	enc := json.NewEncoder(&b)
	enc.SetEscapeHTML(false)
	_ = enc.Encode(body)
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_, _ = w.Write([]byte(strings.TrimSuffix(b.String(), "\n")))
}

// httpError is an HTTPException(status) with its default detail.
func httpError(w http.ResponseWriter, status int) {
	writeJSON(w, status, map[string]string{"detail": http.StatusText(status)})
}

// emptyStatus is Starlette's Response(status_code=status): no body, no Content-Type.
func emptyStatus(w http.ResponseWriter, status int) {
	w.Header()["Content-Type"] = nil
	w.WriteHeader(status)
}

// redirect is RedirectResponse(location, 303) with the location kept as given.

	w.Header().Set("Location", location)
	w.WriteHeader(http.StatusSeeOther)
}

// audit is db.audit. Without an explicit owner the row takes the workspace
// require_login set; actor_user_id and impersonator_user_id come from it too.
func (d deps) audit(ctx context.Context, g *gated, at time.Time, action, detail, actor string, owner *int64) error {
	var actorID, impersonator *int64
	if g != nil {
		id := g.h.Account.ID
		actorID = &id
		if g.h.Impersonating {
			impersonator = &id
		}
		if owner == nil {
			ws := g.h.Workspace.ID
			owner = &ws
		}
	}
	_, err := d.pool.Exec(ctx, `INSERT INTO audit (at, actor, action, detail, owner_user_id, actor_user_id, impersonator_user_id)
		VALUES ($1, $2, $3, $4, $5, $6, $7)`, session.UTCNow(at), actor, action, detail, owner, actorID, impersonator)
	return err
}

func ptr(v int64) *int64 { return &v }

// remembers is auth.session_remembers: bool(payload.get("rm")).
func remembers(p map[string]any) bool {
	switch v := p["rm"].(type) {
	case nil:
		return false
	case bool:
		return v
	case int64:
		return v != 0
	case float64:
		return v != 0
	case string:
		return v != ""
	}
	return true
}

// clearCookie is response.delete_cookie(name, path="/", secure, httponly, samesite="lax").
func clearCookie(name string, secure bool) *http.Cookie {
	return &http.Cookie{Name: name, Value: "", Path: "/", MaxAge: -1, Expires: time.Unix(0, 0),
		HttpOnly: true, Secure: secure, SameSite: http.SameSiteLaxMode}
}

// attach is auth.attach_session for a fresh token.
func (d deps) attach(w http.ResponseWriter, userID, sv int64, opt session.IssueOptions, at time.Time) error {
	secret, err := secure.LoadSecret(d.getenv)
	if err != nil {
		return err
	}
	token, err := session.IssueSession(secret, userID, sv, opt, at)
	if err != nil {
		return err
	}
	http.SetCookie(w, session.SessionCookie(token, opt.Remember, hostweb.SecureCookies(d.getenv)))
	return nil
}

// unsuppress is the second half of lifecycle_mail.resubscribe: drop the
// onboarding suppression of every contact address of the workspace.
func (d deps) unsuppress(r *http.Request, userID int64) error {
	ctx := r.Context()
	secret, err := secure.LoadSecret(d.getenv)
	if err != nil {
		return err
	}
	rows, err := d.pool.Query(ctx, `SELECT contact_email FROM legal_entity WHERE owner_user_id = $1 ORDER BY id`, userID)
	if err != nil {
		return err
	}
	var found []string
	for rows.Next() {
		var raw *string
		if err := rows.Scan(&raw); err != nil {
			rows.Close()
			return err
		}
		if raw == nil {
			continue
		}
		if addr := normaliseEmail(*raw); addr != "" && !contains(found, addr) {
			found = append(found, addr)
		}
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return err
	}
	for _, addr := range found {
		if _, err := d.pool.Exec(ctx, `DELETE FROM mail_suppression WHERE email_hash = $1 AND scope = $2`,
			EmailHash(secret, addr), suppressScope); err != nil {
			return err
		}
	}
	return nil
}

func contains(list []string, s string) bool {
	for _, v := range list {
		if v == s {
			return true
		}
	}
	return false
}

// EmailHash is lifecycle_mail.email_hash for a normalised address.
func EmailHash(secret, normalised string) string {
	m := hmac.New(sha256.New, []byte(secret))
	m.Write([]byte("mail-suppression\x00" + normalised))
	return hex.EncodeToString(m.Sum(nil))
}

var emailRE = regexp.MustCompile(`^[^@\s]+@[^@\s]+\.[^@\s]+$`)

// normaliseEmail is mail.normalise_email for the two forms a contact address
// takes: a bare address or "Name <addr>" (as ubyport.normaliseEmail).
func normaliseEmail(raw string) string {
	s := pyStrip(raw)
	if strings.ContainsAny(s, ",\"") || strings.Count(s, "<") > 1 || strings.Count(s, ">") > 1 {
		return ""
	}
	addr := s
	if i := strings.Index(s, "<"); i >= 0 {
		j := strings.Index(s, ">")
		if j < i || pyStrip(s[j+1:]) != "" {
			return ""
		}
		addr = s[i+1 : j]
	}
	addr = strings.ToLower(pyStrip(addr))
	if addr == "" || !emailRE.MatchString(addr) || utf8.RuneCountInString(addr) > 254 {
		return ""
	}
	return addr
}

// secondFactor is the shared body of two_factor_move and two_factor_disable,
// up to and including reset_totp. done is true when the response is written.
func (d deps) secondFactor(w http.ResponseWriter, r *http.Request, notEnabled, failPath, failAction string) (h *session.Host, account *session.LoginAccount, done bool) {
	at := d.now()
	if !hostweb.CSRF(w, r, d.getenv, at) {
		return nil, nil, true
	}
	h, ok := d.login(w, r, at)
	if !ok {
		return nil, nil, true
	}
	ctx := r.Context()
	account, err := session.AccountByID(ctx, d.pool, h.Account.ID)
	if err != nil || account == nil {
		serverError(w, errOr(err))
		return nil, nil, true
	}
	if account.TOTPEnabled == 0 {
		hostweb.Back(w, r, notEnabled, none, none)
		return nil, nil, true
	}
	blocked, err := session.Account2FABlocked(ctx, d.pool, account.ID, at)
	if err != nil {
		serverError(w, err)
		return nil, nil, true
	}
	if blocked {
		hostweb.Back(w, r, failPath, none, flashFor(r, "auth.error.code_locked"))
		return nil, nil, true
	}
	good, err := session.VerifySecondFactor(ctx, d.pool, d.getenv, account, formStr(r, "code", ""), at)
	if err != nil {
		serverError(w, err)
		return nil, nil, true
	}
	if !good {
		if err := session.RecordAccount2FAFailure(ctx, d.pool, account.ID, at); err != nil {
			serverError(w, err)
			return nil, nil, true
		}
		if err := d.audit(ctx, &gated{h}, at, failAction, "", account.Username, ptr(account.ID)); err != nil {
			serverError(w, err)
			return nil, nil, true
		}
		hostweb.Back(w, r, failPath, none, flashFor(r, "auth.error.code_invalid"))
		return nil, nil, true
	}
	if err := session.ResetTOTP(ctx, d.pool, account.ID); err != nil {
		serverError(w, err)
		return nil, nil, true
	}
	refreshed, err := session.AccountByID(ctx, d.pool, account.ID)
	if err != nil || refreshed == nil {
		serverError(w, errOr(err))
		return nil, nil, true
	}
	return h, refreshed, false
}

// twoFactorMove is two_factor_move: the old phone proves itself, setup runs
// again; the new cookie counts as a fresh login (no issued_at).
func (d deps) twoFactorMove(w http.ResponseWriter, r *http.Request) {
	h, refreshed, done := d.secondFactor(w, r, "/account/2fa/setup", "/settings", "two_factor_move_failed")
	if done {
		return
	}
	at := d.now()
	if err := d.audit(r.Context(), &gated{h}, at, "two_factor_moved", "", refreshed.Username, ptr(refreshed.ID)); err != nil {
		serverError(w, err)
		return
	}
	remember := remembers(h.Payload)
	if err := d.attach(w, refreshed.ID, refreshed.SessionVersion, session.IssueOptions{Remember: remember}, at); err != nil {
		serverError(w, err)
		return
	}
	redirect(w, "/account/2fa/setup?moved=1")
}

// twoFactorDisable is two_factor_disable; the new cookie keeps the login time.
func (d deps) twoFactorDisable(w http.ResponseWriter, r *http.Request) {
	h, refreshed, done := d.secondFactor(w, r, securityPath, securityPath, "two_factor_disable_failed")
	if done {
		return
	}
	at := d.now()
	if err := d.audit(r.Context(), &gated{h}, at, "two_factor_disabled", "", refreshed.Username, ptr(refreshed.ID)); err != nil {
		serverError(w, err)
		return
	}
	remember := remembers(h.Payload)
	iat := session.IssuedAt(h.Payload)
	if err := d.attach(w, refreshed.ID, refreshed.SessionVersion, session.IssueOptions{Remember: remember, IssuedAt: &iat}, at); err != nil {
		serverError(w, err)
		return
	}
	hostweb.Back(w, r, securityPath, flashFor(r, "flash.accounts.twofa_disabled"), none)
}

// securityPromptSeen is passkeys.security_prompt_seen: 401 signed out, else 204.
func (d deps) securityPromptSeen(w http.ResponseWriter, r *http.Request) {
	at := d.now()
	if !hostweb.CSRF(w, r, d.getenv, at) {
		return
	}
	h, ok := d.current(w, r, at)
	if !ok {
		return
	}
	if h == nil {
		emptyStatus(w, http.StatusUnauthorized)
		return
	}
	if _, err := d.pool.Exec(r.Context(), `UPDATE user_account SET two_factor_prompted_at = $1 WHERE id = $2 AND two_factor_prompted_at IS NULL`,
		session.UTCNow(at), h.Account.ID); err != nil {
		serverError(w, err)
		return
	}
	emptyStatus(w, http.StatusNoContent)
}
