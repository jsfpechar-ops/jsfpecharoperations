"""Brief 1700e: the ČÚZK credit on /legal equals the Go constant, once."""
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]


def test_legal_page_carries_the_go_credit_line():
    go = (ROOT / "go/internal/ruian/attribution.go").read_text(encoding="utf-8")
    credit = re.search(r'const Attribution = "([^"]+)"', go).group(1)
    page = (ROOT / "App/app/templates/legal.html").read_text(encoding="utf-8")
    assert page.count(credit) == 1
