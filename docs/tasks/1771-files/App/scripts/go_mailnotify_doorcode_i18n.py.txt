"""Dump the i18n strings the Go door-code mail port needs, for every language.

Run from App/ with the project venv:

    /var/tmp/ubyvenv/bin/python scripts/go_mailnotify_doorcode_i18n.py

Writes go/internal/mailnotify/doorcode_i18n_strings.json (embedded with go:embed by
go/internal/mailnotify/doorcode_i18n.go). It is a second dump next to
i18n_strings.json, which is left untouched. Only the keys the door-code builders read
are dumped; the host notice keys are taken by prefix (every ``mail.door_code_notice.*``
and ``mail.door_code_reason.*`` key), so a new variant needs no edit here.
A language that lacks a key is left out of its table on purpose: the Go lookup then
falls back to English exactly as ``host_i18n.lookup`` and ``i18n.translator`` do.
"""
from __future__ import annotations

import json
import os
import sys

APP_DIR = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))
REPO_DIR = os.path.abspath(os.path.join(APP_DIR, ".."))
OUT = os.path.join(REPO_DIR, "go", "internal", "mailnotify", "doorcode_i18n_strings.json")
sys.path.insert(0, APP_DIR)

from app import host_i18n, i18n  # noqa: E402

# Guest keys read by build_door_code and _fee_block (the ``i18n`` catalogue).
GUEST_KEYS = [
    "mail_door_code_subject",
    "door_code_title",
    "door_code_times",
    "door_code_first_use",
    "mail_completion_fee_heading",
    "mail_completion_fee_total",
    "mail_completion_fee_calc_title",
    "mail_completion_fee_once",
    "mail_completion_fee_already",
    "fee_guest_n",
    "fee_person_line",
    "fee_no_rails",
    "fee_rail_iban",
    "fee_payee",
    "fee_iban",
    "fee_amount",
    "fee_amount_label",
    "fee_vs_label",
    "fee_qr_alt",
    "fee_revolut_button",
    "fee_rail_cash",
    "fee_cash_default",
    "mail_guest_footer_about",
    "mail_guest_footer_why",
    "mail_guest_footer_host_label",
    "mail_guest_footer_help",
    "mail_link_fallback",
]

# Host key prefixes read by _door_code_notice, _door_code_reason, door_code_delayed_notice.
HOST_PREFIXES = ["mail.door_code_notice.", "mail.door_code_reason."]


def _pick_guest(catalogue: dict) -> dict:
    return {lang: {k: catalogue[lang][k] for k in GUEST_KEYS if k in catalogue[lang]}
            for lang in sorted(catalogue)}


def _pick_host(catalogue: dict) -> dict:
    out = {}
    for lang in sorted(catalogue):
        table = catalogue[lang]
        out[lang] = {k: table[k] for k in sorted(table)
                     if any(k.startswith(p) for p in HOST_PREFIXES)}
    return out


def main() -> None:
    doc = {
        "_comment": "Generated by App/scripts/go_mailnotify_doorcode_i18n.py. Do not edit by hand.",
        "guest": _pick_guest(i18n.STRINGS),
        "host": _pick_host(host_i18n.STRINGS),
    }
    with open(OUT, "w", encoding="utf-8") as fh:
        json.dump(doc, fh, ensure_ascii=False, indent=1, sort_keys=True)
        fh.write("\n")
    missing = {
        lang: [k for k in GUEST_KEYS if k not in i18n.STRINGS[lang]]
        for lang in i18n.LANGUAGES
    }
    print("wrote", OUT)
    print("guest keys missing per language:", {k: v for k, v in missing.items() if v})


if __name__ == "__main__":
    main()
