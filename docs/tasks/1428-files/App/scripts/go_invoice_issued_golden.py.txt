def build(case: dict) -> dict:
    content = mail_notify.build_invoice_issued(
        lang=case["lang"],
        property_name=case["issuer"],
        number=case["number"],
        total=case["total"],
        download_url=case["download_url"],
        host=case.get("host"),
        stay_property=case.get("stay_property") or "",
    )
    return {
        "name": case["name"],
        "input": {k: v for k, v in case.items() if k != "name"},
        "subject": content["subject"],
        "text": content["text"],
        "html": content["html"],
    }


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", action="store_true", help="fail when golden is stale")
    args = parser.parse_args()
    cases = [build(c) for c in CASES]
    doc = {
        "_comment": "Generated by App/scripts/go_invoice_issued_golden.py. Do not edit by hand.",
        "public_base_url": PUBLIC_BASE,
        "cases": cases,
    }
    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    if args.check:
        with open(OUT, encoding="utf-8") as f:
            on_disk = json.load(f)
        if on_disk != doc:
            print("stale golden", file=sys.stderr)
            sys.exit(1)
        print("ok")
        return
    with open(OUT, "w", encoding="utf-8") as f:
        json.dump(doc, f, ensure_ascii=False, indent=1)
        f.write("\n")
    print(OUT)


if __name__ == "__main__":
    main()
