"""Dump the guest i18n strings BuildInvoiceIssued needs into go/internal/mail.

Run from App/ with the project venv:

    .venv/bin/python scripts/go_mail_invoice_issued_i18n.py

Writes go/internal/mail/guest_i18n_strings.json (go:embed). Keys come straight from
the guest catalogue; nothing is hand-copied.
"""
from __future__ import annotations

import json
import os
import sys

APP_DIR = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))
REPO_DIR = os.path.abspath(os.path.join(APP_DIR, ".."))
OUT = os.path.join(REPO_DIR, "go", "internal", "mail", "guest_i18n_strings.json")
sys.path.insert(0, APP_DIR)

from app import i18n  # noqa: E402

GUEST_KEYS = [
    "mail_guest_footer_about",
    "mail_guest_footer_why",
    "mail_guest_footer_host_label",
    "mail_guest_footer_help",
    "invoice_mail_issued_subject",
    "invoice_mail_issued_subject_stay",
    "invoice_mail_issued_intro",
    "invoice_mail_issued_button",
    "mail_invoice_title",
    "mail_invoice_number",
    "mail_invoice_property",
    "mail_invoice_total",
]


def _pick(catalogue: dict, keys: list[str]) -> dict:
    out = {}
    for lang in sorted(catalogue):
        table = catalogue[lang]
        picked = {key: table[key] for key in keys if key in table}
        out[lang] = picked
    return out


def main() -> None:
    doc = {
        "_comment": "Generated by App/scripts/go_mail_invoice_issued_i18n.py. Do not edit by hand.",
        "guest": _pick(i18n.STRINGS, GUEST_KEYS),
        "guest_languages": list(i18n.LANGUAGES),
        "guest_default_languages": list(i18n.DEFAULT_GUEST_LANGS.split(",")),
        "guest_default_language": i18n.DEFAULT_LANGUAGE,
    }
    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    with open(OUT, "w", encoding="utf-8") as f:
        json.dump(doc, f, ensure_ascii=False, indent=1)
        f.write("\n")
    print(OUT)


if __name__ == "__main__":
    main()
