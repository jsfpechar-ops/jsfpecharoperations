// Settings is the settings table as db.get_setting and db.set_setting use it.
type Settings interface {

// PoolSettings is Settings on PostgreSQL.
func PoolSettings(pool *pgxpool.Pool) Settings { return poolSettings{pool: pool} }

type poolSettings struct{ pool *pgxpool.Pool }

// Get is db.get_setting (db.py 2110-2112).
func (p poolSettings) Get(ctx context.Context, key string) (string, error) {
	var value *string
	err := p.pool.QueryRow(ctx, `SELECT value FROM settings WHERE key = $1`, key).Scan(&value)
	if errors.Is(err, pgx.ErrNoRows) {
		return "", nil
	}
	if err != nil || value == nil {
		return "", err
	}
	return *value, nil
}

// Set is db.set_setting (db.py 2124-2129).
func (p poolSettings) Set(ctx context.Context, key, value string) error {
	_, err := p.pool.Exec(ctx,
		`INSERT INTO settings (key, value) VALUES ($1, $2) ON CONFLICT (key) DO UPDATE SET value = excluded.value`,
		key, value)
	return err
}

func indexOf(list []string, s string) int {
	for i, v := range list {
		if v == s {
			return i
		}
	}
	return -1
}

// Load is property_setup.load: the stored state, or nil when the row is
// missing, empty, not JSON or not a JSON object.
func Load(ctx context.Context, s Settings, apartmentID int64) (*State, error) {
	raw, err := s.Get(ctx, key(apartmentID))
	if err != nil {
		return nil, err
	}
	return parseState(raw), nil
}

// parseState is the body of property_setup.load after db.get_setting.
func parseState(raw string) *State {
	if raw == "" {
		return nil
	}
	top, ok := decodeAll(raw)
	if !ok {
		return nil
	}
	obj, ok := top.(map[string]any)
	if !ok {
		return nil
	}
	state := &State{
		Steps:      map[string]string{},
		Finished:   truthy(obj["finished"]),
		Hidden:     truthy(obj["hidden"]),
		Legacy:     truthy(obj["legacy"]),
		Celebrated: truthy(obj["celebrated"]),
		Tested:     truthy(obj["tested"]),
	}
	// Python walks steps.items() in document order, a repeated key keeping
	// its first place and its last value; a later alias overwrites an earlier one.
	_, topValues, _ := orderedObject([]byte(raw))
	keys, values, ok := orderedObject(topValues["steps"])
	if !ok {
		return state
	}
	for _, k := range keys {
		var value any
		if err := json.Unmarshal(values[k], &value); err != nil {
			continue
		}
		mark, isString := value.(string)
		step := normaliseStep(k)
		if !isString || !in(KnownSteps, step) || !in(Marks, mark) {
			continue
		}
		if mark == "waiting" && step != "police" {
			continue
		}
		state.Steps[step] = mark
	}
	return state
}

// decodeAll is json.loads: exactly one JSON value, numbers kept as text so
// 1e400 parses as Python's float does. Python's NaN and Infinity literals
// are not JSON and fail here (brief 1290 section 2).
func decodeAll(raw string) (any, bool) {
	dec := json.NewDecoder(strings.NewReader(raw))
	dec.UseNumber()
	var v any
	if err := dec.Decode(&v); err != nil {
		return nil, false
	}
	if _, err := dec.Token(); err != io.EOF {
		return nil, false
	}
	return v, true
}

// orderedObject reads a JSON object's keys in first-seen order and the last
// value of each key, as Python's dict does. ok is false for anything else.
func orderedObject(raw []byte) ([]string, map[string]json.RawMessage, bool) {
	if len(raw) == 0 {
		return nil, nil, false
	}
	dec := json.NewDecoder(strings.NewReader(string(raw)))
	dec.UseNumber()
	tok, err := dec.Token()
	if err != nil || tok != json.Delim('{') {
		return nil, nil, false
	}
	var keys []string
	values := map[string]json.RawMessage{}
	for dec.More() {
		tok, err := dec.Token()
		if err != nil {
			return nil, nil, false
		}
		k, _ := tok.(string)
		var value json.RawMessage
		if err := dec.Decode(&value); err != nil {
			return nil, nil, false
		}
		if _, seen := values[k]; !seen {
			keys = append(keys, k)
		}
		values[k] = value
	}
	return keys, values, true
}

// truthy is Python's bool() of a json.loads value.
func truthy(v any) bool {
	switch t := v.(type) {
	case nil:
		return false
	case bool:
		return t
	case string:
		return t != ""
	case json.Number:
		f, _ := strconv.ParseFloat(string(t), 64)
		return f != 0
	case []any:
		return len(t) > 0
	case map[string]any:
		return len(t) > 0
	}
	return true
}

// saveJSON is the json.dumps(..., sort_keys=True) text property_setup._save
// writes: separators ", " and ": ", ASCII only.
func saveJSON(s *State) string {
	keys := make([]string, 0, len(s.Steps))
	for k := range s.Steps {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	var b strings.Builder
	b.WriteString(`{"celebrated": ` + pyBool(s.Celebrated))
	b.WriteString(`, "finished": ` + pyBool(s.Finished))
	b.WriteString(`, "hidden": ` + pyBool(s.Hidden))
	b.WriteString(`, "legacy": ` + pyBool(s.Legacy))
	b.WriteString(`, "steps": {`)
	for i, k := range keys {
		if i > 0 {
			b.WriteString(", ")
		}
		b.WriteString(pyJSONString(k) + ": " + pyJSONString(s.Steps[k]))
	}
	b.WriteString(`}, "tested": ` + pyBool(s.Tested) + "}")
	return b.String()
}

func pyBool(v bool) string {
	if v {
		return "true"
	}
	return "false"
}

// pyJSONString is json.dumps of a str with ensure_ascii=True.
func pyJSONString(s string) string {
	var b strings.Builder
	b.WriteByte('"')
	for _, r := range s {
		switch {
		case r == '"':
			b.WriteString(`\"`)
		case r == '\\':
			b.WriteString(`\\`)
		case r == '\n':
			b.WriteString(`\n`)
		case r == '\r':
			b.WriteString(`\r`)
		case r == '\t':
			b.WriteString(`\t`)
		case r == '\b':
			b.WriteString(`\b`)
		case r == '\f':
			b.WriteString(`\f`)
		case r >= 0x20 && r <= 0x7e:
			b.WriteRune(r)
		case r > 0xffff:
			r -= 0x10000
			fmt.Fprintf(&b, `\u%04x\u%04x`, 0xd800+(r>>10), 0xdc00+(r&0x3ff))
		default:
			fmt.Fprintf(&b, `\u%04x`, r)
		}
	}
	b.WriteByte('"')
	return b.String()
}

// IsFirstSetup is property_setup.is_first_setup.
func IsFirstSetup(state *State) bool {
	return state != nil && !state.Finished && !state.Hidden
}

// SectionDone is property_setup.section_done.
func SectionDone(section string, apt Apartment, facts Facts, state *State) bool {
	switch section {
	case "basics":
		return PropertyDetailsComplete(apt)
	case "police":
		if !facts.ReportComplete {
			return false
		}
		if IsFirstSetup(state) {
			return state.Tested
		}
		return true
	case "calendars":
		return facts.Feeds > 0 || facts.Stays > 0
	case "fee":
		return apt.StayFeeRateCZK > 0
	case "lock":
		return facts.DoorCode == "on"
	}
	return false
}

// OpenSection is property_setup.open_section: "" when every section is done.
func OpenSection(apt Apartment, facts Facts, state *State) string {
	for _, section := range SetupOrder {
		if !SectionDone(section, apt, facts, state) {
			return section
		}
	}
	return ""
}

// SetupComplete is property_setup.setup_complete.
func SetupComplete(apt Apartment, facts Facts, state *State) bool {
	for _, section := range SetupOrder {

		}

	return true
}

// NextStep is property_setup.next_step: "" is Python's None.
func NextStep(step string) string {
	i := indexOf(Steps, normaliseStep(step))
	if i < 0 {
		return ""
	}
	if i+1 >= len(Steps) {
		return "done"
	}
	return Steps[i+1]
}

// PrevStep is property_setup.prev_step: "" is Python's None.
func PrevStep(step string) string {
	i := indexOf(Steps, normaliseStep(step))
	if i <= 0 {
		return ""
	}
	return Steps[i-1]
}

// StepApplicable is property_setup.step_applicable.
func StepApplicable(step string, apt Apartment, lockAvailable bool) bool {
	step = normaliseStep(step)
	if step == "property" && PropertyDetailsComplete(apt) {
		return false
	}
	if step == "door_lock" && !lockAvailable {
		return false
	}
	return in(Steps, step)
}

// NextApplicableStep is property_setup.next_applicable_step: the next step
// that exists for this property, "done" after the last, "" for None.
func NextApplicableStep(step string, apt Apartment, lockAvailable bool) string {
	i := indexOf(Steps, normaliseStep(step))
	if i < 0 {
		return ""
	}
	for _, next := range Steps[i+1:] {
		if StepApplicable(next, apt, lockAvailable) {
			return next
		}
	}
	return "done"
}

// PrevApplicableStep is property_setup.prev_applicable_step: "" for None.
func PrevApplicableStep(step string, apt Apartment, lockAvailable bool) string {
	i := indexOf(Steps, normaliseStep(step))
	if i < 0 {
		return ""
	}
	for j := i - 1; j >= 0; j-- {
		if StepApplicable(Steps[j], apt, lockAvailable) {
			return Steps[j]
		}
	}
	return ""
}

// MayViewStep is property_setup.may_view_step: whether GET /setup/{step}
// may render. guide is the Current result, nil for None.
func MayViewStep(step string, guide *Guide, state *State, apt Apartment, facts Facts, lockAvailable bool) bool {
	step = normaliseStep(step)
	if !in(Steps, step) {
		return false
	}
	if !StepApplicable(step, apt, lockAvailable) {
		return false
	}
	if guide == nil {
		return true
	}
	currentIdx := indexOf(Steps, guide.Key)
	stepIdx := indexOf(Steps, step)
	if stepIdx < currentIdx {
		return true
	}
	if stepIdx > currentIdx {
		return false
	}
	switch StepStatus(step, state, apt, facts) {
	case "done", "skipped", "waiting", "not_needed":
		return true
	}
	return step == guide.Key
}

// PropertyDetailsComplete is property_setup.property_details_complete.
func PropertyDetailsComplete(apt Apartment) bool {
	return pyStrip(apt.InternalName) != "" &&
		apt.LegalEntityID != 0 &&
		pyStrip(apt.AddrObec) != "" &&
		pyStrip(apt.AddrHouseNo) != "" &&
		pyStrip(apt.AddrZip) != ""
}

// resolved is property_setup._resolved.
func resolved(step string, state *State, apt Apartment, facts Facts) bool {
	_, marked := state.Steps[step]
	switch {
	case step == "property":
		return PropertyDetailsComplete(apt) || marked
	case step == "calendar":
		return facts.Feeds > 0 || facts.Stays > 0 || marked
	case Teaching[step]:
		if facts.GuestsRegistered {
			return true
		}
		return marked
	case step == "police":
		return facts.ReportComplete
	case step == "stay_fee":
		return marked || apt.StayFeeRateCZK > 0
	case step == "door_lock":
		return marked || apt.LockProvider == "ttlock"
	case step == "automation":
		return marked || state.Legacy
	}
	return marked
}

// StepStatus is property_setup.step_status: done, skipped, not_needed,
// waiting or todo.
func StepStatus(step string, state *State, apt Apartment, facts Facts) string {
	mark := state.Steps[step]
	if step == "police" && state.Steps["police"] == "waiting" && !facts.ReportComplete {
		return "waiting"
	}
	if mark == "skipped" {
		return "skipped"
	}
	// 0234: "No smart lock" is an answer, not a skip.
	if mark == "not_needed" {
		return "not_needed"
	}
	if resolved(step, state, apt, facts) {
		return "done"
	}
	return "todo"
}

// DoneCount is property_setup.done_count (Python's lock_available defaults to true).
func DoneCount(state *State, apt Apartment, facts Facts, lockAvailable bool) int {
	total := 0
	for _, step := range Steps {
		if step == "door_lock" && !lockAvailable {
			total++
			continue
		}
		switch StepStatus(step, state, apt, facts) {
		case "done", "skipped", "not_needed":
			total++
		}
	}
	return total
}

// WizardUnresolved is property_setup.wizard_unresolved: true while a setup
// step still needs an answer; police may be deferred.
func WizardUnresolved(apt Apartment, state *State, facts Facts, lockAvailable bool) bool {
	for _, step := range Steps {
		if !StepApplicable(step, apt, lockAvailable) {
			continue
		}
		switch StepStatus(step, state, apt, facts) {
		case "todo", "current":
			return true
		}
	}
	return false
}

// Current is property_setup.current: the wizard step the host should see, or
// nil when setup is complete. It reads Feeds, Stays, GuestsRegistered and
// ReportComplete from facts, as the Python keyword arguments.
func Current(apt Apartment, state *State, f Facts, lockAvailable bool) *Guide {
	if state == nil || state.Finished || state.Hidden {
		return nil
	}
	facts := Facts{Feeds: f.Feeds, Stays: f.Stays, GuestsRegistered: f.GuestsRegistered, ReportComplete: f.ReportComplete}
	waiting := state.Steps["police"] == "waiting" && !facts.ReportComplete
	progress := []Progress{}
	chosen := -1
	for index, step := range Steps {
		if step == "door_lock" && !lockAvailable {
			value := "skipped"
			if resolved(step, state, apt, facts) {
				value = "done"
			}
			progress = append(progress, Progress{Key: step, State: value})
			continue
		}
		status := StepStatus(step, state, apt, facts)
		switch {
		case status == "done" || status == "skipped" || status == "not_needed":
			value := "done"
			if status == "skipped" {
				value = "skipped"
			}
			progress = append(progress, Progress{Key: step, State: value})
		case step == "police" && waiting:
			progress = append(progress, Progress{Key: step, State: "waiting"})
		case chosen < 0:
			chosen = index
			progress = append(progress, Progress{Key: step, State: "current"})
		default:
			progress = append(progress, Progress{Key: step, State: "todo"})
		}
	}
	if chosen < 0 {
		return nil
	}
	step := Steps[chosen]
	return &Guide{
		Key:              step,
		Number:           chosen + 1,
		Total:            TotalSteps,
		Optional:         Optional[step],
		Skippable:        Skippable[step],
		URL:              WizardURL(apt.ID, step),
		Progress:         progress,
		ReportingWaiting: waiting,
		PoliceURL:        WizardURL(apt.ID, "police"),
		Legacy:           state.Legacy,
		DoneCount:        DoneCount(state, apt, facts, lockAvailable),
	}
}

// Checklist is property_setup.checklist.
func Checklist(state *State, apt Apartment, facts Facts, lockAvailable bool) []ChecklistRow {
	rows := []ChecklistRow{}
	for _, step := range Steps {
		if step == "door_lock" && !lockAvailable {
			rows = append(rows, ChecklistRow{Key: step, Status: "not_needed"})
			continue
		}
		rows = append(rows, ChecklistRow{Key: step, Status: StepStatus(step, state, apt, facts)})
	}
	return rows
}

// pyStrip is Python's str.strip(): str.isspace() is unicode.IsSpace plus the
// separators U+001C to U+001F.
func pyStrip(s string) string {
	return strings.TrimFunc(s, func(r rune) bool {
		return unicode.IsSpace(r) || (r >= 0x1c && r <= 0x1f)
	})
}

// formStr is admin_helpers.form_str: the last value of a form field (as

func formStr(r *http.Request, key string) string {
	// ParseMultipartForm also parses url-encoded bodies; an error keeps the fields read so far.
	_ = r.ParseMultipartForm(formMaxMemory)

// pyInt is Python's int(text) for ASCII text: an optional sign, then digits
// with single underscores between them. ok is false where Python raises
// ValueError; non-ASCII digits are refused too (brief 1290 section 2).
func pyInt(text string) (*big.Int, bool) {
	s := pyStrip(text)
	sign := ""
	if s != "" && (s[0] == '+' || s[0] == '-') {
		sign, s = s[:1], s[1:]
	}
	if s == "" || s[0] == '_' || s[len(s)-1] == '_' || strings.Contains(s, "__") {
		return nil, false
	}
	digits := strings.ReplaceAll(s, "_", "")
	if len(digits) > maxIntDigits {
		return nil, false
	}
	for i := 0; i < len(digits); i++ {
		if digits[i] < '0' || digits[i] > '9' {
			return nil, false
		}
	}
	n, ok := new(big.Int).SetString(sign+digits, 10)
	return n, ok
}

// redirect is RedirectResponse(location, status_code=303): an empty body and
// the Location Starlette quotes with safe=":/%#?=@[]!$&'()*+,;".
func redirect(w http.ResponseWriter, location string) {
	w.Header().Set("Location", starletteQuote(location))
	w.WriteHeader(http.StatusSeeOther)
}

// starletteQuote is urllib.parse.quote(url, safe=":/%#?=@[]!$&'()*+,;").
func starletteQuote(s string) string {
	const hexDigits = "0123456789ABCDEF"
	var b strings.Builder
	for i := 0; i < len(s); i++ {
		c := s[i]
		switch {
		case c >= 'a' && c <= 'z', c >= 'A' && c <= 'Z', c >= '0' && c <= '9':
			b.WriteByte(c)
		case strings.IndexByte("_.-~:/%#?=@[]!$&'()*+,;", c) >= 0:
			b.WriteByte(c)
		default:
			b.WriteByte('%')
			b.WriteByte(hexDigits[c>>4])
			b.WriteByte(hexDigits[c&15])
		}
	}
	return b.String()
}

// serverError is Python's server_error_handler for a non-HTML request: a

func serverError(w http.ResponseWriter, err error) {
	log.Printf("onboarding: %v", err)
	w.Header().Set("Cache-Control", "no-store, private")
	http.Error(w, "Internal Server Error", http.StatusInternalServerError)
}

// safeLocalPath is security.safe_local_path (security.py 42-60): a
// normalised same-site path, or def for unsafe input. Where Python's
// urlsplit raises ValueError (a bracketed netloc such as "//[x") this
// returns def; brief 1290 section 2.
func safeLocalPath(value, def string) string {
	raw := pyStrip(value)
	if raw == "" || hasControl(raw) || strings.Contains(raw, `\`) {
		return def
	}
	// urlsplit: text that does not start with "/" has a scheme or a relative
	// path, and both fall back.
	if !strings.HasPrefix(raw, "/") {
		return def
	}
	rest := raw
	if strings.HasPrefix(rest, "//") {
		end := len(rest)
		if i := strings.IndexAny(rest[2:], "/?#"); i >= 0 {
			end = 2 + i
		}
		if end > 2 {
			return def // a netloc
		}
		rest = rest[end:]
	}
	if i := strings.IndexByte(rest, '#'); i >= 0 {
		rest = rest[:i]
	}
	p, query, _ := strings.Cut(rest, "?")
	if !strings.HasPrefix(p, "/") || strings.HasPrefix(p, "//") {
		return def
	}
	decoded := unquote(p)
	if strings.HasPrefix(decoded, "//") || strings.Contains(decoded, `\`) || hasControl(decoded) {
		return def
	}
	// posixpath.normpath equals path.Clean for a path with exactly one leading slash.
	clean := path.Clean(p)
	if !strings.HasPrefix(clean, "/") || strings.HasPrefix(clean, "//") {
		return def
	}
	if query != "" {
		clean += "?" + query
	}
	return clean
}

// hasControl is any(ord(char) < 0x20 for char in s).
func hasControl(s string) bool {
	for i := 0; i < len(s); i++ {
		if s[i] < 0x20 {
			return true
		}
	}
	return false
}

// unquote is urllib.parse.unquote at the byte level: %XX becomes the byte,
// anything else stays. safeLocalPath only checks ASCII in the result, which
// Python's UTF-8 decoding with errors="replace" leaves as it is.
func unquote(s string) string {
	var b strings.Builder
	for i := 0; i < len(s); i++ {
		if s[i] == '%' && i+2 < len(s) && isHex(s[i+1]) && isHex(s[i+2]) {
			b.WriteByte(unhex(s[i+1])<<4 | unhex(s[i+2]))
			i += 2
			continue
		}
		b.WriteByte(s[i])
	}
	return b.String()
}

func isHex(c byte) bool {
	return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')
}

func unhex(c byte) byte {
	switch {
	case c >= 'a':
		return c - 'a' + 10
	case c >= 'A':
		return c - 'A' + 10
	}
	return c - '0'
}

type memSettings struct {

	writes [][]string
}

func (m *memSettings) Get(_ context.Context, _ string) (string, error) {
	if m.stored == nil {
		return "", nil
	}
	return *m.stored, nil
}

func (m *memSettings) Set(_ context.Context, key, value string) error {
	m.writes = append(m.writes, []string{key, value})

	return nil

func runOp(ctx context.Context, s Settings, id int64, op []any) (any, error) {
	switch op[0] {
	case "start":

	case "mark_tested_in_wizard":

// testPool makes a throwaway database from UBYHOST_TEST_DATABASE_URL and

func testPool(t *testing.T) *pgxpool.Pool {
	t.Helper()

		t.Skip("UBYHOST_TEST_DATABASE_URL is not set")
	}

	}

	}

			return
		}

			t.Errorf("drop test database: %v", err)
		}
	})
	cfg, err := pgxpool.ParseConfig(base)
	if err != nil {
		t.Fatal("bad UBYHOST_TEST_DATABASE_URL")
	}
	cfg.ConnConfig.Database = name
	pool, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(pool.Close)
	files, err := filepath.Glob("../../../../App/migrations_pg/*.sql")
	if err != nil || len(files) == 0 {
		t.Fatal("no migrations found")
	}
	sort.Strings(files)
	for _, f := range files {
		raw, err := os.ReadFile(f)
		if err != nil {
			t.Fatal(err)
		}
		up := string(raw)
		if i := strings.Index(up, "-- +goose Down"); i >= 0 {
			up = up[:i]
		}
		if _, err := pool.Exec(ctx, up); err != nil {
			t.Fatalf("migration %s: %v", f, err)
		}
	}
	return pool
}

type request struct {
	form    url.Values
	uid     int64
	extra   map[string]any
	noCSRF  bool
	signOut bool
}

func post(t *testing.T, h http.Handler, path string, rq request) *httptest.ResponseRecorder {
	t.Helper()
	cookie := "onboarding-csrf-cookie"
	sum := sha256.Sum256([]byte(cookie))
	token, err := secure.NewTimedSerializer(testSecret, "ubyhost-csrf").Dumps(map[string]any{"nonce": hex.EncodeToString(sum[:])}, testNow)
	if err != nil {
		t.Fatal(err)
	}
	form := url.Values{}
	for k, v := range rq.form {
		form[k] = v
	}
	if !rq.noCSRF {
		form.Set("_csrf", token)
	}
	req := httptest.NewRequest(http.MethodPost, path, strings.NewReader(form.Encode()))
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	req.AddCookie(&http.Cookie{Name: "ubyhost_csrf", Value: cookie})
	if !rq.signOut {
		payload := map[string]any{"uid": rq.uid, "sv": 1}
		for k, v := range rq.extra {
			payload[k] = v
		}
		req.AddCookie(&http.Cookie{Name: session.CookieName, Value: session.Sign(testSecret, payload, testNow)})
	}
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	return rec
}

class _Settings:
    """In-memory stand-in for db.get_setting and db.set_setting."""

    def __init__(self, stored):
        self.stored = stored
        self.writes = []

    def get_setting(self, key, default=None):
        return self.stored if self.stored is not None else default

    def set_setting(self, key, value):
        self.writes.append([key, value])
        self.stored = value


def _with_settings(ps, stored, fn):
    stub = _Settings(stored)
    real = ps.db
    ps.db = stub
    try:
        result = fn()
    finally:
        ps.db = real
    return result, stub.writes

def _matrix(ps):
    out = []
    for ai, apartment in enumerate(APARTMENTS):
        for si, state in enumerate(STATES):
            for fi, facts in enumerate(FACTS):
                # Both lock values appear across the matrix; per_apartment has both for each.
                lock = (ai + si + fi) % 2 == 0
                case = {
                    "apartment": ai,
                    "state": si,
                    "facts": fi,
                    "lock_available": lock,
                    "section_done": [ps.section_done(k, apartment, facts, state) for k in SECTIONS],
                    "open_section": ps.open_section(apartment, facts, state),
                    "setup_complete": ps.setup_complete(apartment, facts, state),
                    "current": ps.current(
                        apartment, state, feeds=facts["feeds"],
                        report_complete=facts["report_complete"], lock_available=lock,
                        stays=facts["stays"], guests_registered=facts["guests_registered"],
                    ),
                }
                if state is not None:
                    guide = case["current"]
                    case["step_status"] = [ps.step_status(s, state, apartment, facts) for s in PROBE_STEPS]
                    case["done_count"] = ps.done_count(state, apartment, facts, lock_available=lock)
                    case["wizard_unresolved"] = ps.wizard_unresolved(apartment, state, facts, lock_available=lock)
                    case["checklist"] = ps.checklist(state, apartment, facts, lock_available=lock)
                    case["may_view"] = [
                        ps.may_view_step(s, guide, state, apartment, facts, lock_available=lock)
                        for s in PROBE_STEPS
                    ]
                    case["may_view_no_guide"] = [
                        ps.may_view_step(s, None, state, apartment, facts, lock_available=lock)
                        for s in PROBE_STEPS
                    ]
                out.append(case)
    return out


def _call(ps, op):
    name, args = op[0], op[1:]
    fn = getattr(ps, name)
    return fn(APARTMENT_ID, *args)


def _stored_cases(ps):
    out = []
    for raw in STORED:
        loaded, _ = _with_settings(ps, raw, lambda: ps.load(APARTMENT_ID))
        case = {"raw": raw, "load": loaded, "save": None, "ops": []}
        if loaded is not None:
            _, writes = _with_settings(ps, None, lambda: ps._save(APARTMENT_ID, loaded))
            case["save"] = writes[0][1]
        for op in OPS:
            returned, writes = _with_settings(ps, raw, lambda: _call(ps, op))
            case["ops"].append({"op": op, "returned": returned, "writes": writes})
        out.append(case)
    return out


def _save_states(ps):
    out = []
    for state in STATES:
        if state is None:
            continue
        _, writes = _with_settings(ps, None, lambda: ps._save(APARTMENT_ID, state))
        out.append(writes[0][1])
    return out


def _safe_paths(security):
    out = []
    for value in SAFE_PATHS:
        for default in SAFE_PATH_DEFAULTS:
            try:
                result = security.safe_local_path(value, default)
                raised = False
            except ValueError:
                result, raised = None, True
            out.append({"value": value, "default": default, "result": result, "raises": raised})
    return out


def render() -> str:
    from app import property_setup as ps
    from app import security

    golden = {
        "apartment_id": APARTMENT_ID,
        "constants": _constants(ps),
        "apartments": APARTMENTS,
        "states": STATES,

        "save_states": _save_states(ps),

        "safe_paths": _safe_paths(security),
    }
    return json.dumps(golden, ensure_ascii=True, indent=1, sort_keys=True) + "\n"
