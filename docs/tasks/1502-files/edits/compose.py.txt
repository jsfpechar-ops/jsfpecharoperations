import sys
p = 'deploy/lightsail/docker-compose.yml'
s = open(p, encoding='utf-8').read()
start = s.index('\n  ubyhost-go:\n')
end = s.index('\n    volumes:\n', start) + 1
block = s[start:end]
old_comment = '    # (no SES/AWS, Turnstile, Grafana or backup secrets). A bare key is passed\n'
new_comment = '    # (no SES/AWS, Grafana or backup secrets). A bare key is passed\n'
posthog = '\n      POSTHOG_PROJECT_API_KEY:\n'
terms = '\n      UBYHOST_TERMS_VERSION:\n'
for anchor in (posthog, terms):
    if block.count(anchor) != 1:
        sys.exit(f'STOP: anchor not found once: {anchor.strip()}')
added = []
block = block.replace(old_comment, new_comment)
if '\n      UBYHOST_TRUSTED_PROXY_CIDRS:' not in block:
    block = block.replace(terms, terms + '      UBYHOST_TRUSTED_PROXY_CIDRS:\n')
    added.append('UBYHOST_TRUSTED_PROXY_CIDRS')
turnstile = [k for k in ('TURNSTILE_HOSTNAMES', 'TURNSTILE_SECRET', 'TURNSTILE_SITE_KEY')
             if f'\n      {k}:' not in block]
if turnstile:
    block = block.replace(posthog, posthog + ''.join(f'      {k}:\n' for k in turnstile))
    added += turnstile
if new_comment not in block:
    sys.exit('STOP: the ubyhost-go comment line was not found')
open(p, 'w', encoding='utf-8').write(s[:start] + block + s[end:])
print(f'added {len(added)} keys to ubyhost-go')
