// isPySpace is Python's str.isspace for the characters statements and names can hold.
func isPySpace(r rune) bool { return unicode.IsSpace(r) || (r >= 0x1c && r <= 0x1f) }

// pyStrip is str.strip().
func pyStrip(s string) string { return strings.TrimFunc(s, isPySpace) }

// pyLStrip is str.lstrip().
func pyLStrip(s string) string { return strings.TrimLeftFunc(s, isPySpace) }

// pyLines is str.splitlines(): it splits after \n, \r, \r\n, \v, \f, \x1c-\x1e,
// \x85, U+2028 and U+2029 and drops the break.
func pyLines(s string) []string {
	var out []string
	var cur strings.Builder
	rs := []rune(s)
	for i := 0; i < len(rs); i++ {
		switch r := rs[i]; r {
		case '\n', '\v', '\f', 0x1c, 0x1d, 0x1e, 0x85, 0x2028, 0x2029:
		case '\r':
			if i+1 < len(rs) && rs[i+1] == '\n' {
				i++
			}
		default:
			cur.WriteRune(r)
			continue
		}
		out = append(out, cur.String())
		cur.Reset()
	}
	if cur.Len() > 0 {
		out = append(out, cur.String())
	}
	return out
}

// cut is s[:n] where n counts characters, as a Python slice does.
func cut(s string, n int) string {
	i := 0
	for idx := range s {
		if i == n {
			return s[:idx]
		}
		i++
	}
	return s
}

// floorDiv is Python's a // b for b > 0.
func floorDiv(a, b int64) int64 {
	q := a / b
	if a%b != 0 && a < 0 {
		q--
	}
	return q
}

func deref(p *string) string {
	if p == nil {

func itoa(n int64) string { return strconv.FormatInt(n, 10) }

// csvField quotes a field the way csv.writer does with QUOTE_MINIMAL.
func csvField(f string) string {
	if strings.ContainsAny(f, ";\"\r\n") {
		return "\"" + strings.ReplaceAll(f, "\"", "\"\"") + "\""
	}
	return f
}

// csvBytes is the shared writer of the accountant's files: a BOM, semicolons
// and \r\n, so Czech Excel opens it straight away (billing_admin.export_csv,
// billing_analytics._csv_bytes).
func csvBytes(header []string, rows [][]string) []byte {
	var b strings.Builder
	b.WriteString(bom)
	for _, row := range append([][]string{header}, rows...) {
		for i, f := range row {
			if i > 0 {
				b.WriteString(";")
			}
			b.WriteString(csvField(f))
		}
		b.WriteString("\r\n")
	}
	return []byte(b.String())
}

// dayOf keeps only the year, month and day of t (zero means now), in UTC.
func dayOf(t time.Time) time.Time {
	if t.IsZero() {
		t = time.Now()
	}
	return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, time.UTC)
}

// queryAll runs sql and scans every row. The result is never nil, so it
// marshals as [] like the Python list.
func queryAll[T any](ctx context.Context, q DB, scan func(pgx.Rows) (T, error), sql string, args ...any) ([]T, error) {
	rows, err := q.Query(ctx, sql, args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []T{}
	for rows.Next() {
		v, err := scan(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, v)
	}
	return out, rows.Err()
}

// nextMonth is billing_admin._next_month (lines 59-61): the first day after the month.
func nextMonth(month string) (string, error) {
	if len(month) < 7 {
		return "", fmt.Errorf("bad month %q", month)
	}
	year, err1 := strconv.Atoi(month[:4])
	mon, err2 := strconv.Atoi(month[5:7])
	if err1 != nil || err2 != nil {
		return "", fmt.Errorf("bad month %q", month)
	}
	if mon == 12 {
		return fmt.Sprintf("%d-01-01", year+1), nil
	}
	return fmt.Sprintf("%d-%02d-01", year, mon+1), nil
}

// Kc is the page's formatter, routes/admin_billing._kc (lines 25-26): whole
// crowns (floored) with a space between thousands, 1234567 haler -> "12 345".
func Kc(haler int64) string {
	n := floorDiv(haler, 100)
	neg := n < 0
	var digits string
	if neg {
		digits = strconv.FormatUint(uint64(-n), 10)
	} else {
		digits = strconv.FormatUint(uint64(n), 10)
	}
	var b strings.Builder
	if neg {
		b.WriteString("-")
	}
	for i, c := range digits {
		if i > 0 && (len(digits)-i)%3 == 0 {
			b.WriteString(" ")
		}
		b.WriteRune(c)
	}
	return b.String()
}

// ParseGPC is bank_import.parse_gpc (lines 72-95): one 075 line per
// transaction, fixed columns counted in characters, credits only. A line whose
// amount field is not a number is Python's ValueError and returns an error.
func ParseGPC(text string) ([]Payment, error) {
	rows := []Payment{}
	for _, raw := range pyLines(text) {
		rs := []rune(raw)
		if !strings.HasPrefix(raw, "075") || len(rs) < 97 {
			continue
		}
		for len(rs) < 128 {
			rs = append(rs, ' ')
		}

			continue
		}
		received := gpcDate(string(rs[122:128]))
		if received == "" {
			received = gpcDate(string(rs[91:97]))
		}
		if received == "" {
			continue
		}
		tx := strings.TrimLeft(pyStrip(string(rs[35:48])), "0")

// csvAmount is bank_import._csv_amount (lines 109-117): "1 234,50" or
// "1.234,50" in crowns to haler, rounded half to even. ok is false for what
// Python turns into None. Not ported: "inf" and "infinity" crash Python (an
// OverflowError) and read as not a number here; the hex and unicode-digit
// forms float() accepts read as not a number too.
func csvAmount(value string) (int64, bool) {
	text := strings.ReplaceAll(strings.ReplaceAll(value, " ", ""), "\u00a0", "")
	if strings.Contains(text, ",") && strings.Contains(text, ".") {
		text = strings.ReplaceAll(text, ".", "")
	}
	text = pyStrip(strings.ReplaceAll(text, ",", "."))
	if !floatRE.MatchString(text) {
		return 0, false
	}
	f, err := strconv.ParseFloat(strings.ReplaceAll(text, "_", ""), 64)
	if err != nil {
		return 0, false
	}
	f = math.RoundToEven(f * 100)
	if math.IsInf(f, 0) || math.Abs(f) > 9e18 {
		return 0, false
	}
	return int64(f), true
}

func csvDate(value string) string {
	text := pyStrip(value)
	for _, p := range []struct {
		re  *regexp.Regexp
		dmy bool
	}{{dmyRE, true}, {ymdRE, false}} {
		m := p.re.FindStringSubmatch(text)
		if m == nil {
			continue
		}
		a, _ := strconv.Atoi(m[1])
		b, _ := strconv.Atoi(m[2])
		c, _ := strconv.Atoi(m[3])
		if p.dmy {
			return validDate(c, b, a)
		}
		return validDate(a, b, c)
	}
	return ""
}

// ParseCSV is bank_import.parse_csv (lines 133-169): a CSV export with a
// header row; columns are found by their Czech or English names.
func ParseCSV(text string) ([]Payment, error) {
	lines := pyLines(text)
	start := -1
	for i, line := range lines {
		f := fold(line)
		if strings.Contains(f, "castka") || strings.Contains(f, "amount") {
			start = i
			break
		}
	}
	if start < 0 {
		return nil, &StatementError{"no_header"}
	}
	body := strings.Join(lines[start:], "\n")
	delimiter := ','
	if strings.Count(body, ";") >= strings.Count(body, ",") {
		delimiter = ';'
	}
	reader := csv.NewReader(strings.NewReader(body))
	reader.Comma = delimiter
	reader.LazyQuotes = true
	reader.FieldsPerRecord = -1
	head, err := reader.Read()
	if err != nil {
		return nil, err
	}
	header := make([]string, len(head))
	for i, h := range head {
		header[i] = fold(h)
	}
	cols := map[string]int{}
	for _, f := range csvFields {
		found := false
		for _, name := range f.names {
			for i, h := range header {
				if h == name {
					cols[f.field] = i
					found = true
					break
				}
			}
			if found {
				break
			}
		}
	}
	_, hasAmount := cols["amount"]
	_, hasDate := cols["date"]
	if !hasAmount || !hasDate {
		return nil, &StatementError{"no_columns"}
	}
	rows := []Payment{}
	for {
		record, err := reader.Read()
		if err == io.EOF {
			break
		}
		if err != nil {
			return nil, err
		}
		blank := true
		for _, c := range record {
			if pyStrip(c) != "" {
				blank = false
			}
		}
		if blank {
			continue
		}
		cell := func(field string) string {
			i, ok := cols[field]
			if !ok || i >= len(record) {
				return ""
			}
			return pyStrip(record[i])
		}
		amount, okAmount := csvAmount(cell("amount"))
		received := csvDate(cell("date"))
		if !okAmount || received == "" || amount <= 0 {
			continue
		}
		tx := cell("id")
		account := cell("account")
		if account != "" && cell("bank") != "" && !strings.Contains(account, "/") {
			account = account + "/" + cell("bank")
		}
		id := rowID(received, itoa(amount), cell("vs"), account, cell("message"))
		if tx != "" {
			id = "csv-" + tx
		}
		rows = append(rows, Payment{
			BankTxID:       id,
			ReceivedOn:     received,
			AmountHaler:    amount,
			VS:             vsOf(cell("vs")),
			CounterAccount: account,
			Message:        cut(cell("message"), 140),
		})
	}
	return rows, nil
}

// Parse is bank_import.parse (lines 172-176): bytes in, rows out, no database
// and no outbound request. A statement starting with 074 is GPC, else CSV.
func Parse(data []byte) ([]Payment, error) {
	text, err := decodeText(data)
	if err != nil {
		return nil, err
	}
	if strings.HasPrefix(pyLStrip(text), "074") {
		return ParseGPC(text)
	}
	return ParseCSV(text)
}

// GetPortfolio is billing_analytics.portfolio (lines 46-90): every billing
// account that has a plan, priced by billing.Quote so the cockpit can never
// disagree with an issued invoice.
func GetPortfolio(ctx context.Context, q DB, today time.Time) (Portfolio, error) {
	day := dayOf(today)
	customers, err := queryAll(ctx, q, func(r pgx.Rows) (Customer, error) {
		var c Customer
		var interval *string
		err := r.Scan(&c.OwnerUserID, &c.Status, &c.Plan, &interval, &c.ApartmentsBilled,
			&c.AddonTTLock, &c.AddonWhitelabel, &c.CarryCZK, &c.CancelAtPeriodEnd,
			&c.DiscountPercent, &c.DiscountReason, &c.TrialEndsAt, &c.PeriodStart, &c.PeriodEnd,
			&c.DisplayName, &c.Email, &c.Username)
		c.BillingInterval = deref(interval)
		return c, err
	}, `SELECT b.owner_user_id, b.status, b.plan, b.billing_interval, b.apartments_billed,
		b.addon_ttlock, b.addon_whitelabel, b.carry_czk, b.cancel_at_period_end,
		b.discount_percent, b.discount_reason,
		b.trial_ends_at, b.period_start, b.period_end,
		u.display_name, u.email, u.username
		FROM billing_account b JOIN user_account u ON u.id = b.owner_user_id
		WHERE b.plan IS NOT NULL ORDER BY b.owner_user_id`)
	if err != nil {
		return Portfolio{}, err
	}
	out := Portfolio{Customers: customers, ByStatus: map[string]int64{}, Today: isoDay(day)}
	for n := range out.Customers {
		c := &out.Customers[n]
		c.Quote, err = billing.Quote(int(c.ApartmentsBilled), c.Plan, c.BillingInterval,
			c.AddonTTLock != 0, c.AddonWhitelabel != 0, int(c.DiscountPercent))
		if err != nil {
			return Portfolio{}, err
		}
		c.NetCZK = int64(c.Quote.Total) + c.CarryCZK
		c.MonthlyCZK = int64(billing.Monthly(int(c.NetCZK), c.BillingInterval))
		if mrrStates[c.Status] {
			out.MRRCZK += c.MonthlyCZK
			if c.BillingInterval == "year" {
				out.ARRCZK += c.NetCZK
			} else {
				out.ARRCZK += c.MonthlyCZK * 12
			}
		}
		out.ByStatus[c.Status]++
		c.TrialEnds = cut(deref(c.TrialEndsAt), 10)
		if c.Status == "trialing" && c.TrialEnds != "" {
			ends, err := parseDay(c.TrialEnds)
			if err != nil {
				return Portfolio{}, fmt.Errorf("trial_ends_at of owner %d: %w", c.OwnerUserID, err)
			}
			left := daysBetween(ends, day)
			c.TrialDaysLeft = &left
		}
	}
	return out, nil
}

// GetMovement is billing_analytics.movement (lines 285-322): the month over
// month MRR change decomposed, from billing_month_snapshot. month is YYYY-MM.
func GetMovement(ctx context.Context, q DB, month string) (Movement, error) {
	if len(month) < 7 {
		return Movement{}, fmt.Errorf("bad month %q", month)
	}
	var year, mon int
	if _, err := fmt.Sscanf(month[:4], "%d", &year); err != nil {
		return Movement{}, fmt.Errorf("bad month %q", month)
	}
	if _, err := fmt.Sscanf(month[5:7], "%d", &mon); err != nil {
		return Movement{}, fmt.Errorf("bad month %q", month)
	}
	prev := fmt.Sprintf("%d-%02d", year, mon-1)
	if mon == 1 {
		prev = fmt.Sprintf("%d-12", year-1)
	}
	monthMRR := func(m string) (int64, error) {
		var s int64
		err := q.QueryRow(ctx, `SELECT COALESCE(SUM(monthly_czk), 0)::bigint FROM billing_month_snapshot
			WHERE month = $1 AND status IN ('active', 'overdue', 'canceling')`, m).Scan(&s)
		return s, err
	}
	cur, err := monthMRR(month)
	if err != nil {
		return Movement{}, err
	}
	before, err := monthMRR(prev)
	if err != nil {
		return Movement{}, err
	}
	type row struct {
		status     string
		monthly    int64
		prevCZK    *int64
		prevStatus *string
	}
	rows, err := queryAll(ctx, q, func(r pgx.Rows) (row, error) {
		var x row
		var owner int64
		return x, r.Scan(&owner, &x.status, &x.monthly, &x.prevCZK, &x.prevStatus)
	}, `SELECT s.owner_user_id, s.status, s.monthly_czk, p.monthly_czk AS prev_czk, p.status AS prev_status
		FROM billing_month_snapshot s LEFT JOIN billing_month_snapshot p
		ON p.owner_user_id = s.owner_user_id AND p.month = $1 WHERE s.month = $2`, prev, month)
	if err != nil {
		return Movement{}, err
	}
	var parts MovementParts
	for _, r := range rows {
		was := r.prevCZK != nil
		var prevCZK int64
		if was {
			prevCZK = *r.prevCZK
		}
		paying := mrrStates[r.status]
		wasPaying := was && r.prevStatus != nil && mrrStates[*r.prevStatus]
		switch {
		case paying && !wasPaying:
			parts.New += r.monthly
		case !paying && wasPaying:
			parts.Churn -= prevCZK
		case paying && wasPaying:
			if r.monthly > prevCZK {
				parts.Expansion += r.monthly - prevCZK
			} else if r.monthly < prevCZK {
				parts.Contraction -= prevCZK - r.monthly
			}
		}
	}
	return Movement{Month: month, Previous: prev, TotalChange: cur - before, Parts: parts, CurrentMRR: cur}, nil
}

	CollectedCZK int64  `json:"collected_czk"`

// Trend is billing_analytics.trend (lines 341-353): billed and collected per
// issue/paid month, the last `months` of them (the page passes 12).
func Trend(ctx context.Context, q DB, months int) ([]TrendRow, error) {
	type pair struct {
		m string
		h int64
	}
	scan := func(r pgx.Rows) (pair, error) {
		var p pair
		return p, r.Scan(&p.m, &p.h)
	}
	billed, err := queryAll(ctx, q, scan, `SELECT substr(issue_date, 1, 7) AS m, COALESCE(SUM(total_haler), 0)::bigint AS h
		FROM platform_invoice WHERE voided = 0 GROUP BY substr(issue_date, 1, 7)`)
	if err != nil {
		return nil, err
	}
	paidRows, err := queryAll(ctx, q, scan, `SELECT substr(paid_on, 1, 7) AS m, COALESCE(SUM(total_haler), 0)::bigint AS h
		FROM platform_invoice WHERE voided = 0 AND paid_on IS NOT NULL GROUP BY substr(paid_on, 1, 7)`)
	if err != nil {
		return nil, err
	}
	paid := map[string]int64{}
	for _, p := range paidRows {
		paid[p.m] = floorDiv(p.h, 100)
	}
	out := []TrendRow{}
	for _, b := range billed {
		out = append(out, TrendRow{Month: b.m, BilledCZK: floorDiv(b.h, 100), CollectedCZK: paid[b.m]})
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Month < out[j].Month })
	if months >= 0 && len(out) > months {
		out = out[len(out)-months:]
	}
	return out, nil
}

func exportAllFiles(ctx context.Context, q DB, today time.Time) ([]exportFile, error) {

			itoa(int64(c.Quote.TTLock)), itoa(int64(c.Quote.Whitelabel)), itoa(c.CarryCZK),
			itoa(c.NetCZK), itoa(c.MonthlyCZK), cut(deref(c.PeriodStart), 10),
			cut(deref(c.PeriodEnd), 10), cancel,
		})

		number, kind, corrects, issue, due, from, to string

// ExportAllZip is billing_analytics.export_all_zip: the bulk export, full
// history, three CSV files and no PDF, Excel-ready.
func ExportAllZip(ctx context.Context, q DB, today time.Time) ([]byte, error) {
	files, err := exportAllFiles(ctx, q, today)

		return nil, err

	var buf bytes.Buffer
	zw := zip.NewWriter(&buf)
	for _, f := range files {
		w, err := zw.Create(f.Name)
		if err != nil {
			return nil, err
		}
		if _, err := w.Write(f.Data); err != nil {
			return nil, err
		}
	}
	if err := zw.Close(); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil

					x, err := n.Int64()
					if err != nil {
						t.Fatalf("%s: %v", table.Table, err)
					}
					args[i] = x
				}
			}
			if _, err := pool.Exec(ctx, sql, args...); err != nil {
				t.Fatalf("seed %s %v: %v", table.Table, row, err)
			}
		}
	}
	used := map[string]bool{}
	compared := 0
	for i, s := range g.Scenarios {
		call, ok := dispatch[s.Fn]
		if !ok {
			t.Fatalf("scenario %d: no Go call for %q", i, s.Fn)
		}
		used[s.Fn] = true
		got, err := call(ctx, pool, s.Args, mustDay(t, s.Today))

		}
		compared++

		}
	}
	for fn := range dispatch {
		if !used[fn] {
			t.Errorf("no golden scenario for %s", fn)
		}
	}
	if compared != len(g.Scenarios) {
		t.Fatalf("compared %d of %d scenarios", compared, len(g.Scenarios))
	}
}

// It skips when the variable is unset.
func testPool(t *testing.T) *pgxpool.Pool {
	t.Helper()
	base := os.Getenv("UBYHOST_TEST_DATABASE_URL")
	if base == "" {
		t.Skip("UBYHOST_TEST_DATABASE_URL is not set")
	}
	ctx := context.Background()
	name := fmt.Sprintf("ubyhost_go_billingweb_%d", time.Now().UnixNano())
	admin, err := pgx.Connect(ctx, base)
	if err != nil {
		t.Fatal("admin connect failed")
	}
	if _, err := admin.Exec(ctx, "CREATE DATABASE "+name); err != nil {
		admin.Close(ctx)
		t.Fatal(err)
	}
	admin.Close(ctx)
	t.Cleanup(func() {
		c, err := pgx.Connect(context.Background(), base)
		if err != nil {
			t.Errorf("admin connect for cleanup failed")
			return
		}
		defer c.Close(context.Background())
		if _, err := c.Exec(context.Background(), "DROP DATABASE IF EXISTS "+name+" WITH (FORCE)"); err != nil {
			t.Errorf("drop test database: %v", err)
		}
	})
	cfg, err := pgxpool.ParseConfig(base)
	if err != nil {
		t.Fatal("bad UBYHOST_TEST_DATABASE_URL")
	}
	cfg.ConnConfig.Database = name
	pool, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(pool.Close)
	files, err := filepath.Glob("../../../../App/migrations_pg/*.sql")
	if err != nil || len(files) == 0 {
		t.Fatal("no migrations found")
	}
	sort.Strings(files)
	for _, f := range files {
		raw, err := os.ReadFile(f)
		if err != nil {
			t.Fatal(err)
		}
		up := string(raw)
		if i := strings.Index(up, "-- +goose Up"); i >= 0 {
			up = up[i+len("-- +goose Up"):]
		}
		if i := strings.Index(up, "-- +goose Down"); i >= 0 {
			up = up[:i]
		}
		if _, err := pool.Exec(ctx, up); err != nil {
			t.Fatalf("%s: %v", filepath.Base(f), err)
		}
	}
	return pool
}

    line = "075" + "0000001234567890"
    line += account.rjust(16, "0")[:16]
    line += tx.rjust(13, "0")[:13]
    line += (amount if isinstance(amount, str) else str(amount)).rjust(12, "0")[:12]
    line += code + vs.rjust(10, "0")[:10] + "0" * 20 + d91 + message.ljust(20)[:20] + "00203" + d122.ljust(6)

def _parse_case(bank_import, name, data):
    case = {"name": name, "input_b64": base64.b64encode(data).decode("ascii")}
    try:
        case["rows"] = bank_import.parse(data)
    except bank_import.StatementError as exc:
        case["error"] = "StatementError"
        case["reason"] = str(exc)
    except ValueError:
        case["error"] = "ValueError"
    return case

def _json_default(value):
    if isinstance(value, date):
        return value.isoformat()
    raise TypeError(type(value))


def _clean(value):
    """Round-trip through JSON, so dates become strings and rows become dicts."""
    return json.loads(json.dumps(value, default=_json_default))


def _seed(db) -> None:
    for table in SEED:
        cols = ", ".join(table["columns"])
        marks = ", ".join("?" for _ in table["columns"])
        for row in table["rows"]:
            db.execute(f"INSERT INTO {table['table']} ({cols}) VALUES ({marks})", row)


def _zip_files(blob: bytes) -> dict:
    with zipfile.ZipFile(io.BytesIO(blob)) as zf:
        return {name: zf.read(name).decode("utf-8") for name in sorted(zf.namelist())}


def render() -> str:

    from app import bank_import, billing_admin, billing_analytics, db

            return cls.current

    billing_admin.date = Frozen
    billing_analytics.date = Frozen
    db.init_db()
    _seed(db)

    def kc_route(haler):
        return f"{int(haler) // 100:,}".replace(",", " ")

    def admin_page(mesic, q, state, day):
        # The context of routes/admin_billing.py billing_admin_page (lines 33-69).
        month = billing_admin.month_or_current(mesic)

            "unpaid_total": kc_route(sum(int(r["total_haler"]) for r in unpaid)),

            "movement": billing_analytics.movement(month),

        }

        "next_month_start": [
            {"in": d.isoformat(), "out": billing_analytics.next_month_start(d)}
