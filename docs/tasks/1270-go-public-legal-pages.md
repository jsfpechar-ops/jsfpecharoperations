How it works (the orchestrator wrote and dry-ran every file below; you copy them, you do not write code):

- **Gate (change nothing):** Caddy sends a path to Go only when it is listed in `UBYHOST_GO_PATHS` (`deploy/lightsail/caddy/Caddyfile.cloudflare` line 79: `@go path {$UBYHOST_GO_PATHS:/__go_off__}`). Until the owner adds these five paths, Python serves them.

Never edit it by hand.
- **Golden pages:** the same script renders each case with the FastAPI TestClient on a throwaway SQLite database and saves the normalized HTML (ASCII whitespace runs to one space, `> <` to `><`, trimmed) to `testdata/<case>.html`, and the request, env and expected headers to `testdata/cases.json`. Cases: every page in CS (no query) and EN (`?lang=en`), every page with PostHog on, the language cookie (`en`, `vi`, and `?lang=cs` over cookie `en`), operator without DIČ and with another e-mail, privacy placeholders (non-production, empty operator) and production with an empty operator, sign-up on, sign-up plus Meta on, Meta without sign-up, PostHog on in EN, sub-processors with sign-up, sign-up plus Meta, web push off, another effective date and version, and an http base URL. No token needs stripping: the five pages print no CSRF token (the word "csrf" in privacy pages is the cookie table row `ubyhost_csrf`), and asset versions are fixed `?v=` strings in the templates.
- **Language:** `templating.render` uses `host_i18n.resolve_language(request, default=...)`: `?lang=`, then cookie `ubyhost_lang`, each through `supported_language` (first two letters, en/cs/vi), else `PUBLIC_DEFAULT_LANGUAGE` (`cs`) for a visitor or `DEFAULT_LANGUAGE` (`en`) for a signed-in host; then a public page in a language other than en/cs gets `en`. Python does not read Accept-Language. `hostweb.Lang` cannot be reused (its default is `en` and it has no public-page rule), so `resolveLang` reimplements it, with `TestResolveLang`. Translation reuses `hostweb.Catalog.T` (read-only import), which is `host_i18n.translate`.
- **Escaping:** Jinja escapes with markupsafe (`& < > ' "` to `&amp; &lt; &gt; &#39; &#34;`); Go's `html.EscapeString` writes exactly those. Go's `html/template` would also turn `+` into `&#43;` and rewrite URLs, so 1271 renders with `text/template` and every printed value goes through an escaping method (guarded by a test in 1271).
- `legal_links` (`App/app/templating.py` lines 426-446) uses a lookbehind, which Go's regexp lacks; `legalLinks` scans by hand. `TestLegalLinks` covers it.

def _cookie(header: str) -> dict:
    """One Set-Cookie header as comparable fields (attribute order and case differ between stacks)."""
    parts = [p.strip() for p in header.split(";")]
    name, _, value = parts[0].partition("=")
    out = {"name": name, "value": value, "path": "", "max_age": "", "samesite": "", "secure": False, "httponly": False}
    for attr in parts[1:]:
        key, _, val = attr.partition("=")
        key = key.strip().lower()
        if key == "path":
            out["path"] = val
        elif key == "max-age":
            out["max_age"] = val
        elif key == "samesite":
            out["samesite"] = val.lower()
        elif key == "secure":
            out["secure"] = True
        elif key == "httponly":

    return out

// resolveLang is templating.render's language for a public page:
// host_i18n.resolve_language (?lang=, then the cookie, each through
// supported_language), else PUBLIC_DEFAULT_LANGUAGE for a visitor and
// DEFAULT_LANGUAGE for a signed-in host; a language the public pages do not
// have (vi) becomes "en".
func resolveLang(r *http.Request, signedIn func(*http.Request) bool) string {
	lang := ""
	for _, v := range []string{r.URL.Query().Get("lang"), hostweb.Cookie(r, langCookieName)} {
		if lang = supportedLanguage(v); lang != "" {
			break
		}
	}
	if lang == "" {
		lang = publicDefault
		if signedIn != nil && signedIn(r) {
			lang = appDefault
		}
	}
	if lang != "en" && lang != "cs" {
		lang = "en"
	}
	return lang
}

// supportedLanguage is host_i18n.supported_language: the first two letters,
// lower-cased and trimmed, when the app speaks them (en, cs, vi), else "".
func supportedLanguage(v string) string {
	r := []rune(strings.ToLower(strings.TrimSpace(v)))
	if len(r) > 2 {
		r = r[:2]
	}
	switch s := string(r); s {
	case "en", "cs", "vi":
		return s
	}
	return ""
}

// httpsOrigin is analytics._https_origin: "https://host[:port]" or "".
func httpsOrigin(raw string) string {
	u, err := url.Parse(raw)
	if err != nil || u.Scheme != "https" || u.Hostname() == "" {
		return ""
	}
	host := strings.ToLower(u.Hostname())
	if p := u.Port(); p != "" {
		host += ":" + p
	}
	return "https://" + host
}

// hostAllowed is analytics._host_allowed: an https origin outside PostHog US.

	o := httpsOrigin(raw)
	return o != "" && !strings.Contains(o, "us.i.posthog.com") && !strings.Contains(o, "us-assets.i.posthog.com")
}

// legalPaths are templating._LEGAL_PATHS, in the regex's alternation order.
var legalPaths = []string{"/dpa", "/legal", "/privacy", "/subprocessors", "/terms"}

// legalLinks is templating._legal_links on already escaped text: a legal
// path at the start or after whitespace or "(", and followed by the end,
// whitespace or one of . , ; : ), becomes <a href="/x">/x</a>. Go regexp has
// no lookbehind, so this scans by hand; \s is Unicode whitespace as in Python.
func legalLinks(s string) string {
	var b strings.Builder
	prev := rune(-1) // -1: start of the string
	for i := 0; i < len(s); {
		if prev == -1 || prev == '(' || unicode.IsSpace(prev) {
			if path := legalPathAt(s[i:]); path != "" {
				b.WriteString(`<a href="` + path + `">` + path + `</a>`)
				i += len(path)
				prev = '/' // the last rune of a path is a letter; never a boundary
				continue
			}
		}
		r, size := utf8.DecodeRuneInString(s[i:])
		b.WriteString(s[i : i+size])
		prev = r
		i += size
	}
	return b.String()
}

func legalPathAt(s string) string {
	for _, path := range legalPaths {
		if !strings.HasPrefix(s, path) {
			continue
		}
		rest := s[len(path):]
		if rest == "" {
			return path
		}
		next, _ := utf8.DecodeRuneInString(rest)
		if unicode.IsSpace(next) || strings.ContainsRune(".,;:)", next) {
			return path
		}
	}
	return ""
}
