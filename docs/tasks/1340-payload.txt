package billing

import "errors"

// Error is a billing refusal with a stable code. The codes are the messages of
// the Python ValueError the same rule raises (App/app/billing.py,
// billing_order.py), so the golden file compares them as plain strings.
// Callers test with errors.Is(err, billing.ErrAlreadyOrdered) or compare
// billing.Code(err). A wrapped error keeps its code: the text after the code
// is detail for a log, never for a page.
type Error struct{ code string }

// Error returns the code.
func (e *Error) Error() string { return e.code }

// Code returns the code of the first billing Error in err's chain, or "".
func Code(err error) string {
	var e *Error
	if errors.As(err, &e) {
		return e.code
	}
	return ""
}

// The sentinels. Python raise -> Go error:
//
//	unknown status: <x>                  ErrUnknownStatus
//	one change at a time, from ...       ErrOneChange
//	no ordered plan to change            ErrNoOrderedPlan
//	whitelabel_unavailable               ErrWhitelabelUnavailable
//	discount_out_of_range                ErrDiscountOutOfRange
//	reason_missing                       ErrReasonMissing
//	no_plan                              ErrNoPlan
//	og                                   ErrNoAccount
//	already_ordered                      ErrAlreadyOrdered
//	bad_days                             ErrBadDays
//	unknown plan or interval: <x> <y>    ErrUnknownPlan (from quote)
//
// ErrBadValue has no Python twin: Python takes any object as the value of a
// change, Go takes int, int64, bool and string and refuses the rest.
var (
	ErrUnknownStatus         = &Error{"unknown status"}
	ErrOneChange             = &Error{"one change at a time, from CHANGE_COLUMNS"}
	ErrNoOrderedPlan         = &Error{"no ordered plan to change"}
	ErrWhitelabelUnavailable = &Error{"whitelabel_unavailable"}
	ErrDiscountOutOfRange    = &Error{"discount_out_of_range"}
	ErrReasonMissing         = &Error{"reason_missing"}
	ErrNoPlan                = &Error{"no_plan"}
	ErrNoAccount             = &Error{"og"}
	ErrAlreadyOrdered        = &Error{"already_ordered"}
	ErrBadDays               = &Error{"bad_days"}
	ErrUnknownPlan           = &Error{"unknown plan or interval"}
	ErrBadValue              = &Error{"bad value"}
)

// Account is billing.account: the host's row, or nil when there is none.
func Account(ctx context.Context, q DB, owner int64) (*BillingAccount, error) {
	var a BillingAccount
	var founder, ttlock, white, cancel int
	err := q.QueryRow(ctx, accountSQL, owner).Scan(
		&a.OwnerUserID, &a.Status, &a.Plan, &a.BillingInterval, &a.PaymentMethod,
		&founder, &ttlock, &white, &a.ApartmentsBilled, &a.TrialStartedAt, &a.TrialEndsAt,
		&a.PeriodStart, &a.PeriodEnd, &cancel, &a.PendingChangeJSON, &a.CarryCZK,
		&a.BuyerName, &a.BuyerStreet, &a.BuyerCity, &a.BuyerZip, &a.BuyerICO, &a.BuyerDIC,
		&a.BuyerEmail, &a.UpdatedAt, &a.DiscountPercent, &a.DiscountReason)
	if errors.Is(err, pgx.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	a.Founder, a.AddonTTLock, a.AddonWhitelabel = founder != 0, ttlock != 0, white != 0
	a.CancelAtPeriodEnd = cancel != 0
	return &a, nil
}

}

type auditActorKey struct{}

type auditActor struct {
	userID       int64
	username     string
	impersonator *int64
}

// WithAuditActor is db.set_current_actor for Go: it remembers who is acting so
// the audit rows written with the returned context carry actor_user_id and
// impersonator_user_id, and an actor "host" is replaced by username, exactly as
// db.audit does. impersonatorUserID is the acting user's own id only while they
// are inside another workspace (the page passes nil for an ordinary action).
// Without it the columns stay NULL and the actor string is written as given.
func WithAuditActor(ctx context.Context, userID int64, username string, impersonatorUserID *int64) context.Context {
	return context.WithValue(ctx, auditActorKey{}, auditActor{userID, username, impersonatorUserID})
}

func auditActorFrom(ctx context.Context) (auditActor, bool) {
	a, ok := ctx.Value(auditActorKey{}).(auditActor)
	return a, ok
}

// audit is db.audit. The engine always receives the owner, so Python's
// fallback to the request's current owner (db._current_owner_id) is never
// needed here.
func audit(ctx context.Context, q DB, actor, action, detail string, owner int64) error {
	var actorID, impersonatorID *int64
	if a, ok := auditActorFrom(ctx); ok {
		id := a.userID
		actorID, impersonatorID = &id, a.impersonator
		if actor == "host" {
			actor = a.username
		}
	}
	_, err := q.Exec(ctx, `INSERT INTO audit (at, actor, action, detail, owner_user_id, actor_user_id, impersonator_user_id)
		VALUES ($1, $2, $3, $4, $5, $6, $7)`, ubyport.UTCNow(), actor, action, detail, owner, actorID, impersonatorID)
	return err
}

// lockKey is db.lock_key on PostgreSQL: serialise writers of one row.
func lockKey(ctx context.Context, tx pgx.Tx, name string, key int64) error {
	_, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtext($1), $2::int4)`, name, key)
	return err
}

func contains(list []string, s string) bool {
	for _, x := range list {
		if x == s {
			return true
		}
	}
	return false
}

// SetStatus is billing.set_status: move the account to new if Transitions
// allows it, false otherwise. The audit row is written after the commit.
func SetStatus(ctx context.Context, pool *pgxpool.Pool, owner int64, newStatus, actor string) (bool, error) {
	if !contains(Statuses, newStatus) {
		return false, fmt.Errorf("%w: %s", ErrUnknownStatus, newStatus)
	}
	old, moved, err := moveStatus(ctx, pool, owner, func(old string) bool {
		return contains(Transitions[old], newStatus)
	}, newStatus)
	if err != nil || !moved {
		return false, err
	}
	if err := audit(ctx, pool, actor, "billing_status", old+"->"+newStatus, owner); err != nil {
		return false, err
	}
	return true, nil
}

// moveStatus locks the account, reads its status and, when allowed(old), sets
// newStatus in the same transaction.
func moveStatus(ctx context.Context, pool *pgxpool.Pool, owner int64, allowed func(old string) bool, newStatus string) (string, bool, error) {
	tx, err := pool.Begin(ctx)
	if err != nil {
		return "", false, err
	}
	defer tx.Rollback(ctx)
	if err := lockKey(ctx, tx, "billing_account", owner); err != nil {
		return "", false, err
	}
	var old string
	err = tx.QueryRow(ctx, `SELECT status FROM billing_account WHERE owner_user_id = $1`, owner).Scan(&old)
	if errors.Is(err, pgx.ErrNoRows) {
		return "", false, nil
	}
	if err != nil {
		return "", false, err
	}
	if !allowed(old) {
		return old, false, nil
	}
	if _, err := tx.Exec(ctx, `UPDATE billing_account SET status = $1, updated_at = $2 WHERE owner_user_id = $3`,
		newStatus, ubyport.UTCNow(), owner); err != nil {
		return "", false, err
	}
	if err := tx.Commit(ctx); err != nil {
		return "", false, err
	}
	return old, true, nil
}

// OperatorPause is billing.operator_pause: trial_pending or trialing -> paused,
// for PlaceForOperator only. trial_pending -> paused is the one move outside
// Transitions; SetStatus must keep refusing it. False means already paused or
// another state (Place then decides).
func OperatorPause(ctx context.Context, pool *pgxpool.Pool, owner int64, actor string) (bool, error) {
	old, moved, err := moveStatus(ctx, pool, owner, func(old string) bool {
		return old == "trial_pending" || old == "trialing"
	}, "paused")
	if err != nil || !moved {
		return false, err
	}
	if err := audit(ctx, pool, actor, "billing_status", old+"->paused", owner); err != nil {
		return false, err
	}
	return true, nil
}

// dayOf returns the civil date of t (Y-M-D only); a zero t means now.
func dayOf(t time.Time) time.Time {
	if t.IsZero() {
		t = time.Now()
	}
	return time.Date(t.Year(), t.Month(), t.Day(), 0, 0, 0, 0, time.UTC)
}

func isoDay(t time.Time) string { return t.Format("2006-01-02") }

// first10 is Python's s[:10] for an ASCII timestamp.
func first10(s string) string {
	if len(s) > 10 {
		return s[:10]
	}
	return s
}

func parseDay(s string) (time.Time, error) {
	return time.Parse("2006-01-02", first10(s))
}

func wholeMonths(start, end time.Time) int {
	months := (end.Year()-start.Year())*12 + int(end.Month()) - int(start.Month())
	if end.Day() < start.Day() {
		months--
	}
	if months < 0 {
		return 0
	}
	return months
}

// ChangeResult is the dict of billing.apply_change plus the invoice_id that
// billing_order.change adds (0 = none).
type ChangeResult struct {
	When              string `json:"when"` // now, next_period or none
	PeriodTotalBefore int    `json:"period_total_before"`
	PeriodTotalAfter  int    `json:"period_total_after"`
	Months            int    `json:"months"`
	SupplementaryCZK  int    `json:"supplementary_czk"`
	CarryCZK          int    `json:"carry_czk"`
	InvoiceID         int64  `json:"invoice_id"`
}

// normaliseValue turns a change value into what the column stores.
func normaliseValue(field string, value any) (any, error) {
	asInt := func() (int, bool) {
		switch v := value.(type) {
		case int:
			return v, true
		case int64:
			return int(v), true
		case bool:
			return flag(v), true
		}
		return 0, false
	}
	switch field {
	case "plan", "billing_interval":
		if s, ok := value.(string); ok {
			return s, nil
		}
	default:
		if n, ok := asInt(); ok {
			return n, nil
		}
	}
	return nil, fmt.Errorf("%w: %s %T", ErrBadValue, field, value)
}

// ApplyChange is billing.apply_change: preview (confirm false) or save one
// change to an ordered plan. field must be a key of ChangeColumns (anything
// else is ErrOneChange, as Python's "exactly one key" rule). value is an int
// (apartments, add-ons 0 or 1), a bool (add-ons) or a string (plan, interval).
// A rise starts now; on an annual plan it costs the difference for the
// remaining whole months (supplementary, or carried when under 50 Kc). Anything
// else waits for the next period. Issuing the invoice is not done here.
func ApplyChange(ctx context.Context, pool *pgxpool.Pool, owner int64, field string, value any,
	today time.Time, confirm bool, actor string) (ChangeResult, error) {
	column, ok := ChangeColumns[field]
	if !ok {
		return ChangeResult{}, ErrOneChange
	}
	acc, err := Account(ctx, pool, owner)
	if err != nil {
		return ChangeResult{}, err
	}
	if acc == nil || acc.Plan == "" {
		return ChangeResult{}, ErrNoOrderedPlan
	}
	val, err := normaliseValue(field, value)
	if err != nil {
		return ChangeResult{}, err
	}
	if field == "addon_whitelabel" && val.(int) != 0 && !acc.AddonWhitelabel {
		return ChangeResult{}, ErrWhitelabelUnavailable
	}
	current := stateOf(acc)
	target := current
	switch field {
	case "apartments":
		target.apartments = val.(int)
	case "plan":
		target.plan = val.(string)
	case "billing_interval":
		target.interval = val.(string)
	case "addon_ttlock":
		target.ttlock = val.(int)
	case "addon_whitelabel":
		target.whitelabel = val.(int)
	}
	before, err := current.total()
	if err != nil {
		return ChangeResult{}, err
	}
	after, err := target.total()
	if err != nil {
		return ChangeResult{}, err
	}
	when := "next_period"
	if field != "billing_interval" {
		if after == before {
			when = "none"
		} else if after > before {
			when = "now"
		}
	}
	day := dayOf(today)
	months, difference := 0, 0
	if when == "now" && acc.BillingInterval == "year" && acc.PeriodEnd != "" {
		end, err := parseDay(acc.PeriodEnd)
		if err != nil {
			return ChangeResult{}, err
		}
		months = wholeMonths(day, end)
		difference = int(math.RoundToEven(float64((after-before)*months) / 12))
	}
	supplementary := 0
	if difference >= SupplementaryMinCZK {
		supplementary = difference
	}
	result := ChangeResult{
		When: when, PeriodTotalBefore: before, PeriodTotalAfter: after, Months: months,
		SupplementaryCZK: supplementary, CarryCZK: difference - supplementary,
	}
	if !confirm || when == "none" {
		return result, nil
	}
	if when == "now" {
		// column comes from ChangeColumns, never from the caller.
		if _, err := pool.Exec(ctx, "UPDATE billing_account SET "+column+" = $1, carry_czk = carry_czk + $2, updated_at = $3 WHERE owner_user_id = $4",
			val, result.CarryCZK, ubyport.UTCNow(), owner); err != nil {
			return ChangeResult{}, err
		}
	} else {
		pending, err := pendingWith(acc.PendingChangeJSON, field, val)
		if err != nil {
			return ChangeResult{}, err
		}
		if _, err := pool.Exec(ctx, `UPDATE billing_account SET pending_change_json = $1, updated_at = $2 WHERE owner_user_id = $3`,
			pending, ubyport.UTCNow(), owner); err != nil {
			return ChangeResult{}, err
		}
	}
	if err := audit(ctx, pool, actor, "billing_change", field+" "+when, owner); err != nil {
		return ChangeResult{}, err
	}
	return result, nil
}

// pendingWith is json.dumps({**json.loads(raw or "{}"), field: val}, sort_keys=True):
// sorted keys, ", " and ": " separators, ASCII-only strings.
func pendingWith(raw, field string, val any) (string, error) {
	m := map[string]json.RawMessage{}
	if raw != "" {
		if err := json.Unmarshal([]byte(raw), &m); err != nil {
			return "", err
		}
	}
	switch v := val.(type) {
	case int:
		m[field] = json.RawMessage(strconv.Itoa(v))
	case string:
		m[field] = json.RawMessage(ubyport.PyString(v))
	}
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	parts := make([]string, 0, len(keys))
	for _, k := range keys {
		parts = append(parts, ubyport.PyString(k)+": "+string(m[k]))
	}
	return "{" + strings.Join(parts, ", ") + "}", nil
}

// pyStrip is Python str.strip(): unicode.IsSpace plus the separators
// \x1c-\x1f that Python counts as white space.
func pyStrip(s string) string {
	return strings.TrimFunc(s, func(r rune) bool {
		return unicode.IsSpace(r) || (r >= 0x1c && r <= 0x1f)
	})
}

// DiscountResult is the dict of billing.apply_discount.

// ApplyDiscount is billing.apply_discount: set or clear the operator's
// discount (0443). An issued, unpaid, unvoided invoice-kind document is
// storno'd and reissued for the same period; paid or voided ones are never
// touched. The invoice calls go through inv, in Python's order.
func ApplyDiscount(ctx context.Context, pool *pgxpool.Pool, inv Invoicer, owner int64, percent int,
	reason, actor string, today time.Time) (DiscountResult, error) {
	day := dayOf(today)
	if percent < 0 || percent > 100 {
		return DiscountResult{}, ErrDiscountOutOfRange
	}
	reason = pyStrip(reason)
	if r := []rune(reason); len(r) > 200 {
		reason = string(r[:200])
	}
	if percent > 0 && reason == "" {
		return DiscountResult{}, ErrReasonMissing
	}
	acc, err := Account(ctx, pool, owner)
	if err != nil {
		return DiscountResult{}, err
	}
	if acc == nil || acc.Plan == "" {
		return DiscountResult{}, ErrNoPlan
	}
	before, err := stateOf(acc).total()
	if err != nil {
		return DiscountResult{}, err
	}
	var storedReason any
	if reason != "" {
		storedReason = reason
	}
	if _, err := pool.Exec(ctx, `UPDATE billing_account SET discount_percent = $1, discount_reason = $2, updated_at = $3
		WHERE owner_user_id = $4`, percent, storedReason, ubyport.UTCNow(), owner); err != nil {
		return DiscountResult{}, err
	}
	if err := audit(ctx, pool, actor, "billing_discount", fmt.Sprintf("%d %% %s", percent, reason), owner); err != nil {
		return DiscountResult{}, err
	}
	type period struct {
		id       int64
		from, to string
	}
	rows, err := pool.Query(ctx, `SELECT id, period_from, period_to FROM platform_invoice
		WHERE owner_user_id = $1 AND kind = 'invoice' AND voided = 0 AND paid_on IS NULL ORDER BY id`, owner)
	if err != nil {
		return DiscountResult{}, err
	}
	var open []period
	for rows.Next() {
		var p period
		if err := rows.Scan(&p.id, &p.from, &p.to); err != nil {
			rows.Close()
			return DiscountResult{}, err
		}
		open = append(open, p)
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return DiscountResult{}, err
	}
	reissued := []int64{}
	for _, p := range open {
		if _, err := inv.Storno(ctx, p.id, fmt.Sprintf("sleva %d %% (%s)", percent, reason), day); err != nil {
			return DiscountResult{}, err
		}
		id, err := inv.IssueForPeriod(ctx, owner, p.from, p.to, day)
		if err != nil {
			return DiscountResult{}, err
		}
		reissued = append(reissued, id)
	}
	refreshed, err := Account(ctx, pool, owner)
	if err != nil {
		return DiscountResult{}, err
	}
	if refreshed == nil {
		return DiscountResult{}, fmt.Errorf("billing account %d vanished", owner)
	}
	st := stateOf(refreshed)
	st.discount = percent
	after, err := st.total()
	if err != nil {
		return DiscountResult{}, err
	}
	return DiscountResult{
		Percent: percent, Reason: reason, PeriodTotalBefore: before, PeriodTotalAfter: after,
		ReissuedInvoiceIDs: reissued,
	}, nil
}

// ICOOK is validation.ico_ok: a Czech IC, 8 ASCII digits whose last is the
// mod-11 check digit. (Python also accepts non-ASCII decimal digits; Go does
// not, a known gap.)
func ICOOK(ico string) bool {
	text := pyStrip(ico)
	if len(text) != 8 {
		return false
	}
	weights := []int{8, 7, 6, 5, 4, 3, 2}
	total := 0
	for i, w := range weights {
		d := text[i]
		if d < '0' || d > '9' {
			return false
		}
		total += int(d-'0') * w
	}
	last := text[7]
	if last < '0' || last > '9' {
		return false
	}
	rem := total % 11
	check := 11 - rem
	if rem == 0 {
		check = 1
	} else if rem == 1 {
		check = 0
	}
	return int(last-'0') == check
}

// PeriodEnd is billing_trial.period_end: the last day of a month or year that
// starts on start.
func PeriodEnd(start time.Time, interval string) time.Time {
	start = dayOf(start)
	var year int
	var month time.Month
	if interval == "year" {
		year, month = start.Year()+1, start.Month()
	} else {
		year, month = start.Year()+int(start.Month())/12, time.Month(int(start.Month())%12+1)
	}
	last := time.Date(year, month+1, 0, 0, 0, 0, 0, time.UTC).Day()
	if start.Day() > last {
		// 31 January runs to the end of February.
		return time.Date(year, month, last, 0, 0, 0, 0, time.UTC)
	}
	return time.Date(year, month, start.Day(), 0, 0, 0, 0, time.UTC).AddDate(0, 0, -1)
}

// Place is billing_order.place: save the order. In a trial it starts when the
// trial ends; when paused it starts today and the first invoice is issued and
// mailed at once. actor is written on the order and status audit rows (Python
// writes "host"; pass HostActor or the signed-in username).
func Place(ctx context.Context, pool *pgxpool.Pool, inv Invoicer, owner int64, v OrderValues,
	today time.Time, actor string) (PlaceResult, error) {
	acc, err := Account(ctx, pool, owner)
	if err != nil {
		return PlaceResult{}, err
	}
	if acc == nil {
		return PlaceResult{}, ErrNoAccount
	}
	if !contains(Orderable, acc.Status) {
		return PlaceResult{}, ErrAlreadyOrdered
	}
	if v.AddonWhitelabel && !acc.AddonWhitelabel {
		return PlaceResult{}, ErrWhitelabelUnavailable
	}
	day := dayOf(today)
	apartments, err := Apartments(ctx, pool, owner)
	if err != nil {
		return PlaceResult{}, err
	}
	if _, err := pool.Exec(ctx, `UPDATE billing_account SET plan = $1, billing_interval = $2, addon_ttlock = $3,
		addon_whitelabel = $4, apartments_billed = $5, pending_change_json = NULL,
		buyer_name = $6, buyer_street = $7, buyer_city = $8, buyer_zip = $9, buyer_ico = $10,
		buyer_dic = $11, buyer_email = $12, updated_at = $13 WHERE owner_user_id = $14`,
		v.Plan, v.BillingInterval, flag(v.AddonTTLock), flag(acc.AddonWhitelabel), apartments,
		nullable(v.BuyerName), nullable(v.BuyerStreet), nullable(v.BuyerCity), nullable(v.BuyerZip),
		nullable(v.BuyerICO), nullable(v.BuyerDIC), nullable(v.BuyerEmail), ubyport.UTCNow(), owner); err != nil {
		return PlaceResult{}, err
	}
	if err := audit(ctx, pool, actor, "billing_order", v.Plan+" "+v.BillingInterval, owner); err != nil {
		return PlaceResult{}, err
	}
	if acc.Status != "paused" {
		return PlaceResult{Starts: "trial_end", Date: first10(acc.TrialEndsAt)}, nil
	}
	end := PeriodEnd(day, v.BillingInterval)
	if _, err := pool.Exec(ctx, `UPDATE billing_account SET period_start = $1, period_end = $2, updated_at = $3
		WHERE owner_user_id = $4`, isoDay(day), isoDay(end), ubyport.UTCNow(), owner); err != nil {
		return PlaceResult{}, err
	}
	if _, err := SetStatus(ctx, pool, owner, "active", actor); err != nil {
		return PlaceResult{}, err
	}
	id, err := inv.IssueForPeriod(ctx, owner, isoDay(day), isoDay(end), day)
	if err != nil {
		return PlaceResult{}, err
	}
	if err := inv.Send(ctx, id); err != nil {
		return PlaceResult{}, err
	}
	return PlaceResult{Starts: "now", Date: isoDay(day), InvoiceID: id}, nil
}

// ExtendTrial is billing_order.extend_trial: move a running trial's end date
// (0448). It returns the new YYYY-MM-DD, or "" when refused (no account, not
// trialing, no end date). days outside 1..365 is ErrBadDays.
func ExtendTrial(ctx context.Context, pool *pgxpool.Pool, owner int64, days int, actor string,
	today time.Time) (string, error) {
	day := dayOf(today)
	if days <= 0 || days > 365 {
		return "", ErrBadDays
	}
	acc, err := Account(ctx, pool, owner)
	if err != nil {
		return "", err
	}
	if acc == nil || acc.Status != "trialing" || acc.TrialEndsAt == "" {
		return "", nil
	}
	current, err := parseDay(acc.TrialEndsAt)
	if err != nil {
		return "", err
	}
	if day.After(current) {
		current = day
	}
	newEnd := isoDay(current.AddDate(0, 0, days))
	if _, err := pool.Exec(ctx, `UPDATE billing_account SET trial_ends_at = $1, updated_at = $2 WHERE owner_user_id = $3`,
		newEnd, ubyport.UTCNow(), owner); err != nil {
		return "", err
	}
	if err := audit(ctx, pool, actor, "billing_trial_extended", fmt.Sprintf("+%dd -> %s", days, newEnd), owner); err != nil {
		return "", err
	}
	return newEnd, nil
}

func dbValue(v any) any {
	switch x := v.(type) {

func insertRow(t *testing.T, pool *pgxpool.Pool, table string, row map[string]any) {
	t.Helper()
	cols := make([]string, 0, len(row))
	for c := range row {
		cols = append(cols, c)
	}
	sort.Strings(cols)
	sql := "INSERT INTO " + table + " ("
	marks := ""
	args := make([]any, 0, len(cols))
	for i, c := range cols {
		if i > 0 {
			sql += ", "
			marks += ", "
		}
		sql += c
		marks += fmt.Sprintf("$%d", i+1)
		args = append(args, dbValue(row[c]))
	}
	if _, err := pool.Exec(context.Background(), sql+") VALUES ("+marks+")", args...); err != nil {
		t.Fatalf("insert %s: %v", table, err)
	}
}

// rawAccount reads the row with NULLs kept, in the golden "account_after" form.
func rawAccount(t *testing.T, pool *pgxpool.Pool, owner int64, stamp string) any {
	t.Helper()
	cols := []string{"status", "plan", "billing_interval", "payment_method", "founder", "addon_ttlock",
		"addon_whitelabel", "apartments_billed", "trial_started_at", "trial_ends_at", "period_start",
		"period_end", "cancel_at_period_end", "pending_change_json", "carry_czk", "buyer_name",
		"buyer_street", "buyer_city", "buyer_zip", "buyer_ico", "buyer_dic", "buyer_email",
		"discount_percent", "discount_reason"}
	ints := map[string]bool{"founder": true, "addon_ttlock": true, "addon_whitelabel": true,
		"apartments_billed": true, "cancel_at_period_end": true, "carry_czk": true, "discount_percent": true}
	list := ""
	for _, c := range cols {
		list += c + ", "
	}
	rows, err := pool.Query(context.Background(), "SELECT owner_user_id, "+list+"updated_at FROM billing_account WHERE owner_user_id = $1", owner)
	if err != nil {
		t.Fatal(err)
	}
	defer rows.Close()
	if !rows.Next() {
		return nil
	}
	vals, err := rows.Values()
	if err != nil {
		t.Fatal(err)
	}
	out := map[string]any{"owner_user_id": vals[0]}
	for i, c := range cols {
		out[c] = vals[i+1]
		if ints[c] && vals[i+1] == nil {
			out[c] = int64(0)
		}
	}
	updated := vals[len(cols)+1].(string)
	if updated == stamp {
		out["updated_at"] = "seed"
	} else {
		if !stampRE.MatchString(updated) {
			t.Errorf("updated_at %q is not db.utcnow form", updated)
		}
		out["updated_at"] = "run"
	}
	return roundTrip(t, out)
}

func auditRows(t *testing.T, pool *pgxpool.Pool) []map[string]any {
	t.Helper()
	rows, err := pool.Query(context.Background(),
		"SELECT at, action, detail, actor, owner_user_id, actor_user_id, impersonator_user_id FROM audit ORDER BY id")
	if err != nil {
		t.Fatal(err)
	}
	defer rows.Close()
	out := []map[string]any{}
	for rows.Next() {
		var at, action, detail, actor string
		var owner int64
		var actorID, impID *int64
		if err := rows.Scan(&at, &action, &detail, &actor, &owner, &actorID, &impID); err != nil {
			t.Fatal(err)
		}
		if !stampRE.MatchString(at) {
			t.Errorf("audit row %s: at %q is not db.utcnow form", action, at)
		}
		out = append(out, map[string]any{"action": action, "detail": detail, "actor": actor,
			"owner_user_id": owner, "actor_user_id": actorID, "impersonator_user_id": impID})
	}
	return out
}

// runDB runs one database scenario through the Go engine. The returned value is
// in the golden form; a billing error comes back as its code.
func runDB(t *testing.T, pool *pgxpool.Pool, inv *fakeInvoicer, g goldenFile, sc goldenScenario) (any, string) {
	t.Helper()
	ctx := context.Background()
	if c := sc.ActorCtx; c != nil {
		var imp *int64
		if c.Impersonating {
			imp = &c.UserID
		}
		ctx = WithAuditActor(ctx, c.UserID, c.Username, imp)
	}
	a := sc.Call.Args
	owner := sc.Owner
	today := parseDate(t, sc.Today)
	fail := func(err error) (any, string) {
		if c := Code(err); c != "" {
			return nil, c
		}
		t.Fatalf("%s: %v", sc.Name, err)
		return nil, ""
	}
	switch sc.Call.Fn {
	case "account":
		acc, err := Account(ctx, pool, owner)
		if err != nil {
			return fail(err)
		}
		return roundTrip(t, acctMap(acc, g.Stamp)), ""
	case "ensure_account":
		acc, err := EnsureAccount(ctx, pool, owner)
		if err != nil {
			return fail(err)
		}
		return roundTrip(t, acctMap(acc, g.Stamp)), ""
	case "apartment_count":
		n, err := ApartmentCount(ctx, pool, owner)
		if err != nil {
			return fail(err)
		}
		return roundTrip(t, n), ""
	case "set_status":
		ok, err := SetStatus(ctx, pool, owner, a["new"].(string), a["actor"].(string))
		if err != nil {
			return fail(err)
		}
		return ok, ""
	case "operator_pause":
		ok, err := OperatorPause(ctx, pool, owner, a["actor"].(string))
		if err != nil {
			return fail(err)
		}
		return ok, ""
	case "apply_change", "change":
		var field string
		var value any
		confirm := a["confirm"].(bool)
		if sc.Call.Fn == "apply_change" {
			change := a["change"].(map[string]any)
			// Python's "exactly one key" rule: any other size has no field.
			for k, v := range change {
				if len(change) == 1 {
					field = k
					switch x := v.(type) {
					case string:
						value = x
					case float64:
						value = int(x)
					}
				}
			}
		} else {
			field, value = a["field"].(string), nil
			switch v := a["value"].(type) {
			case string:
				value = v
			case float64:
				value = int(v)
			}
		}
		var res ChangeResult
		var err error
		if sc.Call.Fn == "apply_change" {
			res, err = ApplyChange(ctx, pool, owner, field, value, today, confirm, a["actor"].(string))
		} else {
			res, err = Change(ctx, pool, inv, owner, field, value, confirm, today, HostActor)
		}
		if err != nil {
			return fail(err)
		}
		m := roundTrip(t, res).(map[string]any)
		if sc.Call.Fn == "apply_change" {
			delete(m, "invoice_id")
		} else if res.InvoiceID == 0 {
			m["invoice_id"] = nil
		}
		return m, ""
	case "apply_discount":
		res, err := ApplyDiscount(ctx, pool, inv, owner, int(a["percent"].(float64)), a["reason"].(string), a["actor"].(string), today)
		if err != nil {
			return fail(err)
		}
		return roundTrip(t, res), ""
	case "place", "place_for_operator":
		raw, _ := json.Marshal(a["values"])
		var values OrderValues
		if err := json.Unmarshal(raw, &values); err != nil {
			t.Fatal(err)
		}
		var res PlaceResult
		var err error
		if sc.Call.Fn == "place" {
			res, err = Place(ctx, pool, inv, owner, values, today, HostActor)
		} else {
			res, err = PlaceForOperator(ctx, pool, inv, owner, values, today, a["actor"].(string))
		}
		if err != nil {
			return fail(err)
		}
		m := roundTrip(t, res).(map[string]any)
		if res.InvoiceID == 0 {
			m["invoice_id"] = nil
		}
		return m, ""
	case "cancel":
		ok, err := Cancel(ctx, pool, owner, HostActor)
		if err != nil {
			return fail(err)
		}
		return ok, ""
	case "resume":
		ok, err := Resume(ctx, pool, owner, HostActor)
		if err != nil {
			return fail(err)
		}
		return ok, ""
	case "extend_trial":
		s, err := ExtendTrial(ctx, pool, owner, int(a["days"].(float64)), a["actor"].(string), today)
		if err != nil {
			return fail(err)
		}
		return s, ""
	}
	t.Fatalf("%s: unknown fn %s", sc.Name, sc.Call.Fn)
	return nil, ""
}

			}
			if sc.Error == "" && !reflect.DeepEqual(got, sc.Result) {
				t.Errorf("result\n got  %v\n want %v", got, sc.Result)
			}
			if after := rawAccount(t, pool, sc.Owner, g.Stamp); !reflect.DeepEqual(after, sc.AccountAft) {
				t.Errorf("account after\n got  %v\n want %v", after, sc.AccountAft)

				t.Errorf("audit\n got  %v\n want %v", audit, want)
			}
			calls := roundTrip(t, inv.calls)
			if inv.calls == nil {
				calls = roundTrip(t, [][]any{})
			}
			if want := roundTrip(t, sc.Invoicer); !reflect.DeepEqual(calls, want) {
				t.Errorf("invoicer calls\n got  %v\n want %v", calls, want)

		})

	}

def _row(owner):
    from app import db

    row = db.query_one("SELECT * FROM billing_account WHERE owner_user_id = ?", (owner,))
    return dict(row) if row else None


def _acct(row, collapse):
    """The comparable columns of a billing_account row."""
    if row is None:
        return None
    out = {"owner_user_id": int(row["owner_user_id"])}
    for col in ACCOUNT_COLUMNS:
        value = row[col]
        if value is None:
            value = 0 if col in INT_COLUMNS else ("" if collapse else None)
        out[col] = int(value) if col in INT_COLUMNS else value
    out["updated_at"] = "seed" if row["updated_at"] == STAMP else "run"
    return out


def _reset(db) -> None:
    for table in ("audit", "platform_invoice", "billing_account", "apartment"):
        db.execute(f"DELETE FROM {table}")


def _seed(db, sc) -> None:
    owner = sc["owner"]
    acc = sc["account"]
    if acc is not None:
        cols = ["owner_user_id", "updated_at"] + list(acc)
        vals = [owner, STAMP] + list(acc.values())
        marks = ", ".join("?" for _ in cols)
        db.execute(f"INSERT INTO billing_account ({', '.join(cols)}) VALUES ({marks})", vals)
    active, inactive = sc["apartments"]
    aid = 0
    for flag, count in ((1, active), (0, inactive)):
        for _ in range(count):
            aid += 1
            db.execute(
                "INSERT INTO apartment (id, owner_user_id, internal_name, active, created_at) VALUES (?, ?, ?, ?, ?)",
                (aid, owner, f"flat {aid}", flag, STAMP),
            )
    if active:
        # An apartment of another host never counts.
        db.execute(
            "INSERT INTO apartment (id, owner_user_id, internal_name, active, created_at) VALUES (?, ?, ?, 1, ?)",
            (900, 2 if owner == 1 else 1, "foreign", STAMP),
        )
    for inv in sc["invoices"]:
        marks = ", ".join("?" for _ in INVOICE_COLUMNS)
        db.execute(
            f"INSERT INTO platform_invoice ({', '.join(INVOICE_COLUMNS)}) VALUES ({marks})",
            [inv[c] for c in INVOICE_COLUMNS],
        )


class _Recorder:
    def __init__(self):
        self.calls = []
        self.issued = 0
        self.supplementary = 0

    def issue_for_period(self, owner, period_from, period_to, *, today):
        self.issued += 1
        self.calls.append(["issue_for_period", owner, period_from, period_to, today.isoformat()])
        return 1000 + self.issued

    def issue_supplementary(self, owner, *, amount_czk, description, period_from, period_to, today):
        self.supplementary += 1
        self.calls.append(["issue_supplementary", owner, amount_czk, description,
                           period_from, period_to, today.isoformat()])
        return 2000 + self.supplementary

    def storno(self, invoice_id, *, reason, today):
        self.calls.append(["storno", invoice_id, reason, today.isoformat()])
        return 3000 + invoice_id

    def send(self, invoice_id):
        self.calls.append(["send", invoice_id])
        return None


def _run(sc, rec):
    from app import billing, billing_order, billing_trial

    fn = sc["call"]["fn"]
    a = sc["call"]["args"]
    day = date.fromisoformat(sc["today"])
    if fn == "account":
        return _acct(billing.account(a["owner"]), True)
    if fn == "ensure_account":
        return _acct(billing.ensure_account(a["owner"]), True)
    if fn == "apartment_count":
        return billing.apartment_count(a["owner"])
    if fn == "period_end":
        return billing_trial.period_end(date.fromisoformat(a["start"]), a["interval"]).isoformat()
    if fn == "set_status":
        return billing.set_status(a["owner"], a["new"], actor=a["actor"])
    if fn == "operator_pause":
        return billing.operator_pause(a["owner"], actor=a["actor"])
    if fn == "apply_change":
        return billing.apply_change(a["owner"], a["change"], today=day, confirm=a["confirm"], actor=a["actor"])
    if fn == "apply_discount":
        return billing.apply_discount(a["owner"], a["percent"], reason=a["reason"], actor=a["actor"], today=day)
    if fn == "place":
        return billing_order.place(a["owner"], a["values"], today=day)
    if fn == "place_for_operator":
        return billing_order.place_for_operator(a["owner"], a["values"], actor=a["actor"], today=day)
    if fn == "change":
        return billing_order.change(a["owner"], a["field"], a["value"], confirm=a["confirm"], today=day)
    if fn == "cancel":
        return billing_order.cancel(a["owner"])
    if fn == "resume":
        return billing_order.resume(a["owner"])
    if fn == "extend_trial":
        return billing_order.extend_trial(a["owner"], a["days"], actor=a["actor"], today=day)
    if fn == "read_form":
        values, errors = billing_order.read_form(a["form"])
        return {"values": values, "errors": errors}
    raise SystemExit(f"unknown fn {fn}")


def render() -> str:
    """Run every scenario on a fresh database and return the golden JSON text."""
    from app import billing_invoices, db

    db.init_db()
    for user in USERS:
        db.execute(
            "INSERT INTO user_account (id, username, password_hash, role, created_at) VALUES (?, ?, 'x', 'host', ?)",
            (user["id"], user["username"], STAMP),
        )
    names = [sc["name"] for sc in SCENARIOS]
    assert len(names) == len(set(names)), "duplicate scenario name"
    out = []
    for sc in SCENARIOS:
        _reset(db)
        _seed(db, sc)
        rec = _Recorder()
        saved = {n: getattr(billing_invoices, n) for n in ("issue_for_period", "issue_supplementary", "storno", "send")}
        for n in saved:
            setattr(billing_invoices, n, getattr(rec, n))
        entry = {k: sc[k] for k in ("name", "owner", "today", "account", "apartments", "invoices", "call",
                                    "actor_ctx")}
        ctx = sc["actor_ctx"]
        if ctx:
            db.set_current_actor(ctx["user_id"], ctx["username"], impersonating=ctx["impersonating"])
        try:
            entry["result"] = _run(sc, rec)
        except ValueError as exc:
            entry["error"] = str(exc).split(":")[0].strip()
        finally:
            db.set_current_actor(None)
            for n, f in saved.items():
                setattr(billing_invoices, n, f)
        entry["account_after"] = _acct(_row(sc["owner"]), False)
        entry["audit"] = [
            {"action": r["action"], "detail": r["detail"], "actor": r["actor"],
             "owner_user_id": r["owner_user_id"], "actor_user_id": r["actor_user_id"],
             "impersonator_user_id": r["impersonator_user_id"]}
            for r in db.query(
                "SELECT action, detail, actor, owner_user_id, actor_user_id, impersonator_user_id "
                "FROM audit ORDER BY id")
        ]
        entry["invoicer"] = rec.calls
        out.append(entry)
    from app import billing, billing_order

    tables = {
        "statuses": list(billing.STATUSES),
        "transitions": {k: sorted(v) for k, v in billing.TRANSITIONS.items()},
        "plans": list(billing.PLANS),
        "intervals": list(billing.INTERVALS),
        "orderable": list(billing_order.ORDERABLE),
        "buyer_fields": list(billing_order.BUYER_FIELDS),
        "change_columns": dict(billing.CHANGE_COLUMNS),
        "supplementary_min_czk": billing.SUPPLEMENTARY_MIN_CZK,
    }
    golden = {"stamp": STAMP, "users": USERS, "tables": tables, "scenarios": out}
    return json.dumps(golden, ensure_ascii=False, indent=1, sort_keys=True) + "\n"
