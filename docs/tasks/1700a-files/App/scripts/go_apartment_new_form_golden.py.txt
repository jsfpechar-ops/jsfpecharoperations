Run from App/ (create the venv first if missing:
python3 -m venv .venv && .venv/bin/pip install -q -r requirements.txt):

    .venv/bin/python scripts/go_apartment_new_form_golden.py --check  exit 1 when it is stale

case in with auth.issue_session and renders GET /apartments/new (with the query

<title> text and visible_lines() of the page area (div.ui-page): the visible
text and the key attributes, not the bytes.
go/internal/web/properties/apartment_new_test.go seeds PostgreSQL from the same
SEED, renders the Go page and compares with the same normaliser.
App/tests/test_go_apartment_new_form_golden.py fails when this file is stale.
Brief 1700a.
"""

# visible_lines: keep in step with apartmentNewVisibleLines in apartment_new_test.go.
DROP = [re.compile(p, re.S | re.I) for p in (
    r"<!--.*?-->", r"<script\b.*?</script>", r"<style\b.*?</style>", r"<svg\b.*?</svg>")]
TOKEN = re.compile(r"<[^>]*>|[^<]+")
TAG = re.compile(r"^<[ \t\n\r\f\v]*(/?)([A-Za-z][A-Za-z0-9]*)")
ATTR = re.compile(
    r"""([^ \t\n\r\f\v=/>"']+)(?:[ \t\n\r\f\v]*=[ \t\n\r\f\v]*(?:"([^"]*)"|'([^']*)'|([^ \t\n\r\f\v>"']+)))?""")
SPACE = re.compile(r"[ \t\n\r\f\v]+")
TITLE = re.compile(r"<title>(.*?)</title>", re.S | re.I)
KEPT = {"id", "href", "action", "method", "name", "value", "type", "selected", "checked", "required", "for",
        "aria-current", "class", "aria-label", "aria-controls", "aria-describedby", "aria-invalid", "autofocus",
        "maxlength", "inputmode", "autocapitalize", "placeholder", "open", "target", "src",
        "data-ruian-suggest", "data-ruian-loading", "data-ruian-none"}


def title_of(page: str) -> str:
    m = TITLE.search(page)
    return SPACE.sub(" ", html.unescape(m.group(1))).strip(" ") if m else ""


def visible_lines(page: str) -> list[str]:
    """The page area as lines: each text run, and each tag that has a KEPT attribute."""
    start = page.find('<div class="ui-page"')
    if start < 0:
        return []
    page = page[start:]
    end = page.find('<div class="page-skeleton"')
    if end >= 0:
        page = page[:end]
    # A script with a src is kept as a "scriptsrc" tag (DROP needs "script" followed by a word break).
    page = re.sub(r"<script\b([^>]*\bsrc=[^>]*)>\s*</script>", r"<scriptsrc\1>", page, flags=re.S | re.I)
    for pattern in DROP:
        page = pattern.sub("", page)
    out = []
    for tok in TOKEN.findall(page):
        if not tok.startswith("<"):
            text = SPACE.sub(" ", html.unescape(tok)).strip(" ")
            if text:
                out.append(text)
            continue
        m = TAG.match(tok)
        if not m or m.group(1) == "/":
            continue
        name = m.group(2).lower()
        attrs = {}
        for a in ATTR.finditer(tok[m.end():]):
            key = a.group(1).lower()
            has_value = "=" in a.group(0)
            value = html.unescape((a.group(2) or "") + (a.group(3) or "") + (a.group(4) or ""))
            if key == "class":
                value = " ".join(value.split())
            attrs[key] = (has_value, value)
        if attrs.get("name", (False, ""))[1] == "_csrf":
            attrs["value"] = (True, "*")
        parts = [k + ("=" + v if has else "") for k, (has, v) in sorted(attrs.items()) if k in KEPT]
        if parts:
            out.append("<" + name + " " + " ".join(parts) + ">")
    return out


def _seed(db) -> None:
    for table in SEED:
        cols = ", ".join(table["columns"])
        marks = ", ".join("?" for _ in table["columns"])
        for row in table["rows"]:
            db.execute(f"INSERT INTO {table['table']} ({cols}) VALUES ({marks})", row)


def render() -> str:
    from fastapi.testclient import TestClient

    from app import auth, db
    from app.main import app

    db.init_db()
    _seed(db)
    client = TestClient(app)
    cases = []
    for name, user, workspace, lang, query in CASES:
        client.cookies.clear()
        cookies = {"ubyhost_lang": lang}
        if user:
            cookies[auth.SESSION_COOKIE] = auth.issue_session(
                user, 1, workspace_user_id=workspace or None,
                impersonation_started_at=int(time.time()) if workspace else None)
        header = "; ".join(f"{k}={v}" for k, v in cookies.items())
        response = client.get("/apartments/new" + query, headers={"Cookie": header}, follow_redirects=False)
        ok = response.status_code == 200
        cases.append({"name": name, "user": user, "workspace": workspace, "lang": lang, "query": query,
                      "status": response.status_code, "location": response.headers.get("location"),
                      "title": title_of(response.text) if ok else "",
                      "lines": visible_lines(response.text) if ok else []})
    golden = {"versions": VERSIONS, "seed": SEED, "cases": cases}
    return json.dumps(golden, ensure_ascii=False, indent=1, sort_keys=True) + "\n"


def main(argv: list[str]) -> int:
    text = render()
    if "--check" in argv:
        if OUT.exists() and OUT.read_text(encoding="utf-8") == text:
            return 0
        print(f"stale: {OUT}; run scripts/go_apartment_new_form_golden.py")
        return 1
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(text, encoding="utf-8")
    print(f"wrote {OUT}")
    return 0
