const apartmentNewSecret = "golden-secret-key-not-for-real-use"

type apartmentNewSeedTable struct {
	Table   string   `json:"table"`
	Columns []string `json:"columns"`
	Rows    [][]any  `json:"rows"`
}

type apartmentNewCase struct {
	Name      string   `json:"name"`
	User      int64    `json:"user"`
	Workspace int64    `json:"workspace"`
	Lang      string   `json:"lang"`
	Query     string   `json:"query"`
	Status    int      `json:"status"`
	Location  *string  `json:"location"`
	Title     string   `json:"title"`
	Lines     []string `json:"lines"`
}

type apartmentNewGolden struct {
	Seed  []apartmentNewSeedTable `json:"seed"`
	Cases []apartmentNewCase      `json:"cases"`
}

func loadApartmentNewGolden(t *testing.T) apartmentNewGolden {
	t.Helper()
	raw, err := os.ReadFile("testdata/apartment_new_form_golden.json")
	if err != nil {
		t.Fatal(err)
	}
	dec := json.NewDecoder(strings.NewReader(string(raw)))
	dec.UseNumber()
	var g apartmentNewGolden
	if err := dec.Decode(&g); err != nil {
		t.Fatal(err)
	}
	if len(g.Cases) < 15 {
		t.Fatal("apartment new form golden file is too small")
	}
	return g
}

// apartmentNewSeed writes the golden SEED to PostgreSQL.
func apartmentNewSeed(t *testing.T, pool *pgxpool.Pool, g apartmentNewGolden) {
	t.Helper()
	for _, table := range g.Seed {
		marks := make([]string, len(table.Columns))
		for i := range marks {
			marks[i] = fmt.Sprintf("$%d", i+1)
		}
		sql := fmt.Sprintf("INSERT INTO %s (%s) VALUES (%s)", table.Table, strings.Join(table.Columns, ", "), strings.Join(marks, ", "))
		for _, row := range table.Rows {
			args := make([]any, len(row))
			for i, v := range row {
				args[i] = sqlValue(v)
			}
			if _, err := pool.Exec(context.Background(), sql, args...); err != nil {
				t.Fatalf("seed %s: %v", table.Table, err)
			}
		}
	}
}

// apartmentNewVisibleLines is visible_lines in
// App/scripts/go_apartment_new_form_golden.py; keep the two in step.
var (
	apartmentNewDrop = []*regexp.Regexp{regexp.MustCompile(`(?s)<!--.*?-->`), regexp.MustCompile(`(?is)<script\b.*?</script>`),
		regexp.MustCompile(`(?is)<style\b.*?</style>`), regexp.MustCompile(`(?is)<svg\b.*?</svg>`)}
	apartmentNewScript = regexp.MustCompile(`(?is)<script\b([^>]*\bsrc=[^>]*)>\s*</script>`)
	apartmentNewToken  = regexp.MustCompile(`<[^>]*>|[^<]+`)
	apartmentNewTag    = regexp.MustCompile(`^<[ \t\n\r\f\v]*(/?)([A-Za-z][A-Za-z0-9]*)`)
	apartmentNewAttr   = regexp.MustCompile(`([^ \t\n\r\f\v=/>"']+)(?:[ \t\n\r\f\v]*=[ \t\n\r\f\v]*(?:"([^"]*)"|'([^']*)'|([^ \t\n\r\f\v>"']+)))?`)
	apartmentNewSpace  = regexp.MustCompile(`[ \t\n\r\f\v]+`)
	apartmentNewTitle  = regexp.MustCompile(`(?is)<title>(.*?)</title>`)
	apartmentNewKept   = map[string]bool{"id": true, "href": true, "action": true, "method": true, "name": true, "value": true,
		"type": true, "selected": true, "checked": true, "required": true, "for": true, "aria-current": true,
		"class": true, "aria-label": true, "aria-controls": true, "aria-describedby": true, "aria-invalid": true,
		"autofocus": true, "maxlength": true, "inputmode": true, "autocapitalize": true, "placeholder": true,
		"open": true, "target": true, "src": true, "data-ruian-suggest": true, "data-ruian-loading": true,
		"data-ruian-none": true}
)

func apartmentNewTitleOf(page string) string {
	m := apartmentNewTitle.FindStringSubmatch(page)
	if m == nil {
		return ""
	}
	return strings.Trim(apartmentNewSpace.ReplaceAllString(html.UnescapeString(m[1]), " "), " ")
}

func apartmentNewVisibleLines(page string) []string {
	start := strings.Index(page, `<div class="ui-page"`)
	if start < 0 {
		return []string{}
	}
	page = page[start:]
	if end := strings.Index(page, `<div class="page-skeleton"`); end >= 0 {
		page = page[:end]
	}
	page = apartmentNewScript.ReplaceAllString(page, "<scriptsrc${1}>")
	for _, re := range apartmentNewDrop {
		page = re.ReplaceAllString(page, "")
	}
	out := []string{}
	type attrVal struct {
		has   bool
		value string
	}
	for _, tok := range apartmentNewToken.FindAllString(page, -1) {
		if !strings.HasPrefix(tok, "<") {
			text := strings.Trim(apartmentNewSpace.ReplaceAllString(html.UnescapeString(tok), " "), " ")
			if text != "" {
				out = append(out, text)
			}
			continue
		}
		m := apartmentNewTag.FindStringSubmatchIndex(tok)
		if m == nil || tok[m[2]:m[3]] == "/" {
			continue
		}
		name := strings.ToLower(tok[m[4]:m[5]])
		attrs := map[string]attrVal{}
		for _, a := range apartmentNewAttr.FindAllStringSubmatch(tok[m[1]:], -1) {
			key := strings.ToLower(a[1])
			v := html.UnescapeString(a[2] + a[3] + a[4])
			if key == "class" {
				v = strings.Join(strings.Fields(v), " ")
			}
			attrs[key] = attrVal{strings.Contains(a[0], "="), v}
		}
		if attrs["name"].value == "_csrf" {
			attrs["value"] = attrVal{true, "*"}
		}
		keys := make([]string, 0, len(attrs))
		for k := range attrs {
			if apartmentNewKept[k] {
				keys = append(keys, k)
			}
		}
		sort.Strings(keys)
		parts := make([]string, len(keys))
		for i, k := range keys {
			parts[i] = k
			if attrs[k].has {
				parts[i] += "=" + attrs[k].value
			}
		}
		if len(parts) > 0 {
			out = append(out, "<"+name+" "+strings.Join(parts, " ")+">")
		}
	}
	return out
}

func TestApartmentNewVisibleLines(t *testing.T) {
	page := `<p>x</p><title> A  &amp; B </title><div class="ui-page"><!-- c --><a class=" a  b " href="/b?a=1&amp;b=2"><span>Back</span></a>` +
		`<input type="hidden" name="_csrf" value="tok"><input required name=q data-ruian-suggest>` +
		`<script src="/s.js?v=1" defer></script><script>var x = 1;</script><svg><path d="M0"/></svg></div>` +
		`<div class="page-skeleton">skip</div>`
	want := []string{"<div class=ui-page>", "<a class=a b href=/b?a=1&b=2>", "Back", "<input name=_csrf type=hidden value=*>",
		"<input data-ruian-suggest name=q required>", "<scriptsrc src=/s.js?v=1>"}
	if got := apartmentNewVisibleLines(page); strings.Join(got, "\n") != strings.Join(want, "\n") {
		t.Fatalf("got %q", got)
	}
	if got := apartmentNewTitleOf(page); got != "A & B" {
		t.Fatalf("title %q", got)
	}
}

func apartmentNewGetenv(k string) string {
	return map[string]string{"UBYHOST_SECRET_KEY": apartmentNewSecret}[k]
}

func apartmentNewMux(pool *pgxpool.Pool) *http.ServeMux {
	mux := http.NewServeMux()

// apartmentNewSignIn adds the language and session cookies of user (0: signed
// out), previewing workspace when it is not 0.
func apartmentNewSignIn(req *http.Request, user, workspace int64, lang string) {
	req.AddCookie(&http.Cookie{Name: "ubyhost_lang", Value: lang})
	if user != 0 {
		payload := map[string]any{"uid": user, "sv": 1}
		if workspace != 0 {
			payload["as"] = workspace
			payload["ast"] = testNow.Unix()
		}
		req.AddCookie(&http.Cookie{Name: session.CookieName, Value: session.Sign(apartmentNewSecret, payload, testNow)})
	}
}

// TestApartmentNewMatchesPython renders each golden case with the Go handler
// and compares the title and the visible lines with the ones Python rendered.
func TestApartmentNewMatchesPython(t *testing.T) {
	g := loadApartmentNewGolden(t)
	pool := testPool(t)
	apartmentNewSeed(t, pool, g)
	mux := apartmentNewMux(pool)
	for _, c := range g.Cases {
		req := httptest.NewRequest(http.MethodGet, "/apartments/new"+c.Query, nil)
		apartmentNewSignIn(req, c.User, c.Workspace, c.Lang)
		rec := httptest.NewRecorder()
		mux.ServeHTTP(rec, req)
		if rec.Code != c.Status {
			t.Errorf("%s: status %d, Python %d", c.Name, rec.Code, c.Status)
			continue
		}
		if c.Location != nil && rec.Header().Get("Location") != *c.Location {
			t.Errorf("%s: Location %q, Python %q", c.Name, rec.Header().Get("Location"), *c.Location)
		}
		if rec.Code == http.StatusNotFound && rec.Body.String() != "Not found." {
			t.Errorf("%s: 404 body %q", c.Name, rec.Body.String())
		}
		if rec.Code != http.StatusOK {
			continue
		}
		body := rec.Body.String()
		if got := apartmentNewTitleOf(body); got != c.Title {
			t.Errorf("%s: title %q, Python %q", c.Name, got, c.Title)
		}
		got := apartmentNewVisibleLines(body)
		if strings.Count(body, `name="_csrf"`) < 1 {
			t.Errorf("%s: the create form has no _csrf field", c.Name)
		}
		for i := 0; i < len(got) || i < len(c.Lines); i++ {
			var gl, wl string
			if i < len(got) {
				gl = got[i]
			}
			if i < len(c.Lines) {
				wl = c.Lines[i]
			}
			if gl != wl {
				t.Errorf("%s: line %d is %q, Python %q", c.Name, i, gl, wl)
				break
			}
		}
	}
}

// TestApartmentNewRenderRefusal draws the form the way brief 1700b does for a
// refused create: typed values, errors, copy fields and status 400.
func TestApartmentNewRenderRefusal(t *testing.T) {
	g := loadApartmentNewGolden(t)
	pool := testPool(t)
	apartmentNewSeed(t, pool, g)
	render := func(form ApartmentNewForm) *httptest.ResponseRecorder {
		h := func(w http.ResponseWriter, r *http.Request) {
			host, err := hostweb.Login(w, r, pool, apartmentNewGetenv, testNow)
			if err != nil || host == nil {
				t.Errorf("login: %v", err)
				return
			}
			form.Entities, form.Peers, err = loadApartmentNewChoices(r.Context(), pool, host.Workspace.ID)
			if err != nil {
				t.Error(err)
				return
			}
			if err := RenderApartmentNew(w, r, layout.Env{Pool: pool, Getenv: apartmentNewGetenv, Now: testNow}, host, form); err != nil {
				t.Error(err)
			}
		}
		req := httptest.NewRequest(http.MethodPost, "/apartments", nil)
		apartmentNewSignIn(req, 4, 0, "en")
		rec := httptest.NewRecorder()
		h(rec, req)
		return rec
	}
	rec := render(ApartmentNewForm{
		Status: http.StatusBadRequest, NameError: true,
		Draft:         map[string]string{"internal_name": "Typed <x>", "city_en": "Praha", "addr_zip": "1200"},
		DraftEntityID: 42, DraftControllerID: 41,
		OperatorError: "flash.error.no_such_controller",
		AddressErrors: map[string]string{"addr_zip": "Enter 5 digits."},
	})
	body := rec.Body.String()
	if rec.Code != http.StatusBadRequest {
		t.Fatalf("status %d, want 400", rec.Code)
	}
	for _, want := range []string{
		`value="Typed &lt;x&gt;"`, `aria-invalid="true" autofocus`, `id="internal_name-error"`, `id="city_en" name="city_en" value="Praha"`,
		`id="addr_zip-error" role="alert">Enter 5 digits.</p>`, `aria-describedby="addr_zip-help addr_zip-error"`,
		`<option value="42" selected>`, `id="controller-message"`, `<option value="41" selected>`, `name="_csrf"`,
	} {
		if !strings.Contains(body, want) {
			t.Errorf("refusal page lacks %q", want)
		}
	}
	if strings.Contains(body, `name="copy_from" value=`) {
		t.Error("a refusal without a copy source carries copy_from")
	}
	rec = render(ApartmentNewForm{
		Status: http.StatusBadRequest, CopyFromID: 41, CopiedFromName: "Loft <A>",
		Prefill:       map[string]string{"automation_mode": "", "stay_fee_rate_czk": "0", "stay_fee_payee": "Me"},
		OperatorError: "apartment.form.operator.choose",
	})
	body = rec.Body.String()
	for _, want := range []string{`name="copy_from" value="41"`, `name="automation_mode" value=""`, `name="stay_fee_payee" value="Me"`,
		`name="default_purpose" value="10"`, `aria-describedby="operator-message" aria-invalid="true"`, `id="operator-message" role="alert">`} {
		if !strings.Contains(body, want) {
			t.Errorf("copy refusal page lacks %q", want)
		}
	}
}

func TestApartmentNewQueryInt(t *testing.T) {
	for _, c := range []struct {
		query string
		n     int64
		ok    bool
	}{
		{"", 0, true}, {"?x=1", 0, true}, {"?k=41", 41, true}, {"?k=%20%2B4_1%20", 41, true}, {"?k=-5", -5, true},
		{"?k=abc", 0, true}, {"?k=1.5", 0, true}, {"?k=1__0", 0, true}, {"?k=%2B", 0, true}, {"?k=1&k=2", 2, true},
		{"?k=%D9%A3", 0, false}, {"?k=99999999999999999999", 0, false},
	} {
		n, ok := apartmentNewQueryInt(httptest.NewRequest(http.MethodGet, "/apartments/new"+c.query, nil), "k")
		if n != c.n || ok != c.ok {
			t.Errorf("%q: got %d %v, want %d %v", c.query, n, ok, c.n, c.ok)
		}
	}
}

// TestApartmentNewTemplateKeys: every key the template or the field table
// reads is in the page catalog in all three languages.
func TestApartmentNewTemplateKeys(t *testing.T) {
	raw, err := os.ReadFile("apartmentnew/apartment_new.tmpl")
	if err != nil {
		t.Fatal(err)
	}
	keys := map[string]bool{}
	for _, m := range regexp.MustCompile(`[({]t "([a-z_.]+)"`).FindAllStringSubmatch(string(raw), -1) {
		keys[m[1]] = true
	}
	v := ApartmentNewForm{}.view("Add property")
	for _, a := range append(append([]apartmentNewAddr{}, v.AddrRow1...), v.AddrRow2...) {
		keys[a.LabelKey] = true
		if a.HintKey != "" {
			keys[a.HintKey] = true
		}
	}
	if len(keys) < 40 {
		t.Fatalf("only %d keys found", len(keys))
	}
	for key := range keys {
		for _, lang := range []string{"en", "cs", "vi"} {
			if _, ok := apartmentNewStrings[lang][key]; !ok {
				t.Errorf("catalog %s lacks %q", lang, key)
			}
		}
	}
}
