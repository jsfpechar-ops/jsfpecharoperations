package invoices

import (
	"encoding/json"
	"net/url"
	"os"
	"testing"
)

type siteInvoiceGolden struct {
	Cases []struct {
		Name   string            `json:"name"`
		Booking map[string]any   `json:"booking"`
		Form   map[string]string `json:"form"`
	} `json:"cases"`
}

func TestBookingIssueFormMatchesGolden(t *testing.T) {
	raw, err := os.ReadFile("../../booking/testdata/stay_invoice_golden.json")
	if err != nil {
		t.Fatal(err)
	}
	var g siteInvoiceGolden
	if err := json.Unmarshal(raw, &g); err != nil {
		t.Fatal(err)
	}
	for _, c := range g.Cases {
		b := c.Booking
		total := int(b["total_czk"].(float64))
		clean := int(b["cleaning_fee_czk"].(float64))
		name, _ := b["guest_name"].(string)
		email, _ := b["guest_email"].(string)
		lang, _ := b["lang"].(string)
		got := BookingIssueForm(total, clean, name, email, lang)
		want := url.Values{}
		for k, v := range c.Form {
			want.Set(k, v)
		}
		if got.Encode() != want.Encode() {
			t.Fatalf("%s: form mismatch\ngot %v\nwant %v", c.Name, got, want)
		}
	}
}
