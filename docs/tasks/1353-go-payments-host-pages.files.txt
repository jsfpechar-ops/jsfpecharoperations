// stripeMode is config.STRIPE_MODE.
func stripeMode(getenv func(string) string) string {
	return strings.ToLower(strings.TrimSpace(getenv("UBYHOST_STRIPE_MODE")))
}

// stripeOff is payments._off.
func stripeOff(getenv func(string) string) bool {
	m := stripeMode(getenv)
	return m != "test" && m != "live"
}

// pySpace is str.isspace for one rune: unicode.IsSpace plus \x1c-\x1f.
func pySpace(r rune) bool { return unicode.IsSpace(r) || (r >= 0x1c && r <= 0x1f) }

// pyStrip is str.strip().
func pyStrip(s string) string { return strings.TrimFunc(s, pySpace) }

// asciiDigits is str.isdigit for ASCII input (Python also takes other
// Unicode digits; the existing Go payment routes make the same choice).
func asciiDigits(s string) bool {
	if s == "" {
		return false
	}
	for _, c := range s {
		if c < '0' || c > '9' {
			return false
		}
	}
	return true
}

// lastQuery is request.query_params.get: the last value, and whether the key is there.
func lastQuery(r *http.Request, key string) (string, bool) {
	v, ok := r.URL.Query()[key]
	if !ok || len(v) == 0 {
		return "", ok
	}
	return v[len(v)-1], true
}

// lastForm is Starlette's form.get: the last value of a field.
func lastForm(r *http.Request, key string) string {
	if err := r.ParseForm(); err != nil {
		return ""
	}
	v := r.PostForm[key]
	if len(v) == 0 {
		return ""
	}
	return v[len(v)-1]
}

var isoDate = regexp.MustCompile(`^([0-9]{4})-([0-9]{1,2})-([0-9]{1,2})$`)

// dateCZ is validation.fmt_date (the date_cz filter): DD.MM.YYYY from the
// first ten characters, or the value unchanged.
func dateCZ(v string) string {
	head := v
	if r := []rune(v); len(r) > 10 {
		head = string(r[:10])
	}
	m := isoDate.FindStringSubmatch(head)
	if m == nil {
		return v
	}
	y, _ := strconv.Atoi(m[1])
	mo, _ := strconv.Atoi(m[2])
	d, _ := strconv.Atoi(m[3])
	t := time.Date(y, time.Month(mo), d, 0, 0, 0, 0, time.UTC)
	if y < 1 || t.Year() != y || int(t.Month()) != mo || t.Day() != d {
		return v
	}
	return fmt.Sprintf("%02d.%02d.%04d", d, mo, y)
}

// pyQuote is urllib.parse.quote(s) with the default safe="/".
func pyQuote(s string) string {
	const hexDigits = "0123456789ABCDEF"
	var b strings.Builder
	for i := 0; i < len(s); i++ {
		c := s[i]
		switch {
		case c >= 'a' && c <= 'z', c >= 'A' && c <= 'Z', c >= '0' && c <= '9', strings.IndexByte("_.-~/", c) >= 0:
			b.WriteByte(c)
		default:
			b.WriteByte('%')
			b.WriteByte(hexDigits[c>>4])
			b.WriteByte(hexDigits[c&15])
		}
	}
	return b.String()
}

// stampAt is db.utcnow() at the given time.
func stampAt(t time.Time) string {
	return t.UTC().Truncate(time.Second).Format("2006-01-02T15:04:05") + "+00:00"
}

// bufferedWriter holds the page until Render succeeds, so a failed render is
// a clean 500 and never half a page.
type bufferedWriter struct {
	header http.Header
	code   int
	body   bytes.Buffer
}

func (b *bufferedWriter) Header() http.Header { return b.header }
func (b *bufferedWriter) WriteHeader(code int) {
	if b.code == 0 {
		b.code = code
	}
}
func (b *bufferedWriter) Write(p []byte) (int, error) {
	if b.code == 0 {
		b.code = http.StatusOK
	}
	return b.body.Write(p)
}

type service struct {
	node   *node
	ID     string // the id when it is a string; get() matches only those
	Name   string
	Amount string
}

func servicesKey(owner int64) string { return fmt.Sprintf("payment_services:%d", owner) }

// listServices is payment_services.list_for: the dict items with a truthy id
// and name. Bad JSON, or JSON that is not a list, is no services.
func listServices(ctx context.Context, pool *pgxpool.Pool, owner int64) ([]service, error) {
	var raw *string
	err := pool.QueryRow(ctx, settingSQL, servicesKey(owner)).Scan(&raw)
	if err != nil && !errors.Is(err, pgx.ErrNoRows) {
		return nil, err
	}
	text := "[]"
	if raw != nil && *raw != "" {
		text = *raw
	}
	root, err := parseJSON(text)
	if err != nil || root.kind != 'a' {
		return nil, nil
	}
	var out []service
	for _, item := range root.items {
		if item.kind != 'o' || !item.get("id").truthy() || !item.get("name").truthy() {
			continue
		}
		s := service{node: item, Name: item.get("name").text(), Amount: item.get("amount_czk").text()}
		if id := item.get("id"); id.kind == 's' {
			s.ID = id.str
		} else {
			s.ID = id.text()
		}
		out = append(out, s)
	}
	return out, nil
}

// findService is payment_services.get.
func findService(list []service, id string) *service {
	for i := range list {
		if n := list[i].node.get("id"); n.kind == 's' && n.str == id {
			return &list[i]
		}
	}
	return nil
}

func saveServices(ctx context.Context, pool *pgxpool.Pool, owner int64, list []*node) error {
	var b bytes.Buffer
	(&node{kind: 'a', items: list}).dump(&b)
	_, err := pool.Exec(ctx, setSettingSQL, servicesKey(owner), b.String())
	return err
}

func nodes(list []service) []*node {
	out := make([]*node, 0, len(list)+1)
	for _, s := range list {
		out = append(out, s.node)
	}
	return out
}

var errService = errors.New("service refused")

// addService is payment_services.add.
func addService(ctx context.Context, pool *pgxpool.Pool, owner int64, name string, amount int64, amountOK bool) error {
	name = strings.Join(strings.FieldsFunc(name, pySpace), " ")
	if r := []rune(name); len(r) > maxName {
		name = string(r[:maxName])
	}
	if name == "" || !amountOK || amount < 1 || amount > maxAmount {
		return errService
	}
	list, err := listServices(ctx, pool, owner)
	if err != nil {
		return err
	}
	if len(list) >= maxServices {
		return errService
	}
	var raw [4]byte
	if _, err := rand.Read(raw[:]); err != nil {
		return err
	}
	item := &node{kind: 'o', keys: []string{"id", "name", "amount_czk"}, vals: []*node{
		{kind: 's', str: hex.EncodeToString(raw[:])},
		{kind: 's', str: name},
		{kind: 'n', num: strconv.FormatInt(amount, 10)},
	}}
	return saveServices(ctx, pool, owner, append(nodes(list), item))
}

// removeService is payment_services.remove.
func removeService(ctx context.Context, pool *pgxpool.Pool, owner int64, id string) error {
	list, err := listServices(ctx, pool, owner)
	if err != nil {
		return err
	}
	var kept []*node
	for _, s := range list {
		if n := s.node.get("id"); n.kind == 's' && n.str == id {
			continue
		}
		kept = append(kept, s.node)
	}
	if len(kept) == len(list) {
		return nil
	}
	if kept == nil {
		kept = []*node{}
	}
	return saveServices(ctx, pool, owner, kept)
}

func consentAccepted(ctx context.Context, pool *pgxpool.Pool, owner int64) (bool, error) {
	var id int64
	err := pool.QueryRow(ctx, consentSQL, owner, consentAction, "version="+consentVersion).Scan(&id)
	if errors.Is(err, pgx.ErrNoRows) {
		return false, nil

func hasBankAccount(ctx context.Context, pool *pgxpool.Pool, owner int64) (bool, error) {
	var iban, bank *string
	err := pool.QueryRow(ctx, bankSQL, owner).Scan(&iban, &bank)
	if errors.Is(err, pgx.ErrNoRows) {
		return false, nil
	}
	if err != nil {
		return false, err
	}
	return (iban != nil && *iban != "") || (bank != nil && *bank != ""), nil
}

type stay struct {
	ID    int64
	Label string
}

// loadStays is payments._stays with _stay_label.
func loadStays(ctx context.Context, pool *pgxpool.Pool, owner int64) ([]stay, error) {
	rows, err := pool.Query(ctx, staysSQL, owner)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []stay
	for rows.Next() {
		var (
			s                        stay
			summary, from, to, apart string
		)
		if err := rows.Scan(&s.ID, &summary, &from, &to, &apart); err != nil {
			return nil, err
		}
		name := pyStrip(summary)
		if name == "" {
			name = pyStrip(apart)
		}
		s.Label = fmt.Sprintf("%s (%s → %s)", name, from, to)
		out = append(out, s)
	}
	return out, rows.Err()
}

type step struct {
	Key, Status, Href string
	N                 int
	Cur, Help         bool
}

type entityView struct{ Name, Email string }

type requestView struct {
	ID                      int64
	Reason, Note, Status    string
	Amount                  int64
	NoStay                  bool
	Created, Tone           string
	CanLink, CanSend, HasQR bool
	ShowMenu, MenuSend      bool
}

type overviewData struct {
	StripeOff, Complete, Blocked bool
	State, Next                  string
	Steps                        []step
	Entity                       *entityView
	Services                     []service
	Requests                     []requestView
	HowSteps                     []int
}

func loadOverview(ctx context.Context, pool *pgxpool.Pool, getenv func(string) string, owner int64) (overviewData, error) {
	d := overviewData{StripeOff: stripeOff(getenv), State: "none", HowSteps: []int{1, 2, 3}}
	err := pool.QueryRow(ctx, accountSQL, owner, stripeMode(getenv)).Scan(&d.State)
	if err != nil && !errors.Is(err, pgx.ErrNoRows) {
		return d, err
	}
	d.Complete = d.State == "complete"
	d.Blocked = d.State == "rejected" || d.State == "disabled"

	// _setup
	consent, err := consentAccepted(ctx, pool, owner)
	if err != nil {
		return d, err
	}
	var (
		entityID    int64
		name, email string
		hasEntity   bool
		detailsHref = "/entities"
	)
	err = pool.QueryRow(ctx, entitySQL, owner).Scan(&entityID, &name, &email)
	switch {
	case err == nil:
		hasEntity = true
		detailsHref = fmt.Sprintf("/entities?edit=%d", entityID)
	case !errors.Is(err, pgx.ErrNoRows):
		return d, err
	}
	hasDetails := hasEntity && pyStrip(name) != ""
	stripeStatus := map[string]string{"pending": "waiting", "action_required": "attention"}[d.State]
	if stripeStatus == "" {
		stripeStatus = "todo"
	}
	status := func(ok bool) string {
		if ok {
			return "done"
		}
		return "todo"
	}
	d.Steps = []step{
		{Key: "consent", Status: status(consent)},
		{Key: "details", Status: status(hasDetails), Href: detailsHref},
		{Key: "stripe", Status: stripeStatus, Help: true},
		{Key: "live", Status: "todo", Help: true},
	}
	cur := ""
	for i := range d.Steps {
		d.Steps[i].N = i + 1
		if cur == "" && d.Steps[i].Status != "done" {
			cur = d.Steps[i].Key
			d.Steps[i].Cur = true
		}
	}
	switch {
	case !consent:
		d.Next = "consent"
	case !hasDetails:
		d.Next = "details"
	default:
		d.Next = "stripe"
	}
	if hasDetails {
		d.Entity = &entityView{Name: name, Email: email}
	}

	if d.Services, err = listServices(ctx, pool, owner); err != nil {
		return d, err
	}
	rows, err := pool.Query(ctx, requestsSQL, owner)
	if err != nil {
		return d, err
	}
	defer rows.Close()
	for rows.Next() {
		var (
			v           requestView
			rail, guest string
			created     string
		)
		if err := rows.Scan(&v.ID, &v.Reason, &v.Amount, &rail, &v.Status, &guest, &v.Note, &v.NoStay, &created); err != nil {
			return d, err
		}
		v.Created = dateCZ(created)
		switch v.Status {
		case "paid":
			v.Tone = "p-ok"
		case "sent", "awaiting_payment":
			v.Tone = "p-warn"
		default:
			v.Tone = "p-n"
		}
		open := v.Status == "draft" || v.Status == "sent" || v.Status == "awaiting_payment"
		v.CanLink = rail == "card" && open
		v.CanSend = v.Status == "draft" && guest != ""
		v.HasQR = v.Status == "draft" && rail == "qr"
		v.MenuSend = v.CanSend && v.CanLink
		v.ShowMenu = v.MenuSend || v.HasQR
		if !(v.Reason == "custom" && v.Note != "") {
			v.Note = ""
		}
		d.Requests = append(d.Requests, v)
	}
	return d, rows.Err()
}

// OverviewHandler serves GET /payments (payments_page).
func OverviewHandler(pool *pgxpool.Pool, getenv func(string) string, now func() time.Time) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		at := now()
		h, err := hostweb.Login(w, r, pool, getenv, at)
		if err != nil {
			fail(w, err)
			return
		}
		if h == nil {
			return
		}
		q := r.URL.Query()
		_, hasService := q["service"]
		_, hasStay := q["reservation_id"]
		if hasService || hasStay {
			target := "/payments/new"
			if r.URL.RawQuery != "" {
				target += "?" + r.URL.RawQuery
			}
			w.Header().Set("Location", target)
			w.WriteHeader(http.StatusSeeOther)
			return
		}
		d, err := loadOverview(r.Context(), pool, getenv, h.Workspace.ID)
		if err != nil {
			fail(w, err)
			return
		}
		render(w, r, pool, getenv, at, h, layout.Page{Title: "payments.title", BodyKind: "list", V2: true}, "payments", d)
	}
}

// REASONS in payment_request.py.
var reasons = []string{"repair", "extra_guest", "late_checkout", "cleaning", "damage", "custom"}

// NewHandler serves GET /payments/new (payments_new_page).
func NewHandler(pool *pgxpool.Pool, getenv func(string) string, now func() time.Time) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		at := now()
		h, err := hostweb.Login(w, r, pool, getenv, at)
		if err != nil {
			fail(w, err)
			return
		}
		if h == nil {
			return
		}
		ctx := r.Context()
		owner := h.Workspace.ID
		d := newData{Reasons: reasons}
		if raw, _ := lastQuery(r, "reservation_id"); asciiDigits(raw) {
			d.Selected, _ = strconv.ParseInt(raw, 10, 64)
		}
		list, err := listServices(ctx, pool, owner)
		if err != nil {
			fail(w, err)
			return
		}
		sid, _ := lastQuery(r, "service")
		d.Prefill = findService(list, sid)
		if d.Selected != 0 {
			var email string
			err := pool.QueryRow(ctx, stayEmailSQL, d.Selected, owner).Scan(&email)
			if err != nil && !errors.Is(err, pgx.ErrNoRows) {
				fail(w, err)
				return
			}
			d.GuestEmail = pyStrip(email)
		}
		if d.Stays, err = loadStays(ctx, pool, owner); err != nil {
			fail(w, err)
			return
		}
		bank, err := hasBankAccount(ctx, pool, owner)
		if err != nil {
			fail(w, err)
			return
		}
		d.NoBank = !bank
		render(w, r, pool, getenv, at, h, layout.Page{Title: "payments.new.title", BodyKind: "form", V2: true}, "payments_new", d)
	}
}

// ServiceAddHandler serves POST /payments/services (payment_service_add).
func ServiceAddHandler(pool *pgxpool.Pool, getenv func(string) string, now func() time.Time) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		at := now()
		if !hostweb.CSRF(w, r, getenv, at) {
			return
		}
		h, err := hostweb.Login(w, r, pool, getenv, at)
		if err != nil {
			fail(w, err)
			return
		}
		if h == nil {
			return
		}
		raw := pyStrip(lastForm(r, "amount_czk"))
		if !asciiDigits(raw) {
			backErr(w, r, "/payments", "payments.err.amount", "")
			return
		}
		amount, perr := strconv.ParseInt(raw, 10, 64)
		err = addService(r.Context(), pool, h.Workspace.ID, lastForm(r, "name"), amount, perr == nil)
		if errors.Is(err, errService) {
			backErr(w, r, "/payments", "payments.services.err", "")
			return
		}
		if err != nil {
			fail(w, err)
			return
		}
		backMsg(w, r, "/payments", "payments.services.saved")
	}
}

package pages

import (
	"bytes"
	"encoding/json"
	"errors"
	"io"
	"math"
	"strconv"
	"strings"
)

// node is one decoded JSON value with object keys kept in order, so a
// rewrite of the payment_services setting keeps what Python's json.loads and
// json.dumps keep: every item's keys, in their order.
type node struct {
	kind  byte // 's' string, 'n' number, 'b' bool, 'z' null, 'a' array, 'o' object
	str   string
	num   string // the number literal as written
	b     bool
	items []*node
	keys  []string
	vals  []*node
}

// parseJSON is json.loads for a whole document. A duplicate key keeps its
// first position and its last value, as a Python dict does.
func parseJSON(raw string) (*node, error) {
	dec := json.NewDecoder(strings.NewReader(raw))
	dec.UseNumber()
	n, err := parseNode(dec)
	if err != nil {
		return nil, err
	}
	if _, err := dec.Token(); err != io.EOF {
		return nil, errors.New("trailing data")
	}
	return n, nil
}

func parseNode(dec *json.Decoder) (*node, error) {
	tok, err := dec.Token()
	if err != nil {
		return nil, err
	}
	switch v := tok.(type) {
	case json.Delim:
		switch v {
		case '[':
			n := &node{kind: 'a'}
			for dec.More() {
				item, err := parseNode(dec)
				if err != nil {
					return nil, err
				}
				n.items = append(n.items, item)
			}
			if _, err := dec.Token(); err != nil {
				return nil, err
			}
			return n, nil
		case '{':
			n := &node{kind: 'o'}
			for dec.More() {
				kt, err := dec.Token()
				if err != nil {
					return nil, err
				}
				key, ok := kt.(string)
				if !ok {
					return nil, errors.New("object key")
				}
				val, err := parseNode(dec)
				if err != nil {
					return nil, err
				}
				if i := n.index(key); i >= 0 {
					n.vals[i] = val
				} else {
					n.keys = append(n.keys, key)
					n.vals = append(n.vals, val)
				}
			}
			if _, err := dec.Token(); err != nil {
				return nil, err
			}
			return n, nil
		}
		return nil, errors.New("unexpected delimiter")
	case string:
		return &node{kind: 's', str: v}, nil
	case json.Number:
		return &node{kind: 'n', num: string(v)}, nil
	case bool:
		return &node{kind: 'b', b: v}, nil
	case nil:
		return &node{kind: 'z'}, nil
	}
	return nil, errors.New("unexpected token")
}

func (n *node) index(key string) int {
	for i, k := range n.keys {
		if k == key {
			return i
		}
	}
	return -1
}

// get is dict.get: the value under key, or nil.
func (n *node) get(key string) *node {
	if n == nil || n.kind != 'o' {
		return nil
	}
	if i := n.index(key); i >= 0 {
		return n.vals[i]
	}
	return nil
}

// isFloat reports whether the literal is a Python float (has . e or E).
func (n *node) isFloat() bool { return strings.ContainsAny(n.num, ".eE") }

// truthy is Python's bool() of the loaded value.
func (n *node) truthy() bool {
	if n == nil {
		return false
	}
	switch n.kind {
	case 's':
		return n.str != ""
	case 'n':
		if n.isFloat() {
			f, _ := strconv.ParseFloat(n.num, 64)
			return f != 0
		}
		return strings.Trim(n.num, "-0") != ""
	case 'b':
		return n.b
	case 'a':
		return len(n.items) > 0
	case 'o':
		return len(n.keys) > 0
	}
	return false
}

// text is what Jinja prints for the value: str() for scalars. A list or an
// object prints as its JSON (Python prints its repr; only hand-edited
// settings can hold one).
func (n *node) text() string {
	if n == nil {
		return "" // a missing key is Jinja's undefined
	}
	switch n.kind {
	case 's':
		return n.str
	case 'n':
		return pyNumber(n)
	case 'b':
		if n.b {
			return "True"
		}
		return "False"
	case 'z':
		return "None"
	}
	var b bytes.Buffer
	n.dump(&b)
	return b.String()
}

// dump writes json.dumps(value, ensure_ascii=False): ", " and ": "
// separators, non-ASCII kept, no HTML escaping.
func (n *node) dump(b *bytes.Buffer) {
	switch n.kind {
	case 's':
		dumpString(b, n.str)
	case 'n':
		b.WriteString(pyNumber(n))
	case 'b':
		if n.b {
			b.WriteString("true")
		} else {
			b.WriteString("false")
		}
	case 'z':
		b.WriteString("null")
	case 'a':
		b.WriteByte('[')
		for i, item := range n.items {
			if i > 0 {
				b.WriteString(", ")
			}
			item.dump(b)
		}
		b.WriteByte(']')
	case 'o':
		b.WriteByte('{')
		for i, k := range n.keys {
			if i > 0 {
				b.WriteString(", ")
			}
			dumpString(b, k)
			b.WriteString(": ")
			n.vals[i].dump(b)
		}
		b.WriteByte('}')
	}
}

// pyNumber is repr() of the loaded number: an int as its digits, a float
// in Python's shortest repr ("100.0", "1e+16", "1.5e-05").
func pyNumber(n *node) string {
	if !n.isFloat() {
		if i, err := strconv.ParseInt(n.num, 10, 64); err == nil {
			return strconv.FormatInt(i, 10)
		}
		neg := strings.HasPrefix(n.num, "-")
		digits := strings.TrimLeft(strings.TrimPrefix(n.num, "-"), "0")
		if digits == "" {
			return "0"
		}
		if neg {
			return "-" + digits
		}
		return digits
	}
	f, err := strconv.ParseFloat(n.num, 64)
	if err != nil || math.IsInf(f, 0) {
		if f < 0 {
			return "-Infinity"
		}
		return "Infinity"
	}
	return pyFloat(f)
}

func pyFloat(f float64) string {
	if f == 0 {
		if math.Signbit(f) {
			return "-0.0"
		}
		return "0.0"
	}
	s := strconv.FormatFloat(f, 'e', -1, 64) // d.ddde±XX
	mant, expText, _ := strings.Cut(s, "e")
	exp, _ := strconv.Atoi(expText)
	neg := strings.HasPrefix(mant, "-")
	digits := strings.Replace(strings.TrimPrefix(mant, "-"), ".", "", 1)
	sign := ""
	if neg {
		sign = "-"
	}
	if exp < -4 || exp >= 16 {
		m := digits[:1]
		if len(digits) > 1 {
			m += "." + digits[1:]
		}
		es := "+"
		if exp < 0 {
			es = "-"
			exp = -exp
		}
		e := strconv.Itoa(exp)
		if len(e) < 2 {
			e = "0" + e
		}
		return sign + m + "e" + es + e
	}
	if exp < 0 {
		return sign + "0." + strings.Repeat("0", -exp-1) + digits
	}
	if len(digits) <= exp+1 {
		return sign + digits + strings.Repeat("0", exp+1-len(digits)) + ".0"
	}
	return sign + digits[:exp+1] + "." + digits[exp+1:]
}

// dumpString is json.dumps of a str with ensure_ascii=False.
func dumpString(b *bytes.Buffer, s string) {
	const hexDigits = "0123456789abcdef"
	b.WriteByte('"')
	for _, r := range s {
		switch r {
		case '"':
			b.WriteString(`\"`)
		case '\\':
			b.WriteString(`\\`)
		case '\n':
			b.WriteString(`\n`)
		case '\r':
			b.WriteString(`\r`)
		case '\t':
			b.WriteString(`\t`)
		case '\b':
			b.WriteString(`\b`)
		case '\f':
			b.WriteString(`\f`)
		default:
			if r < 0x20 {
				b.WriteString(`\u00`)
				b.WriteByte(hexDigits[r>>4])
				b.WriteByte(hexDigits[r&15])
			} else {
				b.WriteRune(r)
			}
		}
	}
	b.WriteByte('"')
}

var (

var (

	fieldFlags = []string{"checked", "selected", "disabled", "required", "readonly", "open", "hidden"}
	fieldTags  = map[string]bool{"form": true, "input": true, "select": true, "option": true, "textarea": true, "button": true, "details": true, "label": true}

func bodyRegion(page string) string {
	start := strings.Index(page, `<div class="ui-page"`)
	if start < 0 {
		return ""
	}
	end := strings.Index(page[start:], `<div class="page-skeleton"`)
	if end < 0 {
		return ""
	}
	return page[start : start+end]
}

func summarise(page string) summaryData {
	region := bodyRegion(page)
	for _, re := range dropRes {
		region = re.ReplaceAllString(region, " ")
	}
	region = commentRe.ReplaceAllString(region, " ")
	out := summaryData{Fields: []string{}, Links: []string{}}
	for _, m := range tagRe.FindAllStringSubmatch(region, -1) {
		if m[1] != "" {
			continue
		}
		tag := strings.ToLower(m[2])
		attrs := map[string]*string{}
		for _, a := range attrRe.FindAllStringSubmatchIndex(m[3], -1) {
			name := strings.ToLower(m[3][a[2]:a[3]])
			var val *string
			for g := 4; g <= 8; g += 2 {
				if a[g] >= 0 {
					v := html.UnescapeString(m[3][a[g]:a[g+1]])
					val = &v
					break
				}
			}
			attrs[name] = val
		}
		if v, ok := attrs["name"]; ok && v != nil && *v == "_csrf" {
			c := "CSRF"
			attrs["value"] = &c
		}
		if v, ok := attrs["href"]; ok && tag == "a" {
			if v == nil {
				out.Links = append(out.Links, "")
			} else {
				out.Links = append(out.Links, *v)
			}
		}
		if fieldTags[tag] {
			parts := []string{tag}
			for _, k := range fieldAttrs {
				if v := attrs[k]; v != nil {
					parts = append(parts, k+"="+*v)
				}
			}
			for _, k := range fieldFlags {
				if _, ok := attrs[k]; ok {
					parts = append(parts, k)
				}
			}
			out.Fields = append(out.Fields, strings.Join(parts, " "))
		}
	}
	text := html.UnescapeString(tagRe.ReplaceAllString(region, " "))
	out.Text = strings.Join(strings.FieldsFunc(text, pySpace), " ")
	return out
}

func readRows(t *testing.T, pool *pgxpool.Pool) rowsData {
	t.Helper()
	ctx := context.Background()
	out := rowsData{Settings: map[string]string{}, Audit: []map[string]any{}}
	rows, err := pool.Query(ctx, `SELECT key, COALESCE(value, '') FROM settings WHERE key LIKE 'payment_services:%' ORDER BY key`)
	if err != nil {
		t.Fatal(err)
	}
	for rows.Next() {
		var k, v string
		if err := rows.Scan(&k, &v); err != nil {
			t.Fatal(err)
		}
		out.Settings[k] = newID.ReplaceAllString(v, `"id": "<new>"`)
	}
	rows.Close()
	rows, err = pool.Query(ctx, `SELECT actor, action, detail, owner_user_id, actor_user_id, impersonator_user_id FROM audit
		WHERE action = 'payments_terms_accepted' ORDER BY owner_user_id, id`)
	if err != nil {
		t.Fatal(err)
	}
	defer rows.Close()
	for rows.Next() {
		var (
			actor, action, detail string
			owner                 *int64
			actorID, imp          *int64
		)
		if err := rows.Scan(&actor, &action, &detail, &owner, &actorID, &imp); err != nil {
			t.Fatal(err)
		}
		num := func(p *int64) any {
			if p == nil {
				return nil
			}
			return float64(*p)
		}
		out.Audit = append(out.Audit, map[string]any{"actor": actor, "action": action, "detail": detail,
			"owner_user_id": num(owner), "actor_user_id": num(actorID), "impersonator_user_id": num(imp)})
	}
	return out
}

// fakeLinks answers SendPaymentLink from golden.json's link_results.
type fakeLinks struct{ results map[string]string }

func (f *fakeLinks) SendPaymentLink(_ context.Context, id int64) (string, error) {
	v := f.results[fmt.Sprint(id)]
	if v == "STRIPE_OFF" {
		return "", payments.ErrStripeOff
	}
	return v, nil
}

const csrfCookie = "golden-csrf-nonce"

func csrfToken(t *testing.T, secret string, now time.Time) string {
	t.Helper()
	sum := sha256.Sum256([]byte(csrfCookie))
	tok, err := secure.NewTimedSerializer(secret, "ubyhost-csrf").Dumps(map[string]any{"nonce": hex.EncodeToString(sum[:])}, now)
	if err != nil {
		t.Fatal(err)
	}
	return tok
}

func newMux(pool *pgxpool.Pool, g golden, mode string, links payments.LinkSender) *http.ServeMux {
	env := func(k string) string {
		if k == "UBYHOST_STRIPE_MODE" {
			return mode
		}
		return g.Env[k]
	}
	mux := http.NewServeMux()
	mux.Handle("GET /payments", OverviewHandler(pool, env, time.Now))
	mux.Handle("GET /payments/new", NewHandler(pool, env, time.Now))
	mux.Handle("POST /payments/services", ServiceAddHandler(pool, env, time.Now))
	mux.Handle("POST /payments/services/{id}/delete", ServiceDeleteHandler(pool, env, time.Now))
	mux.Handle("GET /payments/requests/{id}/share", ShareHandler(pool, env, time.Now, links))
	mux.Handle("GET /payments/consent", ConsentHandler(pool, env, time.Now))
	mux.Handle("POST /payments/consent", ConsentAcceptHandler(pool, env, time.Now))
	return mux
}

func call(t *testing.T, pool *pgxpool.Pool, g golden, c goldenCase, links payments.LinkSender) *httptest.ResponseRecorder {
	t.Helper()
	now := time.Now()
	var req *http.Request
	if c.Method == http.MethodPost {
		form := url.Values{}
		for k, v := range c.Form {
			form.Set(k, v)
		}
		form.Set("_csrf", csrfToken(t, g.Secret, now))
		req = httptest.NewRequest(c.Method, c.Path, strings.NewReader(form.Encode()))
		req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
		req.AddCookie(&http.Cookie{Name: "ubyhost_csrf", Value: csrfCookie})
	} else {
		req = httptest.NewRequest(c.Method, c.Path, nil)
	}
	req.Host = "testserver"
	req.AddCookie(&http.Cookie{Name: "ubyhost_lang", Value: c.Lang})
	if c.UID != 0 {
		req.AddCookie(&http.Cookie{Name: session.CookieName, Value: session.Sign(g.Secret,
			map[string]any{"uid": c.UID, "sv": int64(0), "iat": now.Unix(), "rf": now.Unix()}, now)})
	}
	rec := httptest.NewRecorder()
	newMux(pool, g, c.Mode, links).ServeHTTP(rec, req)
	return rec
}

func check(t *testing.T, c goldenCase, rec *httptest.ResponseRecorder) {
	t.Helper()
	if rec.Code != c.Status {
		t.Fatalf("status %d, Python %d (Location %q)", rec.Code, c.Status, rec.Header().Get("Location"))
	}
	if got := rec.Header().Get("Location"); got != c.Location {
		t.Errorf("Location\n go %q\n py %q", got, c.Location)
	}
	body := rec.Body.String()
	if c.Page == nil {
		if c.Status != http.StatusSeeOther && body != c.Body {
			t.Errorf("body %q, Python %q", body, c.Body)
		}
		return
	}
	if m := titleRe.FindStringSubmatch(body); m == nil || html.UnescapeString(strings.TrimSpace(m[1])) != html.UnescapeString(c.Title) {
		t.Errorf("title %v, Python %q", m, c.Title)
	}
	got := summarise(body)
	if got.Text != c.Page.Text {
		t.Errorf("text\n go %s\n py %s", got.Text, c.Page.Text)
	}
	if !reflect.DeepEqual(got.Fields, c.Page.Fields) {
		t.Errorf("fields\n go %q\n py %q", got.Fields, c.Page.Fields)
	}
	if !reflect.DeepEqual(got.Links, c.Page.Links) {
		t.Errorf("links\n go %q\n py %q", got.Links, c.Page.Links)
	}
}

// TestGoldenMatchesPython runs every Python case on the Go handlers: the same
// status, Location, title, visible text, form fields and links, and after
// each write the same payment_services settings and consent audit rows.
func TestGoldenMatchesPython(t *testing.T) {
	pool := testPool(t)
	g := loadGolden(t)
	seed(t, pool, g.Fixture)
	links := &fakeLinks{results: g.Links}
	if len(g.Cases) == 0 || len(g.Writes) == 0 {
		t.Fatal("golden.json has no cases")
	}
	for _, c := range g.Cases {
		t.Run(c.Name, func(t *testing.T) { check(t, c, call(t, pool, g, c, links)) })
	}
	for _, c := range g.Writes {
		ok := t.Run(c.Name, func(t *testing.T) {
			check(t, c, call(t, pool, g, c, links))
			got := readRows(t, pool)
			want := *c.Rows
			if !reflect.DeepEqual(got.Settings, want.Settings) {
				t.Errorf("settings\n go %q\n py %q", got.Settings, want.Settings)
			}
			if !reflect.DeepEqual(got.Audit, want.Audit) {
				t.Errorf("audit\n go %v\n py %v", got.Audit, want.Audit)
			}
		})
		if !ok {
			break // later writes build on this one
		}
	}
}

// TestPythonJSON: json.loads then json.dumps(ensure_ascii=False), as
// payment_services rewrites the setting (expected text from Python 3).
func TestPythonJSON(t *testing.T) {
	raw := `[{"id": "a", "n": 1.0, "e": 1e20, "f": 0.00001, "g": 123456789012345678, "h": 2.5e-3, "s": "<&> \u0001\\\"é", "z": null, "b": true, "x": [1, {"k": -0}], "id": "b"}]`
	want := `[{"id": "b", "n": 1.0, "e": 1e+20, "f": 1e-05, "g": 123456789012345678, "h": 0.0025, "s": "<&>` + " " + `\u0001\\\"é", "z": null, "b": true, "x": [1, {"k": 0}]}]`
	n, err := parseJSON(raw)
	if err != nil {
		t.Fatal(err)
	}
	var b bytes.Buffer
	n.dump(&b)
	if b.String() != want {
		t.Errorf("dump\n got %s\nwant %s", b.String(), want)
	}
	for _, bad := range []string{"{not json", "[1] 2", ""} {
		if _, err := parseJSON(bad); err == nil {
			t.Errorf("%q parsed", bad)
		}
	}
}

var hrefRe = regexp.MustCompile(`(?i)href\s*=\s*"([^"]*)"`)

// TestShareNeverLinksJavascript: a stored link or guest e-mail that is a
// javascript: URL never becomes an href (brief 1353, until 1381's safeHref).
func TestShareNeverLinksJavascript(t *testing.T) {
	pool := testPool(t)
	g := loadGolden(t)
	seed(t, pool, g.Fixture)
	if _, err := pool.Exec(context.Background(), `UPDATE payment_request SET guest_email = 'javascript:alert(1)' WHERE id = 72`); err != nil {
		t.Fatal(err)
	}
	links := &fakeLinks{results: map[string]string{"72": "javascript:alert(1)//"}}
	c := goldenCase{Method: http.MethodGet, Path: "/payments/requests/72/share", UID: 2, Lang: "en", Mode: "test"}
	rec := call(t, pool, g, c, links)
	if rec.Code != http.StatusOK {
		t.Fatalf("status %d", rec.Code)
	}
	hrefs := hrefRe.FindAllStringSubmatch(rec.Body.String(), -1)
	if len(hrefs) == 0 {
		t.Fatal("no links")
	}
	for _, m := range hrefs {
		if strings.HasPrefix(strings.ToLower(strings.TrimSpace(html.UnescapeString(m[1]))), "javascript:") {
			t.Errorf("javascript href %q", m[1])
		}
	}
	if !strings.Contains(rec.Body.String(), `href="mailto:javascript%3Aalert%281%29?subject=`) {
		t.Error("mailto link missing or not encoded")
	}
}

// TestRefusals: Go-only edges the golden cannot record.
func TestRefusals(t *testing.T) {
	pool := testPool(t)
	g := loadGolden(t)
	seed(t, pool, g.Fixture)
	links := &fakeLinks{results: g.Links}
	before := readRows(t, pool)

	// FastAPI answers 422 for a non-integer id; Go answers 404.
	rec := call(t, pool, g, goldenCase{Method: http.MethodGet, Path: "/payments/requests/abc/share", UID: 2, Lang: "en", Mode: "test"}, links)
	if rec.Code != http.StatusNotFound {
		t.Errorf("non-integer id: status %d", rec.Code)
	}

	// A POST without a valid token changes nothing and never reaches /payments.
	for _, path := range []string{"/payments/services", "/payments/services/svc-a/delete", "/payments/consent"} {
		req := httptest.NewRequest(http.MethodPost, path, strings.NewReader("name=X&amount_czk=5&agree=1&_csrf=bad"))
		req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
		req.Host = "testserver"
		req.AddCookie(&http.Cookie{Name: "ubyhost_csrf", Value: csrfCookie})
		req.AddCookie(&http.Cookie{Name: session.CookieName, Value: session.Sign(g.Secret,
			map[string]any{"uid": int64(2), "sv": int64(0), "iat": time.Now().Unix()}, time.Now())})
		rec := httptest.NewRecorder()
		newMux(pool, g, "test", links).ServeHTTP(rec, req)
		if strings.HasPrefix(rec.Header().Get("Location"), "/payments") {
			t.Errorf("%s: bad csrf redirected to %q", path, rec.Header().Get("Location"))
		}
	}
	if after := readRows(t, pool); !reflect.DeepEqual(before, after) {
		t.Error("a refused POST changed rows")
	}
}

def body_region(page: str) -> str:
    start = page.find('<div class="ui-page"')
    end = page.find('<div class="page-skeleton"', start)
    if start < 0 or end < 0:

def summary(page: str) -> dict:

    for drop in _DROP:
        region = drop.sub(" ", region)

    fields, links = [], []
    for m in _TAG.finditer(region):
        if m.group(1):
            continue
        tag = m.group(2).lower()
        attrs = {}
        for a in _ATTR.finditer(m.group(3)):
            value = a.group(2) if a.group(2) is not None else a.group(3) if a.group(3) is not None else a.group(4)
            attrs[a.group(1).lower()] = html.unescape(value) if value is not None else None
        if attrs.get("name") == "_csrf":
            attrs["value"] = "CSRF"
        if tag == "a" and "href" in attrs:
            links.append(attrs["href"] or "")
        if tag in ("form", "input", "select", "option", "textarea", "button", "details", "label"):
            parts = [tag]
            parts += [f"{k}={attrs[k]}" for k in _FIELD_ATTRS if attrs.get(k) is not None]
            parts += [k for k in _FLAGS if k in attrs]
            fields.append(" ".join(parts))
    text = html.unescape(_TAG.sub(" ", region))
    return {"text": " ".join(text.split()), "fields": fields, "links": links}

def main() -> int:
    for key in ("UBYHOST_DATABASE_URL", "UBYHOST_TEST_DATABASE_URL"):

    def call(case):
        name, method, path, uid, lang, mode, form = case
        config.STRIPE_MODE = mode
        client = TestClient(app, base_url="http://testserver")
        client.cookies.set("ubyhost_lang", lang)
        if uid:
            client.cookies.set(auth.SESSION_COOKIE, auth.issue_session(uid, 0))
        if method == "GET":
            response = client.get(path, follow_redirects=False)
        else:
            nonce = "golden-csrf-nonce"
            client.cookies.set(security.CSRF_COOKIE, nonce)
            token = security._csrf_serializer().dumps({"nonce": security._nonce_binding(nonce)})
            response = client.request(method, path, data={**form, "_csrf": token}, follow_redirects=False)
        out = {"name": name, "method": method, "path": path, "uid": uid, "lang": lang, "mode": mode,
               "form": form, "status": response.status_code, "location": response.headers.get("location", "")}
        if response.status_code == 200:
            out["page"] = summary(response.text)
            out["title"] = re.search(r"<title>(.*?)</title>", response.text, re.S).group(1).strip()
        elif response.status_code != 303:
            out["body"] = response.text
        return out

    def rows():
        svc = {r["key"]: re.sub(r'"id": "[0-9a-f]{8}"', '"id": "<new>"', r["value"] or "")
               for r in db.query("SELECT key, value FROM settings WHERE key LIKE 'payment_services:%' ORDER BY key")}
        audit = [dict(r) for r in db.query(
            "SELECT actor, action, detail, owner_user_id, actor_user_id, impersonator_user_id FROM audit"

    data = {"secret": SECRET, "env": env, "fixture": [{"table": t, "row": r} for t, r in FIXTURE],

        if not OUT.exists() or OUT.read_text(encoding="utf-8") != text:
            print(f"stale: {OUT.relative_to(APP.parent)}")
            return 1
        return 0
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(text, encoding="utf-8")
    print(f"wrote {OUT.relative_to(APP.parent)}: {len(cases)} cases, {len(writes)} writes")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
