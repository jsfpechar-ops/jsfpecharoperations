// supportReason is auth.support_reason: whitespace folded, 5 to 300 characters.
func supportReason(value string) string {
	text := strings.Join(strings.FieldsFunc(value, pySpace), " ")
	if n := len([]rune(text)); n < 5 || n > 300 {
		return ""
	}
	return text

// impersonate is user_impersonate: open a host's workspace.
func (d deps) impersonate(w http.ResponseWriter, r *http.Request) {
	at := d.now()
	if !hostweb.CSRF(w, r, d.getenv, at) {
		return
	}
	userID, ok := pathID(w, r, "user_id")
	if !ok {
		return
	}
	h, ok := d.admin(w, r)
	if !ok {
		return
	}
	ctx := r.Context()
	var found int64
	err := d.pool.QueryRow(ctx, `SELECT id FROM user_account WHERE id = $1 AND active = 1`, userID).Scan(&found)
	if errors.Is(err, pgx.ErrNoRows) {
		adminsOnly(w, r)
		return
	}
	if err != nil {
		serverError(w, err)
		return
	}
	reason := supportReason(formStr(r, "reason", ""))
	if reason == "" {
		reason = "support"
	}
	iat := session.IssuedAt(h.Payload)
	if err := d.attach(w, h.Account.ID, h.Account.SessionVersion, session.IssueOptions{Workspace: userID, IssuedAt: &iat}, at); err != nil {
		serverError(w, err)
		return
	}

		h.Account.Username, ptr(userID)); err != nil {
		serverError(w, err)
		return
	}
	redirect(w, "/")
}

// ownerIDs is [int(v) for v in form.getlist("owner_ids") if str(v).isdigit()] as json.dumps writes it.
func ownerIDs(r *http.Request) string {
	form(r)
	parts := []string{}
	for _, v := range r.PostForm["owner_ids"] {
		if pyIsDigit(v) {
			if n, err := strconv.ParseInt(v, 10, 64); err == nil {
				parts = append(parts, itoa(n))
			}
		}
	}
	return "[" + strings.Join(parts, ", ") + "]"
}

// approx is int(raw) if raw.isdigit() else None.
func approx(raw string) any {
	if !pyIsDigit(raw) {
		return nil
	}
	n, err := strconv.ParseInt(raw, 10, 64)
	if err != nil {
		return nil
	}
	return n
}

// incidentUpdate is incident_update (incidents.mark_timestamp, incidents.update).
func (d deps) incidentUpdate(w http.ResponseWriter, r *http.Request) {
	at := d.now()
	if !hostweb.CSRF(w, r, d.getenv, at) {
		return
	}
	id, ok := pathID(w, r, "incident_id")
	if !ok {
		return
	}
	h, ok := d.admin(w, r)
	if !ok {
		return
	}
	ctx := r.Context()
	var found int64
	err := d.pool.QueryRow(ctx, `SELECT id FROM security_incident WHERE id = $1`, id).Scan(&found)
	if errors.Is(err, pgx.ErrNoRows) {
		hostweb.Back(w, r, incidentsPath, none, flashFor(r, "flash.error.no_such_incident"))
		return
	}
	if err != nil {
		serverError(w, err)
		return
	}
	now := session.UTCNow(at)
	action := formStr(r, "action", "")
	if contains(incidentTimestamps, action) {
		if _, err := d.pool.Exec(ctx, `UPDATE security_incident SET `+action+` = $1, updated_at = $2 WHERE id = $3`, now, now, id); err != nil {
			serverError(w, err)
			return
		}
	}
	var sets []string
	var vals []any
	for _, field := range []string{"summary", "data_categories", "risk_level", "notes"} {
		if formHas(r, field) {
			vals = append(vals, formStr(r, field, ""))
			sets = append(sets, field+" = $"+strconv.Itoa(len(vals)))
		}
	}
	if formHas(r, "approx_subjects") {
		vals = append(vals, approx(formStr(r, "approx_subjects", "")))
		sets = append(sets, "approx_subjects = $"+strconv.Itoa(len(vals)))
	}
	if len(sets) > 0 {
		vals = append(vals, now, id)
		sets = append(sets, "updated_at = $"+strconv.Itoa(len(vals)-1))
		if _, err := d.pool.Exec(ctx, `UPDATE security_incident SET `+strings.Join(sets, ", ")+` WHERE id = $`+strconv.Itoa(len(vals)), vals...); err != nil {
			serverError(w, err)
			return
		}
	}
	if err := d.audit(ctx, &gated{h}, at, "incident_updated", "incident="+itoa(id), h.Account.Username, nil); err != nil {
		serverError(w, err)
		return
	}
	hostweb.Back(w, r, incidentsPath, flashFor(r, "flash.incidents.saved"), none)
}

// adsConversionsCSV is ads_conversions_csv: 404 while sign-up is off, then
// the admin gate; the export marks the rows uploaded.
func (d deps) adsConversionsCSV(w http.ResponseWriter, r *http.Request) {
	at := d.now()
	if !hostweb.CSRF(w, r, d.getenv, at) {
		return
	}
	if !signup.Enabled(d.getenv) {
		httpError(w, http.StatusNotFound)
		return
	}
	h, ok := d.admin(w, r)
	if !ok {
		return
	}
	again := formStr(r, "again", "") == "1"
	ctx := signup.WithAuditActor(r.Context(), h.Account.ID, h.Account.Username, h.Impersonating)
	body, count, err := signup.ExportConversions(ctx, d.pool, d.getenv, at, again)
	if err != nil {
		serverError(w, err)
		return
	}
	yes := "no"
	if again {
		yes = "yes"
	}
	if err := d.audit(ctx, &gated{h}, at, "ads_conversions_exported", "rows="+strconv.Itoa(count)+"; again="+yes, h.Account.Username, nil); err != nil {
		serverError(w, err)
		return
	}
	w.Header().Set("Content-Type", "text/csv; charset=utf-8")
	w.Header().Set("Content-Disposition", `attachment; filename="ubyhost-ads-conversions.csv"`)
	w.Header().Set("Cache-Control", "no-store")
	w.WriteHeader(http.StatusOK)
	_, _ = io.WriteString(w, body)
}

			return
		}
	}
	hostweb.Back(w, r, back, none, none)
}

	for in, want := range map[string]string{"  needs\n help\x1cnow ": "needs help now", "abc": "", " a b c ": "a b c", strings.Repeat("x", 301): ""} {

var stamp = regexp.MustCompile(`^\d{4}-\d\d-\d\dT`)

func norm(v any) *string {
	if v == nil {
		return nil
	}
	s := fmt.Sprint(v)
	if stamp.MatchString(s) && !strings.HasPrefix(s, "2001-") && !strings.HasPrefix(s, "2099-") {
		s = "<now>"
	}
	return &s
}

func dump(t *testing.T, ctx context.Context, pool *pgxpool.Pool) map[string][][]*string {
	t.Helper()
	out := map[string][][]*string{}
	for name, sql := range watch {
		rows, err := pool.Query(ctx, sql)
		if err != nil {
			t.Fatal(err)
		}
		out[name] = [][]*string{}
		for rows.Next() {
			vals, err := rows.Values()
			if err != nil {
				t.Fatal(err)
			}
			row := make([]*string, len(vals))
			for i, v := range vals {
				row[i] = norm(v)
			}
			out[name] = append(out[name], row)
		}
		rows.Close()
	}
	return out
}

		table := item[0].(string)
		row := item[1].(map[string]any)
		cols := make([]string, 0, len(row))
		for c := range row {
			cols = append(cols, c)
		}
		sort.Strings(cols)
		marks := make([]string, len(cols))
		vals := make([]any, len(cols))
		for i, c := range cols {
			marks[i] = "$" + strconv.Itoa(i+1)
			vals[i] = seedValue(row[c], now)
		}
		if _, err := pool.Exec(ctx, "INSERT INTO "+table+" ("+strings.Join(cols, ", ")+") VALUES ("+strings.Join(marks, ", ")+")", vals...); err != nil {
			t.Fatalf("seed %s: %v", table, err)
		}
	}
	for _, table := range tables {
		if table == "mail_suppression" {
			continue
		}
		if _, err := pool.Exec(ctx, fmt.Sprintf(`SELECT setval(pg_get_serial_sequence('%s', 'id'), COALESCE((SELECT MAX(id) FROM %s), 0) + 1, false)`, table, table)); err != nil {
			t.Fatal(err)
		}
	}
}

// sessionToken is the script's session_cookie: auth.issue_session with
// times relative to now, through 1280's IssueSession.
func sessionToken(t *testing.T, secret string, spec map[string]int64, now time.Time) string {
	t.Helper()
	sv := int64(1)
	if v, ok := spec["sv"]; ok {
		sv = v
	}
	iat := now.Unix() - 60
	if v, ok := spec["iat"]; ok {
		iat = now.Unix() + v
	}
	opt := session.IssueOptions{Workspace: spec["as"], Remember: spec["rm"] != 0, IssuedAt: &iat}
	if v, ok := spec["ast"]; ok {
		ast := now.Unix() + v
		opt.ImpersonationStartedAt = &ast
	}
	token, err := session.IssueSession(secret, spec["uid"], sv, opt, now)
	if err != nil {
		t.Fatal(err)
	}
	return token
}

func parseCookie(t *testing.T, secret, line string, now time.Time) cookieRec {
	t.Helper()
	parts := strings.Split(line, ";")
	name, value, _ := strings.Cut(strings.TrimSpace(parts[0]), "=")
	rec := cookieRec{Name: name, Attrs: map[string]string{}, Empty: value == "" || value == `""`}
	for _, p := range parts[1:] {
		k, v, _ := strings.Cut(strings.TrimSpace(p), "=")
		k = strings.ToLower(k)
		switch k {
		case "expires":
			rec.Attrs[k] = "set"
		case "httponly", "secure":
			rec.Attrs[k] = "true"
		case "samesite":
			rec.Attrs[k] = strings.ToLower(v)
		default:
			rec.Attrs[k] = v
		}
	}
	if name == "ubyhost_session" && !rec.Empty {
		var payload map[string]any
		if err := secure.NewTimedSerializer(secret, "ubyhost-session").Loads(value, 400*24*time.Hour, time.Now().Add(time.Second), &payload); err != nil {
			t.Fatalf("session cookie does not verify: %v", err)
		}
		for _, k := range []string{"iat", "rf", "ast"} {
			if v, ok := payload[k].(float64); ok {
				payload[k] = float64(int64(v) - now.Unix())
			}
		}
		rec.Payload = payload
	}
	return rec
}

// sameOffsets compares session payloads, the time offsets within 2 seconds.
func samePayload(got, want map[string]any) bool {
	if len(got) != len(want) {
		return false
	}
	for k, w := range want {
		g, ok := got[k]
		if !ok {
			return false
		}
		if k == "iat" || k == "rf" || k == "ast" {
			d := g.(float64) - w.(float64)
			if d < -2 || d > 2 {
				return false
			}
			continue
		}
		if !reflect.DeepEqual(g, w) {
			return false
		}
	}
	return true
}

func str(p *string) string {
	if p == nil {
		return "<nil>"
	}
	return *p
}

func header(h http.Header, k string) *string {
	if v, ok := h[http.CanonicalHeaderKey(k)]; ok && len(v) > 0 {
		return &v[0]
	}
	return nil
}

func TestRoutesGoldenDatabase(t *testing.T) {
	g := loadGolden(t)
	pool := testPool(t)
	ctx := context.Background()
	sum := sha256.Sum256([]byte(g.Nonce))
	csrfToken, err := secure.NewTimedSerializer(g.Secret, "ubyhost-csrf").Dumps(map[string]any{"nonce": hex.EncodeToString(sum[:])}, time.Now())
	if err != nil {
		t.Fatal(err)
	}
	for _, c := range g.Cases {
		t.Run(c.Name, func(t *testing.T) {
			now := time.Now()
			loadSeed(t, ctx, pool, append(append([][2]any{}, g.Seed...), c.Seed...), now)
			env := map[string]string{}
			for k, v := range g.Env {
				env[k] = v
			}
			for k, v := range c.Env {
				env[k] = v
			}
			getenv := func(k string) string { return env[k] }
			mux := http.NewServeMux()
			for _, r := range Routes(pool, getenv, time.Now) {
				h := r.Handler
				// main.go's metrics.wrap sets hardenHeaders first; this is its Cache-Control.
				mux.Handle(r.Pattern, http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
					w.Header().Set("Cache-Control", "no-store, private")
					h(w, req)
				}))
			}
			csrf := c.CSRF == nil || *c.CSRF
			var pairs []string
			for _, p := range c.Form {
				pairs = append(pairs, url.QueryEscape(p[0])+"="+url.QueryEscape(p[1]))
			}
			if csrf && c.Method == http.MethodPost {
				pairs = append(pairs, "_csrf="+url.QueryEscape(csrfToken))
			}
			target := "http://testserver" + c.Path
			if c.Query != "" {
				target += "?" + c.Query
			}
			req := httptest.NewRequest(c.Method, target, strings.NewReader(strings.Join(pairs, "&")))
			if c.Method == http.MethodPost {
				req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
			}
			var cookies []string
			if c.Session != nil {
				cookies = append(cookies, "ubyhost_session="+sessionToken(t, g.Secret, c.Session, now))
			}
			if csrf {
				cookies = append(cookies, "ubyhost_csrf="+g.Nonce)
			}
			if c.Lang != "" {
				cookies = append(cookies, "ubyhost_lang="+c.Lang)
			}
			if len(cookies) > 0 {
				req.Header.Set("Cookie", strings.Join(cookies, "; "))
			}
			for k, v := range c.Headers {
				req.Header.Set(k, v)
			}
			rec := httptest.NewRecorder()
			mux.ServeHTTP(rec, req)
			res := rec.Result()
			w := c.Want
			if res.StatusCode != w.Status {
				t.Fatalf("status %d, want %d (body %q)", res.StatusCode, w.Status, rec.Body.String())
			}
			for _, h := range []struct {
				name string
				want *string
			}{{"Location", w.Location}, {"Content-Type", w.ContentType}, {"Cache-Control", w.CacheControl}, {"Content-Disposition", w.ContentDisposition}} {
				if got := header(res.Header, h.name); str(got) != str(h.want) {
					t.Errorf("%s %q, want %q", h.name, str(got), str(h.want))
				}
			}
			if w.Body != nil && rec.Body.String() != *w.Body {
				t.Errorf("body %q, want %q", rec.Body.String(), *w.Body)
			}
			var got []cookieRec
			for _, line := range res.Header["Set-Cookie"] {
				got = append(got, parseCookie(t, g.Secret, line, now))
			}
			if len(got) != len(w.Cookies) {
				t.Fatalf("cookies %+v, want %+v", got, w.Cookies)
			}
			for i := range got {
				wc := w.Cookies[i]
				if gap := knownGaps[c.Name]; gap != "" && wc.Payload != nil {
					delete(wc.Payload, gap)
					wc.Attrs = got[i].Attrs
				}
				if got[i].Name != wc.Name || got[i].Empty != wc.Empty || !reflect.DeepEqual(got[i].Attrs, wc.Attrs) ||
					(wc.Payload != nil && !samePayload(got[i].Payload, wc.Payload)) {
					t.Errorf("cookie %d: %+v, want %+v", i, got[i], wc)
				}
			}
			if db := dump(t, ctx, pool); !reflect.DeepEqual(db, w.DB) {
				for name := range w.DB {
					if !reflect.DeepEqual(db[name], w.DB[name]) {
						gj, _ := json.Marshal(db[name])
						wj, _ := json.Marshal(w.DB[name])
						t.Errorf("table %s:\n got  %s\n want %s", name, gj, wj)
					}
				}
			}
		})
	}
}

func testPool(t *testing.T) *pgxpool.Pool {
	t.Helper()
	base := os.Getenv("UBYHOST_TEST_DATABASE_URL")
	if base == "" {
		t.Skip("UBYHOST_TEST_DATABASE_URL is not set")
	}
	ctx := context.Background()
	name := fmt.Sprintf("ubyhost_go_webauth_%d", time.Now().UnixNano())
	admin, err := pgx.Connect(ctx, base)
	if err != nil {
		t.Fatal("admin connect failed")
	}
	if _, err := admin.Exec(ctx, "CREATE DATABASE "+name); err != nil {
		admin.Close(ctx)
		t.Fatal(err)
	}
	admin.Close(ctx)
	t.Cleanup(func() {
		c, err := pgx.Connect(context.Background(), base)
		if err != nil {
			return
		}
		defer c.Close(context.Background())
		_, _ = c.Exec(context.Background(), "DROP DATABASE IF EXISTS "+name+" WITH (FORCE)")
	})
	cfg, err := pgxpool.ParseConfig(base)
	if err != nil {
		t.Fatal("bad UBYHOST_TEST_DATABASE_URL")
	}
	cfg.ConnConfig.Database = name
	pool, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(pool.Close)
	files, err := filepath.Glob("../../../../App/migrations_pg/*.sql")
	if err != nil || len(files) == 0 {
		t.Fatal("no migrations found")
	}
	sort.Strings(files)
	for _, f := range files {
		raw, err := os.ReadFile(f)
		if err != nil {
			t.Fatal(err)
		}
		up := string(raw)
		if i := strings.Index(up, "-- +goose Up"); i >= 0 {
			up = up[i+len("-- +goose Up"):]
		}
		if i := strings.Index(up, "-- +goose Down"); i >= 0 {
			up = up[:i]
		}
		if _, err := pool.Exec(ctx, up); err != nil {
			t.Fatalf("%s: %v", filepath.Base(f), err)
		}
	}
	return pool
}
