def load_seed(seed, extra, now):
    with db.connect() as conn:
        for table in TABLES:
            conn.execute(f"DELETE FROM {table}")

def session_cookie(spec, now):
    if not spec:
        return ""
    return auth.issue_session(
        spec["uid"], spec.get("sv", 1),
        workspace_user_id=spec.get("as"),
        remember=bool(spec.get("rm")),
        impersonation_started_at=int(now + spec["ast"]) if "ast" in spec else None,
        issued_at=int(now + spec.get("iat", -60)),
    )


def parse_cookie(line, now):
    parts = [p.strip() for p in line.split(";")]
    name, _, value = parts[0].partition("=")
    attrs = {}
    for part in parts[1:]:
        key, _, val = part.partition("=")
        key = key.lower()
        if key == "expires":
            attrs[key] = "set"
        elif key in ("httponly", "secure"):
            attrs[key] = "true"
        else:
            attrs[key] = val.lower() if key == "samesite" else val
    out = {"name": name, "attrs": attrs, "empty": value in ("", '""')}
    if name == auth.SESSION_COOKIE and not out["empty"]:
        payload = URLSafeTimedSerializer(SECRET, salt="ubyhost-session").loads(value)
        for key in ("iat", "rf", "ast"):
            if key in payload:
                payload[key] = int(round(payload[key] - now))
        out["payload"] = payload
    return out


CSRF_TOKEN = URLSafeTimedSerializer(SECRET, salt="ubyhost-csrf").dumps({"nonce": security._nonce_binding(NONCE)})


def run(case, seed):
    now = time.time()
    load_seed(seed, case.get("seed", []), now)
    patched = case.get("py_config", {})
    saved = {k: getattr(config, k) for k in patched}
    for k, v in patched.items():
        setattr(config, k, v)
    try:
        cookies = []
        token = session_cookie(case.get("session"), now)
        if token:
            cookies.append(f"{auth.SESSION_COOKIE}={token}")
        if case.get("csrf", True):
            cookies.append(f"{security.CSRF_COOKIE}={NONCE}")
        if case.get("lang"):
            cookies.append(f"ubyhost_lang={case['lang']}")
        headers = {"cookie": "; ".join(cookies)} if cookies else {}
        headers.update(case.get("headers", {}))
        pairs = [tuple(p) for p in case.get("form", [])]
        if case.get("csrf", True) and case["method"] == "POST":
            pairs.append(("_csrf", CSRF_TOKEN))
        client = TestClient(app, base_url="http://testserver")
        url = case["path"] + (("?" + case["query"]) if case.get("query") else "")
        if case["method"] == "POST":
            headers["content-type"] = "application/x-www-form-urlencoded"
            response = client.post(url, content=urlencode(pairs), headers=headers, follow_redirects=False)
        else:
            response = client.get(url, headers=headers, follow_redirects=False)
        want = {
            "status": response.status_code,
            "location": response.headers.get("location"),
            "content_type": response.headers.get("content-type"),
            "cache_control": response.headers.get("cache-control"),
            "content_disposition": response.headers.get("content-disposition"),
            "body": None if 300 <= response.status_code < 400 else response.text,
            "cookies": [parse_cookie(line, now) for line in response.headers.get_list("set-cookie")],
            "db": dump(),
        }
    finally:
        for k, v in saved.items():
            setattr(config, k, v)
    case = dict(case)
    case["want"] = want
    return case

    OUT.write_text(json.dumps(out, ensure_ascii=False, indent=1, sort_keys=True) + "\n", encoding="utf-8")
    print(f"wrote {OUT.relative_to(REPO)}")
    catalog = PKG / "i18n.json"
    catalog.write_text(go_host_i18n.render(go_host_i18n.read_keys(PKG / "i18n_keys.txt")), encoding="utf-8")
    print(f"wrote {catalog.relative_to(REPO)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

package signup

import (
	"os"
	"regexp"
	"strings"
	"testing"
)

// TestWithdrawSQLBlanksIDColumns keeps the written-out withdraw statements in
// WithdrawConsent in step with IDColumns: each blanks exactly its platform's columns.
func TestWithdrawSQLBlanksIDColumns(t *testing.T) {
	raw, err := os.ReadFile("register.go")
	if err != nil {
		t.Fatal(err)
	}
	re := regexp.MustCompile("case (Google|Meta):\n\t\tupdate = `UPDATE ad_click SET ([^`]*) WHERE")
	found := re.FindAllStringSubmatch(string(raw), -1)
	if len(found) != len(IDColumns) {
		t.Fatalf("found %d withdraw statements, IDColumns has %d platforms", len(found), len(IDColumns))
	}
	for _, m := range found {
		platform := map[string]string{"Google": Google, "Meta": Meta}[m[1]]
		var blanked []string
		for _, part := range strings.Split(m[2], ", ") {
			if col, ok := strings.CutSuffix(part, " = NULL"); ok {
				blanked = append(blanked, col)
			}
		}
		if strings.Join(blanked, ",") != strings.Join(IDColumns[platform], ",") {
			t.Errorf("%s blanks %v, IDColumns says %v", platform, blanked, IDColumns[platform])
		}
	}
}
